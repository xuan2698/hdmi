def reference(vectors):
    enc_data = 0
    enc_valid = 0
    dec_data = 0
    dec_valid = 0
    out = []
    for idx, row in enumerate(vectors):
        rst = row.get('rst_in', 0)
        enc_valid_in = row.get('enc_valid_in', 0)
        enc_data_in = row.get('enc_data_in', 0)
        dec_valid_in = row.get('dec_valid_in', 0)
        dec_data_in = row.get('dec_data_in', 0)
        if rst:
            enc_data = 0
            enc_valid = 0
            dec_data = 0
            dec_valid = 0
        out.append({
            'enc_valid_out': enc_valid,
            'enc_data_out': enc_data,
            'dec_valid_out': dec_valid,
            'dec_data_out': dec_data
        })
        clk = row.get('clk_in') if isinstance(row, dict) else None
        if clk is not None:
            is_posedge = (clk == 1)
        else:
            is_posedge = (idx % 2 == 0)
        if is_posedge and not rst:
            enc_valid = enc_valid_in
            enc_val = 0
            for i in range(8):
                bit = (enc_data_in >> i) & 1
                pair = 1 if bit else 2
                enc_val |= pair << (2 * i)
            enc_data = enc_val

            dec_valid = dec_valid_in
            dec_val = 0
            for i in range(8):
                pair = (dec_data_in >> (2 * i)) & 3
                if pair == 1:
                    dec_val |= 1 << i
            dec_data = dec_val
    return out
