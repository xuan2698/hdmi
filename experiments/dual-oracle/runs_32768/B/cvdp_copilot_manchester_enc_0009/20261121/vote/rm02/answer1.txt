def reference(vectors):
    enc_valid_out = 0
    enc_data_out = 0
    dec_valid_out = 0
    dec_data_out = 0
    result = []

    for idx, row in enumerate(vectors):
        rst = row.get('rst_in', 0) & 1

        if rst:
            enc_valid_out = 0
            enc_data_out = 0
            dec_valid_out = 0
            dec_data_out = 0

        result.append({
            'enc_valid_out': enc_valid_out,
            'enc_data_out': enc_data_out,
            'dec_valid_out': dec_valid_out,
            'dec_data_out': dec_data_out,
        })

        edge = row.get('edge', row.get('clk_in', None))
        if isinstance(edge, str):
            is_posedge = 'pos' in edge.lower() or edge.lower() in ('1', 'rising', 'up')
        elif edge is None:
            is_posedge = (idx % 2 == 0)
        else:
            is_posedge = (int(edge) == 1)

        if is_posedge and not rst:
            enc_in = row.get('enc_data_in', 0) & 0xff
            dec_in = row.get('dec_data_in', 0) & 0xffff

            enc = 0
            for i in range(8):
                bit = (enc_in >> i) & 1
                pair = 1 if bit else 2
                enc |= pair << (2 * i)

            dec = 0
            for i in range(8):
                pair = (dec_in >> (2 * i)) & 3
                if pair == 1:
                    dec |= 1 << i

            enc_data_out = enc & 0xffff
            enc_valid_out = row.get('enc_valid_in', 0) & 1
            dec_data_out = dec & 0xff
            dec_valid_out = row.get('dec_valid_in', 0) & 1

    return result
