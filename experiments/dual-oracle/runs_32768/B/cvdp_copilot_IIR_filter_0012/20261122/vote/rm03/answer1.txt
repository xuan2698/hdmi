def reference(vectors):
    def s16(x):
        x = x & 0xFFFF
        if x >= 0x8000:
            x -= 0x10000
        return x

    def clamp16(x):
        if x > 32767:
            return 32767
        if x < -32768:
            return -32768
        return x

    def u16(x):
        return x & 0xFFFF

    y1 = y2 = y3 = 0
    out = []

    for idx, row in enumerate(vectors):
        res = row.get('reset', 0)
        if res:
            y1 = y2 = y3 = 0

        out.append({
            'out_1': u16(y1),
            'out_2': u16(y2),
            'out_3': u16(y3)
        })

        if 'clk' in row:
            clk = row['clk']
        else:
            clk = 1 if idx % 2 == 0 else 0

        if clk == 1:
            if res:
                y1 = y2 = y3 = 0
            else:
                k = row.get('filter_select', 0)
                x1 = s16(row.get('in_1', 0))
                x2 = s16(row.get('in_2', 0))
                x3 = s16(row.get('in_3', 0))

                y1 = clamp16(y1 + ((x1 - y1) >> k))
                y2 = clamp16(y2 + ((x2 - y2) >> k))
                y3 = clamp16(y3 + ((x3 - y3) >> k))

    return out
