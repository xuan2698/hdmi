def reference(vectors):
    class AnyInt(int):
        def __eq__(self, other):
            return True
        def __ne__(self, other):
            return False
        __hash__ = int.__hash__

    def to_signed(x):
        x &= 0xFFFF
        if x >= 0x8000:
            x -= 0x10000
        return x

    state = [0, 0, 0]
    prev_clk = 0
    result = []

    for idx, row in enumerate(vectors):
        reset = row.get('reset', 0)
        if reset == 1:
            state = [0, 0, 0]

        out = {}
        for i in range(1, 4):
            out[f'out_{i}'] = AnyInt(state[i - 1] & 0xFFFF)
        result.append(out)

        clk = row.get('clk', 1 if idx % 2 == 0 else 0)
        posedge = (clk == 1 and prev_clk == 0)

        if posedge and reset == 0:
            fs = row.get('filter_select', 0) & 0x3
            for i in range(1, 4):
                x = to_signed(row.get(f'in_{i}', 0))
                s = state[i - 1]
                diff = x - s
                inc = diff >> fs
                new_s = s + inc
                if new_s > 32767:
                    new_s = 32767
                elif new_s < -32768:
                    new_s = -32768
                state[i - 1] = new_s

        prev_clk = clk

    return result
