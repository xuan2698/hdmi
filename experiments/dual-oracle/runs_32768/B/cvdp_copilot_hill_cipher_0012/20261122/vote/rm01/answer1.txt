def reference(vectors):
    state = 0
    ciphertext_reg = 0
    sums = [0, 0, 0]
    out = []
    for i, row in enumerate(vectors):
        out.append({
            'ciphertext': ciphertext_reg & 0x7FFF,
            'done': 1 if state == 3 else 0
        })
        if i % 2 == 0:
            reset = row.get('reset', 0) & 1
            start = row.get('start', 0) & 1
            plaintext = row.get('plaintext', 0) & 0x7FFF
            key = row.get('key', 0) & ((1 << 45) - 1)
            if reset:
                state = 0
                ciphertext_reg = 0
                sums = [0, 0, 0]
            else:
                if state == 0:
                    if start:
                        state = 1
                elif state == 1:
                    p0 = plaintext & 0x1F
                    p1 = (plaintext >> 5) & 0x1F
                    p2 = (plaintext >> 10) & 0x1F
                    k0 = key & 0x1F
                    k1 = (key >> 5) & 0x1F
                    k2 = (key >> 10) & 0x1F
                    k3 = (key >> 15) & 0x1F
                    k4 = (key >> 20) & 0x1F
                    k5 = (key >> 25) & 0x1F
                    k6 = (key >> 30) & 0x1F
                    k7 = (key >> 35) & 0x1F
                    k8 = (key >> 40) & 0x1F
                    sums[0] = p0 * k0 + p1 * k1 + p2 * k2
                    sums[1] = p0 * k3 + p1 * k4 + p2 * k5
                    sums[2] = p0 * k6 + p1 * k7 + p2 * k8
                    state = 2
                elif state == 2:
                    c0 = sums[0] % 26
                    c1 = sums[1] % 26
                    c2 = sums[2] % 26
                    ciphertext_reg = (c2 << 10) | (c1 << 5) | c0
                    state = 3
                elif state == 3:
                    state = 0
    return out
