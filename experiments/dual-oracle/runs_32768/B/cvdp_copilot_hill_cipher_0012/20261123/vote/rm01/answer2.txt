def reference(vectors):
    state = "IDLE"
    cipher = 0
    latched_p = 0
    latched_k = 0
    outs = []
    expect_pos = 1
    for row in vectors:
        try:
            reset = row["reset"]
        except:
            try:
                reset = row.get("reset", 0)
            except:
                reset = 0
        try:
            start = row["start"]
        except:
            try:
                start = row.get("start", 0)
            except:
                start = 0
        try:
            plaintext = row["plaintext"]
        except:
            try:
                plaintext = row.get("plaintext", 0)
            except:
                plaintext = 0
        try:
            key = row["key"]
        except:
            try:
                key = row.get("key", 0)
            except:
                key = 0
        e = None
        try:
            e = row["edge"]
        except:
            try:
                e = row.get("edge", None)
            except:
                e = None
        if e is None:
            try:
                e = row["clk"]
            except:
                try:
                    e = row.get("clk", None)
                except:
                    e = None
        is_pos = expect_pos
        if e is not None:
            if e == "posedge" or e == "pos" or e == 1 or e == "1":
                is_pos = 1
            elif e == "negedge" or e == "neg" or e == 0 or e == "0":
                is_pos = 0
            else:
                is_pos = expect_pos
        outs.append({"ciphertext": cipher & 32767, "done": 1 if state == "DONE" else 0})
        if is_pos:
            if reset:
                state = "IDLE"
                cipher = 0
                latched_p = 0
                latched_k = 0
            else:
                if state == "IDLE":
                    if start:
                        state = "COMPUTE"
                        latched_p = plaintext
                        latched_k = key
                elif state == "COMPUTE":
                    state = "COMPUTE_MOD"
                elif state == "COMPUTE_MOD":
                    p0 = latched_p & 31
                    p1 = (latched_p >> 5) & 31
                    p2 = (latched_p >> 10) & 31
                    k0 = latched_k & 31
                    k1 = (latched_k >> 5) & 31
                    k2 = (latched_k >> 10) & 31
                    k3 = (latched_k >> 15) & 31
                    k4 = (latched_k >> 20) & 31
                    k5 = (latched_k >> 25) & 31
                    k6 = (latched_k >> 30) & 31
                    k7 = (latched_k >> 35) & 31
                    k8 = (latched_k >> 40) & 31
                    c0 = (p0 * k0 + p1 * k1 + p2 * k2) % 26
                    c1 = (p0 * k3 + p1 * k4 + p2 * k5) % 26
                    c2 = (p0 * k6 + p1 * k7 + p2 * k8) % 26
                    cipher = c0 | (c1 << 5) | (c2 << 10)
                    state = "DONE"
                elif state == "DONE":
                    state = "IDLE"
        if expect_pos:
            expect_pos = 0
        else:
            expect_pos = 1
    return outs
