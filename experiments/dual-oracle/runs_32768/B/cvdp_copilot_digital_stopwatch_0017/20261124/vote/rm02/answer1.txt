def reference(vectors):
    counter = 0
    seconds = 0
    minutes = 0
    hour = 0
    out = []
    for i, v in enumerate(vectors):
        if isinstance(v, dict):
            reset = v['reset']
            start_stop = v['start_stop']
        else:
            reset = v[0]
            start_stop = v[1]
        if reset == 1:
            counter = 0
            seconds = 0
            minutes = 0
            hour = 0
        out.append({'seconds': seconds, 'minutes': minutes, 'hour': hour})
        if i % 2 == 0:  # posedge
            if reset == 0 and start_stop == 1:
                if counter == 3:  # CLK_FREQ - 1
                    counter = 0
                    if seconds == 59:
                        seconds = 0
                        if minutes == 59:
                            minutes = 0
                            hour = 1
                        else:
                            minutes += 1
                    else:
                        seconds += 1
                else:
                    counter += 1
    return out
