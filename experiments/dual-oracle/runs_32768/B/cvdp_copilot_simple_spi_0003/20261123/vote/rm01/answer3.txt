def reference(vectors):
    state = 0
    cs_b = 1
    clk = 0
    data = 0
    bits_left = 16
    done = 0
    shift_reg = 0
    prev_clk = 0
    outputs = []
    for row in vectors:
        if isinstance(row, dict):
            if 'edge' in row:
                e = row['edge']
                if isinstance(e, str):
                    e = e.lower()
                    is_pos = e.startswith('p') or e == '1'
                else:
                    is_pos = (e == 1)
                i_rst_b = row.get('i_rst_b', 1)
                i_data_in = row.get('i_data_in', 0)
                i_enable = row.get('i_enable', 0)
                i_fault = row.get('i_fault', 0)
                i_clear = row.get('i_clear', 0)
            elif 'i_clk' in row:
                cur_clk = row['i_clk']
                is_pos = (cur_clk == 1 and prev_clk == 0)
                prev_clk = cur_clk
                i_rst_b = row.get('i_rst_b', 1)
                i_data_in = row.get('i_data_in', 0)
                i_enable = row.get('i_enable', 0)
                i_fault = row.get('i_fault', 0)
                i_clear = row.get('i_clear', 0)
            else:
                is_pos = True
                i_rst_b = row.get('i_rst_b', 1)
                i_data_in = row.get('i_data_in', 0)
                i_enable = row.get('i_enable', 0)
                i_fault = row.get('i_fault', 0)
                i_clear = row.get('i_clear', 0)
        else:
            if len(row) >= 6:
                e = row[0]
                if isinstance(e, str):
                    e = e.lower()
                    is_pos = e.startswith('p') or e == '1'
                else:
                    is_pos = (e == 1)
                i_rst_b, i_data_in, i_enable, i_fault, i_clear = row[1:6]
            elif len(row) == 5:
                is_pos = (len(outputs) % 2 == 0)
                i_rst_b, i_data_in, i_enable, i_fault, i_clear = row
            else:
                is_pos = True
                i_rst_b = row[0] if len(row)>0 else 1
                i_data_in = row[1] if len(row)>1 else 0
                i_enable = row[2] if len(row)>2 else 0
                i_fault = row[3] if len(row)>3 else 0
                i_clear = row[4] if len(row)>4 else 0
        i_rst_b = int(i_rst_b) & 1
        i_data_in = int(i_data_in) & 0xFFFF
        i_enable = int(i_enable) & 1
        i_fault = int(i_fault) & 1
        i_clear = int(i_clear) & 1

        if i_rst_b == 0:
            state = 0
            cs_b = 1
            clk = 0
            data = 0
            bits_left = 16
            done = 0
            shift_reg = 0

        outputs.append({
            'o_spi_cs_b': cs_b,
            'o_spi_clk': clk,
            'o_spi_data': data,
            'o_bits_left': bits_left,
            'o_done': done,
            'o_fsm_state': state
        })

        if is_pos and i_rst_b != 0:
            n_state = state
            n_cs_b = cs_b
            n_clk = clk
            n_data = data
            n_bits_left = bits_left
            n_done = done
            n_shift_reg = shift_reg
            if i_clear == 1:
                n_state = 0
                n_cs_b = 1
                n_clk = 0
                n_data = 0
                n_bits_left = 16
                n_done = 0
                n_shift_reg = 0
            elif i_fault == 1:
                n_state = 3
                n_cs_b = 1
                n_clk = 0
                n_data = 0
                n_bits_left = 16
                n_done = 0
            elif state == 3:
                n_state = 3
                n_cs_b = 1
                n_clk = 0
                n_data = 0
                n_bits_left = 16
                n_done = 0
            elif i_enable == 0:
                n_state = 0
                n_cs_b = 1
                n_clk = 0
                n_data = 0
                n_bits_left = 16
                n_done = 0
                n_shift_reg = 0
            else:
                if state == 0:
                    n_cs_b = 1
                    n_clk = 0
                    n_data = 0
                    n_bits_left = 16
                    n_done = 0
                    if i_enable == 1:
                        n_state = 1
                        n_shift_reg = i_data_in
                    else:
                        n_state = 0
                elif state == 1:
                    n_cs_b = 0
                    n_data = (shift_reg >> 15) & 1
                    n_shift_reg = (shift_reg << 1) & 0xFFFF
                    n_bits_left = bits_left
                    n_done = 0
                    n_state = 2
                elif state == 2:
                    n_clk = 1 - clk
                    n_bits_left = bits_left - 1
                    n_cs_b = cs_b
                    n_data = data
                    n_shift_reg = shift_reg
                    if bits_left == 1:
                        n_state = 0
                        n_done = 1
                        n_cs_b = 1
                        n_clk = 0
                        n_data = 0
                        n_bits_left = 16
                    else:
                        n_state = 1
                        n_done = 0
                elif state == 3:
                    n_state = 3
                    n_cs_b = 1
                    n_clk = 0
                    n_data = 0
                    n_bits_left = 16
                    n_done = 0
            state = n_state
            cs_b = n_cs_b
            clk = n_clk
            data = n_data
            bits_left = n_bits_left
            done = n_done
            shift_reg = n_shift_reg
    return outputs
