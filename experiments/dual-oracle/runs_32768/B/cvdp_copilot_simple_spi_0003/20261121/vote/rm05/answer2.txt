def reference(vectors):
    state = 0
    bits_left = 16
    shift_reg = 0
    cs_b = 1
    spi_clk = 0
    spi_data = 0
    done = 0
    prev_clk = 0
    out = []
    for idx, row in enumerate(vectors):
        if isinstance(row, dict):
            rst = row.get('i_rst_b', row.get('rst_b', 1))
            data = row.get('i_data_in', row.get('data_in', 0))
            en = row.get('i_enable', row.get('enable', 0))
            fault = row.get('i_fault', row.get('fault', 0))
            clear = row.get('i_clear', row.get('clear', 0))
            clk = row.get('i_clk', row.get('clk', None))
            if 'edge' in row:
                e = row['edge']
                if isinstance(e, str):
                    pos = (e.lower() == 'posedge')
                else:
                    pos = (e == 1) if e in (0, 1) else (idx % 2 == 0)
            elif clk is not None:
                pos = (clk == 1 and prev_clk == 0)
                prev_clk = clk
            else:
                pos = (idx % 2 == 0)
        else:
            if len(row) == 5:
                rst, data, en, fault, clear = row
                pos = (idx % 2 == 0)
            elif len(row) == 6:
                clk, rst, data, en, fault, clear = row
                pos = (clk == 1 and prev_clk == 0)
                prev_clk = clk
            else:
                rst = data = en = fault = clear = 0
                pos = (idx % 2 == 0)
        out.append({
            'o_spi_cs_b': int(cs_b),
            'o_spi_clk': int(spi_clk),
            'o_spi_data': int(spi_data),
            'o_bits_left': int(bits_left & 0x1F),
            'o_done': int(done),
            'o_fsm_state': int(state & 0x3)
        })
        if rst == 0:
            state = 0
            bits_left = 16
            shift_reg = 0
            cs_b = 1
            spi_clk = 0
            spi_data = 0
            done = 0
        elif pos:
            if clear:
                state = 0
                bits_left = 16
                shift_reg = 0
                cs_b = 1
                spi_clk = 0
                spi_data = 0
                done = 0
            elif fault:
                if state != 3:
                    done = 1
                else:
                    done = 0
                state = 3
                bits_left = 16
                shift_reg = 0
                cs_b = 1
                spi_clk = 0
                spi_data = 0
            elif en == 0:
                state = 0
                bits_left = 16
                shift_reg = 0
                cs_b = 1
                spi_clk = 0
                spi_data = 0
                done = 0
            else:
                if state == 0:
                    cs_b = 1
                    spi_clk = 0
                    spi_data = 0
                    done = 0
                    bits_left = 16
                    state = 1
                    cs_b = 0
                    shift_reg = data & 0xFFFF
                elif state == 1:
                    cs_b = 0
                    spi_data = (shift_reg >> 15) & 1
                    state = 2
                    done = 0
                elif state == 2:
                    spi_clk = 1 - spi_clk
                    shift_reg = (shift_reg << 1) & 0xFFFF
                    if bits_left == 1:
                        bits_left = 0
                        done = 1
                        state = 0
                        cs_b = 1
                    else:
                        bits_left = (bits_left - 1) & 0x1F
                        done = 0
                        state = 1
                elif state == 3:
                    cs_b = 1
                    spi_clk = 0
                    spi_data = 0
                    done = 0
                    bits_left = 16
    return out
