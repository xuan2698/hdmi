def reference(vectors):
    state = 0
    out = []
    for i, v in enumerate(vectors):
        rst = v.get('rst', 0)
        if rst:
            state = 0
            tr = 1
            to = 0
            dout = 0
        else:
            tr = 1 << (state & 3)
            to = ((tr << 1) | (tr >> 3)) & 0xF
            dout = (v.get('data_in', 0) & 0xF) if v.get('has_data_to_send', 0) else 0

        out.append({'data_out': dout, 'token_received': tr, 'token': to})

        pos = False
        if 'edge' in v:
            e = v['edge']
            if isinstance(e, str):
                pos = e.lower().startswith('pos') or e.lower() == '1'
            else:
                pos = e == 1
        elif 'clk' in v:
            c = v['clk']
            if isinstance(c, str):
                pos = ('pos' in c.lower()) or c == '1'
            else:
                pos = c == 1
        else:
            pos = (i % 2 == 0)

        if pos and not rst:
            state = (state + 1) & 3

    return out
