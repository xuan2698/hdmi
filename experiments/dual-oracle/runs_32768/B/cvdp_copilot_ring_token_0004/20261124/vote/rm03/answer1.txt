def reference(vectors):
    state = 1
    token_reg = 0
    data_reg = 0
    outputs = []

    for idx, row in enumerate(vectors):
        if isinstance(row, dict):
            rst = row.get('rst', 0)
            data_in = row.get('data_in', 0)
            has_data = row.get('has_data_to_send', 0)

            ed = row.get('edge', None)
            if ed is not None:
                if isinstance(ed, str):
                    is_posedge = ('pos' in ed.lower())
                else:
                    is_posedge = (ed == 1 or ed is True)
            else:
                clk = row.get('clk', None)
                if clk is not None:
                    is_posedge = (clk == 1)
                else:
                    is_posedge = (idx % 2 == 0)
        else:
            if len(row) == 4:
                rst, data_in, has_data, clk = row
                if isinstance(clk, str):
                    is_posedge = ('pos' in clk.lower())
                else:
                    is_posedge = (clk == 1)
            elif len(row) == 3:
                rst, data_in, has_data = row
                is_posedge = (idx % 2 == 0)
            else:
                ed = row[0] if len(row) > 0 else 1
                rst = row[1] if len(row) > 1 else 0
                data_in = row[2] if len(row) > 2 else 0
                has_data = row[3] if len(row) > 3 else 0
                if isinstance(ed, str):
                    is_posedge = ('pos' in ed.lower())
                else:
                    is_posedge = (ed == 1)

        if rst:
            state = 1
            token_reg = 0
            data_reg = 0

        outputs.append({
            'data_out': data_reg & 0xF,
            'token_received': state & 0xF,
            'token': token_reg & 0xF
        })

        if is_posedge and not rst:
            if state in (1, 2, 4, 8):
                if state == 1:
                    next_state = 2
                elif state == 2:
                    next_state = 4
                elif state == 4:
                    next_state = 8
                else:
                    next_state = 1
                next_token = state
                next_data = (data_in & 0xF) if has_data else 0
            else:
                next_state = 1
                next_token = 0
                next_data = 0

            state = next_state
            token_reg = next_token
            data_reg = next_data

    return outputs
