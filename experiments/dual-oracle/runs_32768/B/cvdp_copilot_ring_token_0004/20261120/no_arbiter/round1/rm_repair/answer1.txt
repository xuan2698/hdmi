def reference(vectors):
    state = 1
    prev_clk = 0
    res = []
    for i, v in enumerate(vectors):
        rst = v.get('rst', 0)
        data_in = v.get('data_in', 0) & 0xF
        has = v.get('has_data_to_send', 0) & 1

        posedge = None
        e = v.get('edge')
        if isinstance(e, str):
            posedge = e.lower().startswith('p')
        elif e is not None:
            posedge = bool(e)

        if posedge is None:
            if 'clk' in v:
                clk = v['clk']
                posedge = (clk == 1 and prev_clk == 0)
            else:
                posedge = (i % 2 == 0)

        if 'clk' in v:
            prev_clk = v['clk']

        if rst:
            state = 1

        if rst:
            data_out = 0
            token_received = 1
            token = 0
        else:
            token_received = state & 0xF
            if state in (1, 2, 4, 8):
                token = ((state << 1) & 0xF) | (state >> 3)
                data_out = data_in if has else 0
            else:
                token = 0
                data_out = 0

        res.append({
            'data_out': data_out & 0xF,
            'token_received': token_received & 0xF,
            'token': token & 0xF
        })

        if posedge and not rst:
            if state in (1, 2, 4, 8):
                state = ((state << 1) & 0xF) | (state >> 3)
            else:
                state = 1

    return res
