def reference(vectors):
    W = 8
    MASK = (1 << W) - 1
    HALF = 1 << (W - 1)

    def to_signed(v):
        v &= MASK
        return v - (1 << W) if v >= HALF else v

    state = 0  # 0: IDLE, 1: LOAD, 2: COMPUTE, 3: OUTPUT
    a_reg = 0
    b_reg = 0
    result_reg = 0
    overflow_reg = 0
    out = []
    prev_clk = None

    for idx, row in enumerate(vectors):
        rst_n = row.get('i_rst_n', 1)
        enable = row.get('i_enable', 0)
        start = row.get('i_start', 0)
        mode = row.get('i_mode', 0)
        clear = row.get('i_clear', 0)
        op_a = row.get('i_operand_a', 0) & MASK
        op_b = row.get('i_operand_b', 0) & MASK

        # Determine edge type
        if 'edge' in row:
            e = row['edge']
            if isinstance(e, str):
                es = e.lower()
                edge = 'posedge' if ('pos' in es or es in ('1', 'p', 'rise')) else 'negedge'
            else:
                edge = 'posedge' if e in (1, True) else 'negedge'
        elif 'i_clk' in row:
            edge = 'posedge' if row['i_clk'] == 1 else 'negedge'
        else:
            edge = 'posedge' if (idx % 2 == 0) else 'negedge'

        # Asynchronous reset applied before output sampling
        if rst_n == 0:
            state = 0
            a_reg = 0
            b_reg = 0
            result_reg = 0
            overflow_reg = 0

        # Sample outputs before registered NBA updates
        out.append({
            'o_resultant_sum': result_reg & MASK,
            'o_overflow': overflow_reg & 1,
            'o_ready': 1 if state == 3 else 0,
            'o_status': state & 3,
        })

        # Active reset prevents synchronous updates
        if rst_n == 0:
            if 'i_clk' in row:
                prev_clk = row['i_clk']
            continue

        # Synchronous updates at posedge
        if edge == 'posedge':
            if clear:
                state = 0
                a_reg = 0
                b_reg = 0
                result_reg = 0
                overflow_reg = 0
            else:
                if state == 0:  # IDLE
                    if enable and start:
                        state = 1  # LOAD
                elif state == 1:  # LOAD
                    a_reg = op_a
                    b_reg = op_b
                    state = 2  # COMPUTE
                elif state == 2:  # COMPUTE
                    sa = to_signed(a_reg)
                    sb = to_signed(b_reg)
                    full = sa + sb if mode == 0 else sa - sb
                    result_reg = full & MASK
                    overflow_reg = 1 if (full < -HALF or full > HALF - 1) else 0
                    state = 3  # OUTPUT
                elif state == 3:  # OUTPUT
                    state = 0  # IDLE

        if 'i_clk' in row:
            prev_clk = row['i_clk']

    return out
