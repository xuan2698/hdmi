def reference(vectors):
    cnt = 0x1F
    out = []
    next_pos = True
    for row in vectors:
        if isinstance(row, dict):
            r = row.get('reset', 0)
            e = row.get('edge', None)
        elif isinstance(row, (list, tuple)):
            r = row[-1] if len(row) else 0
            e = None
            for v in row:
                if isinstance(v, str) and v in ('posedge', 'negedge'):
                    e = v
                    break
        else:
            r = row
            e = None

        if e is None:
            edge = 'posedge' if next_pos else 'negedge'
            next_pos = not next_pos
        else:
            edge = e
            next_pos = (edge != 'posedge')

        if r:
            cnt = 0x1F

        gate = 1 if 0 < cnt < 0xE else 0
        rel = 1 if cnt < 0x8 else 0

        out.append({
            'cnt_q': cnt & 0x1F,
            'gate_clk_o': gate,
            'release_reset_o': rel
        })

        if edge == 'posedge' and not r and cnt > 0:
            cnt = (cnt - 1) & 0x1F

    return out
