def reference(vectors):
    out = []
    state = 0
    for row in vectors:
        if row.get('areset', 0):
            state = 0
        else:
            x = row.get('x', 0) & 1
            if state == 0:
                if x:
                    state = 1
            elif state == 1:
                if x:
                    state = 2
            else:
                if not x:
                    state = 1
        out.append({'z': 1 if state == 1 else 0})
    return out
