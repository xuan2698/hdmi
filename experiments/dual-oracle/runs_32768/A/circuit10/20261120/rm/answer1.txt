def reference(vectors):
    state = 0
    prev_clk = 0
    out = []
    for row in vectors:
        clk = row.get('clk', 0)
        a = row.get('a', 0)
        b = row.get('b', 0)
        if prev_clk == 0 and clk == 1:
            state = (a & b) | (a & state) | (b & state)
        q = a ^ b ^ state
        out.append({'q': q, 'state': state})
        prev_clk = clk
    return out
