def reference(vectors):
    q = 0
    out = []
    for row in vectors:
        r = row['reset'] if isinstance(row, dict) else row[0]
        if r:
            q = 0
        else:
            dec = (q & 15) + ((q >> 4) & 15) * 10 + ((q >> 8) & 15) * 100 + ((q >> 12) & 15) * 1000
            dec = (dec + 1) % 10000
            d0 = dec % 10
            d1 = (dec // 10) % 10
            d2 = (dec // 100) % 10
            d3 = (dec // 1000) % 10
            q = d0 | (d1 << 4) | (d2 << 8) | (d3 << 12)
        d0 = q & 15
        d1 = (q >> 4) & 15
        d2 = (q >> 8) & 15
        e1 = 1 if d0 == 9 else 0
        e2 = 1 if d0 == 9 and d1 == 9 else 0
        e3 = 1 if d0 == 9 and d1 == 9 and d2 == 9 else 0
        ena = e1 | (e2 << 1) | (e3 << 2)
        out.append({'ena': ena, 'q': q})
    return out
