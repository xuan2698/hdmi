def reference(vectors):
    q = 0
    out = []
    for row in vectors:
        if isinstance(row, dict):
            reset = row.get('reset', 0)
        elif isinstance(row, (list, tuple)):
            reset = row[0] if row else 0
        else:
            reset = row

        if reset & 1:
            q = 0
        else:
            d0 = q & 15
            d1 = (q >> 4) & 15
            d2 = (q >> 8) & 15
            d3 = (q >> 12) & 15
            d0 += 1
            if d0 == 10:
                d0 = 0
                d1 += 1
                if d1 == 10:
                    d1 = 0
                    d2 += 1
                    if d2 == 10:
                        d2 = 0
                        d3 += 1
                        if d3 == 10:
                            d3 = 0
            q = (d3 << 12) | (d2 << 8) | (d1 << 4) | d0

        d0 = q & 15
        d1 = (q >> 4) & 15
        d2 = (q >> 8) & 15
        ena = 0
        if d0 == 9:
            ena |= 1
            if d1 == 9:
                ena |= 2
                if d2 == 9:
                    ena |= 4

        out.append({'ena': ena, 'q': q})

    return out
