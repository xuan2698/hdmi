def reference(vectors):
    q = 0
    out = []
    for row in vectors:
        reset = row['reset'] if isinstance(row, dict) else row[0]
        if reset:
            q = 0
        else:
            ones = q & 0xF
            tens = (q >> 4) & 0xF
            hundreds = (q >> 8) & 0xF
            thousands = (q >> 12) & 0xF
            if ones < 9:
                ones += 1
            else:
                ones = 0
                if tens < 9:
                    tens += 1
                else:
                    tens = 0
                    if hundreds < 9:
                        hundreds += 1
                    else:
                        hundreds = 0
                        if thousands < 9:
                            thousands += 1
                        else:
                            thousands = 0
            q = ones | (tens << 4) | (hundreds << 8) | (thousands << 12)
        
        ones = q & 0xF
        tens = (q >> 4) & 0xF
        hundreds = (q >> 8) & 0xF
        ena = (1 if ones == 9 else 0) | \
              ((1 if (ones == 9 and tens == 9) else 0) << 1) | \
              ((1 if (ones == 9 and tens == 9 and hundreds == 9) else 0) << 2)
        out.append({'ena': ena, 'q': q})
    return out
