def reference(vectors):
    q = 0
    result = []
    for row in vectors:
        if isinstance(row, dict):
            reset = row.get("reset", 0)
        else:
            reset = row[0] if isinstance(row, (list, tuple)) else row
        if reset:
            q = 0
        else:
            q = (q + 1) % 10
        result.append({"q": q})
    return result
