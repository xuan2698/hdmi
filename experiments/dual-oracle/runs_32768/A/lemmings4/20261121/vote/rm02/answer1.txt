def reference(vectors):
    state = 0  # 0=left, 1=right, 2=fall_left, 3=fall_right, 4=dig_left, 5=dig_right, 6=splat
    fall_count = 0
    result = []

    for row in vectors:
        if isinstance(row, dict):
            areset = int(row.get('areset', 0))
            bump_left = int(row.get('bump_left', 0))
            bump_right = int(row.get('bump_right', 0))
            ground = int(row.get('ground', 0))
            dig = int(row.get('dig', 0))
        else:
            areset, bump_left, bump_right, ground, dig = [int(x) for x in row[:5]]

        if areset:
            state = 0
            fall_count = 0
        elif state == 0:
            if ground == 0:
                state = 2
                fall_count = 1
            elif dig:
                state = 4
            elif bump_left or bump_right:
                state = 1
        elif state == 1:
            if ground == 0:
                state = 3
                fall_count = 1
            elif dig:
                state = 5
            elif bump_left or bump_right:
                state = 0
        elif state == 2:
            if ground == 0:
                fall_count += 1
            else:
                if fall_count > 20:
                    state = 6
                else:
                    state = 0
                fall_count = 0
        elif state == 3:
            if ground == 0:
                fall_count += 1
            else:
                if fall_count > 20:
                    state = 6
                else:
                    state = 1
                fall_count = 0
        elif state == 4:
            if ground == 0:
                state = 2
                fall_count = 1
        elif state == 5:
            if ground == 0:
                state = 3
                fall_count = 1
        elif state == 6:
            state = 6
            fall_count = 0

        result.append({
            'walk_left': 1 if state == 0 else 0,
            'walk_right': 1 if state == 1 else 0,
            'aaah': 1 if state == 2 or state == 3 else 0,
            'digging': 1 if state == 4 or state == 5 else 0
        })

    return result
