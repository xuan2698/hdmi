def reference(vectors):
    state = 0
    cnt = 0
    ans = []
    for row in vectors:
        if isinstance(row, dict):
            areset = row.get('areset', 0)
            bump_left = row.get('bump_left', 0)
            bump_right = row.get('bump_right', 0)
            ground = row.get('ground', 1)
            dig = row.get('dig', 0)
        else:
            areset, bump_left, bump_right, ground, dig = row

        if areset:
            state = 0
            cnt = 0
        elif state == 0:
            if ground == 0:
                state = 2
                cnt = 1
            elif dig == 1:
                state = 4
            elif bump_left == 1:
                state = 1
        elif state == 1:
            if ground == 0:
                state = 3
                cnt = 1
            elif dig == 1:
                state = 5
            elif bump_right == 1:
                state = 0
        elif state == 2:
            if ground == 1:
                if cnt > 20:
                    state = 6
                else:
                    state = 0
                cnt = 0
            else:
                cnt += 1
        elif state == 3:
            if ground == 1:
                if cnt > 20:
                    state = 6
                else:
                    state = 1
                cnt = 0
            else:
                cnt += 1
        elif state == 4:
            if ground == 0:
                state = 2
                cnt = 1
        elif state == 5:
            if ground == 0:
                state = 3
                cnt = 1

        ans.append({
            'walk_left': 1 if state == 0 else 0,
            'walk_right': 1 if state == 1 else 0,
            'aaah': 1 if state == 2 or state == 3 else 0,
            'digging': 1 if state == 4 or state == 5 else 0
        })
    return ans
