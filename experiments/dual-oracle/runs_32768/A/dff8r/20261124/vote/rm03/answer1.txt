def reference(vectors):
    q = 0
    result = []
    for row in vectors:
        if isinstance(row, dict):
            d = row.get('d', 0)
            reset = row.get('reset', 0)
        else:
            d = row[0] if len(row) > 0 else 0
            reset = row[1] if len(row) > 1 else 0
        if reset:
            q = 0
        else:
            q = int(d) & 0xFF
        result.append({'q': q})
    return result
