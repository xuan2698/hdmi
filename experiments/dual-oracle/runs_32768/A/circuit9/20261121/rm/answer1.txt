def reference(vectors):
    q = 0
    prev_clk = 0
    out = []
    for row in vectors:
        clk = row['clk']
        if prev_clk == 0 and clk == 1:
            a = row['a']
            if a == 1:
                q = 4
            else:
                if q == 6:
                    q = 0
                else:
                    q += 1
        out.append({'q': q})
        prev_clk = clk
    return out
