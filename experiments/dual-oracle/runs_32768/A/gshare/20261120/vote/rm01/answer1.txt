def reference(vectors):
    pht = [0] * 128
    hist = 0
    result = []

    for row in vectors:
        areset = row.get('areset', 0)
        if areset:
            pht = [0] * 128
            hist = 0
            pc = row.get('predict_pc', 0) & 0x7f
            idx = (pc ^ hist) & 0x7f
            taken = (pht[idx] >> 1) & 1
            result.append({
                'predict_taken': taken,
                'predict_history': hist
            })
            continue

        pv = row.get('predict_valid', 0)
        ppc = row.get('predict_pc', 0) & 0x7f
        tv = row.get('train_valid', 0)
        tt = row.get('train_taken', 0) & 1
        tm = row.get('train_mispredicted', 0) & 1
        th = row.get('train_history', 0) & 0x7f
        tpc = row.get('train_pc', 0) & 0x7f

        idx = (ppc ^ hist) & 0x7f
        taken = (pht[idx] >> 1) & 1
        pred_hist = hist

        next_hist = hist
        if tv and tm:
            next_hist = ((th << 1) & 0x7f) | tt
        elif pv:
            next_hist = ((hist << 1) & 0x7f) | taken

        if tv:
            tidx = (tpc ^ th) & 0x7f
            c = pht[tidx]
            if tt:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            pht[tidx] = c

        hist = next_hist
        result.append({
            'predict_taken': taken,
            'predict_history': pred_hist
        })

    return result
