def reference(vectors):
    state = 0
    outputs = []
    for row in vectors:
        reset = row['reset'] & 1
        data = row['data'] & 1
        if reset:
            state = 0
        else:
            if state == 0:
                state = 1 if data else 0
            elif state == 1:
                state = 2 if data else 0
            elif state == 2:
                state = 3 if data == 0 else 2
            elif state == 3:
                state = 4 if data else 0
            elif state == 4:
                state = 4
        outputs.append({'start_shifting': 1 if state == 4 else 0})
    return outputs
