def reference(vectors):
    hour = 12
    minute = 0
    second = 0
    pm = 0
    out = []
    for row in vectors:
        reset = row.get('reset', 0)
        ena = row.get('ena', 0)
        if reset:
            hour = 12
            minute = 0
            second = 0
            pm = 0
        elif ena:
            second += 1
            if second == 60:
                second = 0
                minute += 1
                if minute == 60:
                    minute = 0
                    if hour == 11:
                        hour = 12
                        pm ^= 1
                    elif hour == 12:
                        hour = 1
                    else:
                        hour += 1
        hh = ((hour // 10) << 4) | (hour % 10)
        mm = ((minute // 10) << 4) | (minute % 10)
        ss = ((second // 10) << 4) | (second % 10)
        out.append({'pm': pm, 'hh': hh, 'mm': mm, 'ss': ss})
    return out
