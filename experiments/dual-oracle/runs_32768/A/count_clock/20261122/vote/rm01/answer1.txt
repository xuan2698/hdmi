def reference(vectors):
    h = 12
    m = 0
    s = 0
    p = 0
    out = []
    for v in vectors:
        if isinstance(v, dict):
            rst = v.get('reset', 0)
            en = v.get('ena', 0)
        else:
            rst = v[0] if len(v) > 0 else 0
            en = v[1] if len(v) > 1 else 0

        if rst:
            h, m, s, p = 12, 0, 0, 0
        elif en:
            s += 1
            if s == 60:
                s = 0
                m += 1
                if m == 60:
                    m = 0
                    if h == 11:
                        h = 12
                        p = 1 - p
                    elif h == 12:
                        h = 1
                    else:
                        h += 1

        hh = (h // 10) * 16 + (h % 10)
        mm = (m // 10) * 16 + (m % 10)
        ss = (s // 10) * 16 + (s % 10)
        out.append({'pm': p, 'hh': hh, 'mm': mm, 'ss': ss})

    return out
