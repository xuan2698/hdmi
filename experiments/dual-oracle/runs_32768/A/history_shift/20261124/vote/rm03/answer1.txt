def reference(vectors):
    history = 0
    mask = 0xFFFFFFFF
    outputs = []
    for row in vectors:
        if row['areset']:
            history = 0
        elif row['train_mispredicted']:
            history = ((row['train_history'] << 1) | row['train_taken']) & mask
        elif row['predict_valid']:
            history = ((history << 1) | row['predict_taken']) & mask
        outputs.append({'predict_history': history})
    return outputs
