def reference(vectors):
    x = 0
    y = 0
    out = []
    for v in vectors:
        if 'x' in v:
            x = v['x']
        if 'y' in v:
            y = v['y']
        y0 = y & 1
        y1 = (y >> 1) & 1
        y2 = (y >> 2) & 1
        Y0 = ((1 - x) & y0) | ((1 - x) & y2) | (x & (1 - y2) & (1 - y0))
        z = y2 | (y1 & y0)
        out.append({'Y0': Y0, 'z': z})
        if 'mid' in v:
            m = v['mid']
            if 'x' in m:
                x = m['x']
            if 'y' in m:
                y = m['y']
    return out
