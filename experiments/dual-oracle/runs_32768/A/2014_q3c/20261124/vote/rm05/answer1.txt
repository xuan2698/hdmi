def reference(vectors):
    out = []
    for v in vectors:
        x = v['x'] & 1
        y = v['y'] & 7
        y2 = (y >> 2) & 1
        y1 = (y >> 1) & 1
        y0 = y & 1

        Y0 = ((y2 | y0) & (1 - x)) | ((1 - y2) & (1 - y0) & x)
        z = y2 | (y1 & y0)

        out.append({'Y0': Y0, 'z': z})
    return out
