def reference(vectors):
    q = 0
    out = []
    for row in vectors:
        if isinstance(row, dict):
            d = row.get('d', 0)
            reset = row.get('reset', 0)
        else:
            d = row[0]
            reset = row[1]
        if reset & 1:
            q = 0x34
        else:
            q = d & 0xFF
        out.append({'q': q})
    return out
