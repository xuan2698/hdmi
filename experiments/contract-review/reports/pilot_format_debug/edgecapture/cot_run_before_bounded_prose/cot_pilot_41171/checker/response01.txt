def reference(vectors):
    # Contract: synchronous active-high reset clears both the sticky output
    # and the previous-input register.  For each bit, a 1->0 transition on
    # `in` sets the corresponding `out` bit, which stays set until reset.
    out = 0
    prev = 0
    result = []
    for row in vectors:
        reset = row['reset'] & 1
        inp = row['in'] & 0xFFFFFFFF
        if reset:
            out = 0
            prev = 0
        else:
            out = (out | (prev & (~inp & 0xFFFFFFFF))) & 0xFFFFFFFF
            prev = inp
        result.append({'out': out})
    return result
