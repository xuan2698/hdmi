def reference(vectors):
    # State: q_state holds the registered 8-bit output value.
    # Initial value: unspecified per spec, but candidate interpretation says zero.
    q_state = 0  # R1_sync_reset / global: initial q is zero

    results = []
    for row in vectors:
        d = row.get('d', 0)
        reset = row.get('reset', 0)

        # R1_sync_reset: active high synchronous reset -> q becomes 0 on posedge
        # R2_dff_capture: reset == 0 -> q captures d on posedge
        # Priority: reset (1) over capture (2)
        if reset == 1:
            q_state = 0  # R1_sync_reset
        else:
            q_state = d & 0xFF  # R2_dff_capture

        results.append({'q': q_state})

    return results
