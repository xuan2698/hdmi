def reference(vectors):
    # State: q_state holds the registered 8-bit output value between clock edges.
    q_state = 0  # power-up unspecified; use 0 as concrete value

    results = []
    for row in vectors:
        d = row.get('d', 0)
        reset = row.get('reset', 0)

        # R1_sync_reset: active high synchronous reset -> q = 0 on posedge
        # R2_dff_capture: reset == 0 -> q = d on posedge
        # Both rules apply at posedge clk; reset has priority 1.
        if reset == 1:
            q_state = 0  # R1_sync_reset
        else:
            q_state = d & 0xFF  # R2_dff_capture

        results.append({'q': q_state & 0xFF})

    return results
