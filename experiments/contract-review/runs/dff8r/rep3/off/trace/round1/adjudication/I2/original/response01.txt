def reference(vectors):
    # State: q_state holds the 8-bit registered output.
    # R1/R2: synchronous reset sampled at posedge clk.
    q_state = 0  # power-up unspecified; use 0 as concrete value
    out = []
    for row in vectors:
        d = row.get('d', 0)
        reset = row.get('reset', 0)
        # R1: reset == 1 at posedge -> q becomes 0 (priority 1)
        # R2: reset == 0 at posedge -> q becomes d (priority 2)
        if reset == 1:
            q_state = 0
        else:
            q_state = d & 0xFF
        out.append({'q': q_state & 0xFF})
    return out
