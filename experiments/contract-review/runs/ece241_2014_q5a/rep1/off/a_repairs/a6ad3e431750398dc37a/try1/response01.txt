def reference(vectors):
    # Contract implementation for a serial 2's complementer Moore FSM.
    # States: S0 (no 1 seen yet), S1 (at least one 1 seen).
    # Outputs are combinational based on current state and x.
    # State updates on posedge clk; async reset forces S0 and z=0.

    out = []
    state = "S0"  # power-up state (unspecified, but we pick S0)

    for row in vectors:
        areset = row.get("areset", 0)
        x = row.get("x", 0)

        # R4: asynchronous reset asserted -> state forced to S0, z=0 (same cycle)
        if areset == 1:
            state = "S0"
            z = 0
            out.append({"z": z})
            continue

        # Combinational output based on current state and x (same cycle)
        # R1: In S0, z = x
        # R2: In S1, z = NOT x
        if state == "S0":
            z = x & 1  # R1
        else:  # S1
            z = (~x) & 1  # R2

        out.append({"z": z})

        # R3: State transition on posedge clk (visible next cycle)
        # S0 -> S1 when x=1; S1 stays S1
        if state == "S0" and x == 1:
            state = "S1"
        # S1 stays S1

    return out
