def reference(vectors):
    # State encoding
    S0 = 0
    S1 = 1

    # Contract rules:
    # R4: asynchronous reset (areset=1) forces state to S0 and z=0 (same cycle).
    #     Candidate interpretation applied: reset is synchronous - state and z
    #     update only on posedge clk when areset=1. We model the sampled behavior:
    #     when areset=1 at the sampled edge, state becomes S0 and z=0.
    # R1: In S0, z = x (combinational, same cycle).
    # R2: In S1, z = NOT x (combinational, same cycle).
    # R3: State transition on posedge: S0->S1 when x=1; S1 stays S1.

    state = S0  # power-up state (unspecified, use S0)
    out = []

    for row in vectors:
        areset = row.get('areset', 0) & 1
        x = row.get('x', 0) & 1

        # Determine current-cycle output based on current state (before edge update)
        # R4: reset forces z=0 (synchronous interpretation: applied at edge,
        #     but visible same cycle per contract; we treat areset as overriding)
        if areset:
            # R4: reset asserted -> z=0
            z = 0
            # R4: state forced to S0 (synchronous: at posedge)
            state = S0
        else:
            # R1 / R2: combinational output based on current state
            if state == S0:
                z = x  # R1
            else:
                z = 1 - x  # R2 (NOT x)

            # R3: state transition on posedge
            if state == S0 and x == 1:
                state = S1
            # S1 stays S1

        out.append({'z': z & 1})

    return out
