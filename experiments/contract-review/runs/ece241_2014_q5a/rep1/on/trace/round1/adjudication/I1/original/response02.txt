def reference(vectors):
    out = []
    state = 0  # A: reset state / no 1 seen yet
    cur_areset = 0
    cur_x = 0

    for row in vectors:
        if "areset" in row:
            cur_areset = row["areset"]
        if "x" in row:
            cur_x = row["x"]

        # reset_async_z0: asynchronous reset forces state A and output z=0
        if cur_areset == 1:
            state = 0

        if cur_areset == 1:
            # reset_async_z0
            state = 0
            z = 0
        else:
            if state == 0:  # A
                if cur_x == 0:
                    # A_x0_to_A_z0
                    state = 0
                    z = 0
                else:
                    # A_x1_to_C_z1
                    state = 2
                    z = 1
            elif state == 1:  # B
                if cur_x == 0:
                    # B_x0_to_C_z1
                    state = 2
                    z = 1
                else:
                    # B_x1_to_B_z0
                    state = 1
                    z = 0
            else:  # C
                if cur_x == 0:
                    # C_x0_to_C_z1
                    state = 2
                    z = 1
                else:
                    # C_x1_to_B_z0
                    state = 1
                    z = 0

        out.append({"z": z})

        # Process optional mid changes after sampling.
        if "mid" in row:
            mid = row["mid"]
            if mid:
                # Try treating mid as a single dict.
                try:
                    v = mid["areset"]
                    cur_areset = v
                    if cur_areset == 1:
                        # reset_async_z0
                        state = 0
                except:
                    pass
                try:
                    v = mid["x"]
                    cur_x = v
                except:
                    pass

                # Try treating mid as a sequence of dicts.
                for m in mid:
                    try:
                        v = m["areset"]
                        cur_areset = v
                        if cur_areset == 1:
                            # reset_async_z0
                            state = 0
                    except:
                        pass
                    try:
                        v = m["x"]
                        cur_x = v
                    except:
                        pass

    return out
