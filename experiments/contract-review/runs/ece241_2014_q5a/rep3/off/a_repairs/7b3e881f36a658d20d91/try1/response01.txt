def reference(vectors):
    # Contract implementation for 2's complementer Moore FSM.
    # States: S0 (copy phase, no 1 seen yet), S1 (invert phase, 1 seen).
    # R3: areset==1 asynchronously forces state=S0 and z=0 (same cycle).
    # R1: state==S0 -> z = x (combinational, same cycle).
    # R2: state==S1 -> z = ~x (combinational, same cycle).
    # R4: posedge clk, areset==0: if state==S0 and x==1 -> next state=S1; else unchanged.
    # Mid changes occur AFTER sampling, before other edge, carried to next row.

    state = 'S0'  # power-up unspecified but we start at S0
    results = []

    for row in vectors:
        areset = row.get('areset', 0)
        x = row.get('x', 0)

        # R3: asynchronous reset forces state to S0 and z to 0 (same cycle)
        if areset == 1:
            state = 'S0'
            z = 0
            results.append({'z': z})
            # mid changes after sampling: apply to next row's inputs
            # (mid changes are carried to start of next row; we handle by
            #  reading row values fresh each iteration, so nothing to do here)
            continue

        # Combinational output based on current state (R1 / R2)
        if state == 'S0':
            z = x & 1  # R1
        else:
            z = (~x) & 1  # R2

        results.append({'z': z})

        # R4: posedge clk transition (visible next cycle)
        if state == 'S0' and x == 1:
            state = 'S1'
        # else state unchanged

    return results
