def reference(vectors):
    # Contract implementation.
    # State encoding: S0 = 0, S1 = 1
    # R3: async reset forces state to S0 and z to 0 (same cycle)
    # R1: state == S0 -> z = x (Moore, registered, visible next cycle)
    # R2: state == S1 -> z = ~x (Moore, registered, visible next cycle)
    # R4: posedge clk, areset==0: if state==S0 and x==1 -> next state = S1; else unchanged

    S0 = 0
    S1 = 1

    state = S0
    z_reg = 0  # registered output

    results = []

    for row in vectors:
        areset = row.get('areset', 0)
        x = row.get('x', 0)

        # R3: asynchronous reset - same cycle effect
        if areset == 1:
            state = S0
            z_reg = 0
            results.append({'z': z_reg})
            continue

        # Normal posedge operation (areset == 0)
        # R1/R2: Moore output based on current state and current x, registered -> visible next cycle
        if state == S0:
            z_next = x & 1          # R1
        else:
            z_next = (~x) & 1       # R2

        # R4: state transition
        if state == S0 and x == 1:
            state = S1
        # else state unchanged

        z_reg = z_next
        results.append({'z': z_reg})

    return results
