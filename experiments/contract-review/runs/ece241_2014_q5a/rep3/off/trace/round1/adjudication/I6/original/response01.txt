def reference(vectors):
    # Contract implementation.
    # State: S0 = copy phase (no 1 seen yet), S1 = invert phase (1 seen).
    # R3: areset==1 asynchronously forces state to S0 and z to 0 (same cycle).
    # R1: state==S0 -> z = x (combinational, same cycle).
    # R2: state==S1 -> z = ~x (combinational, same cycle).
    # R4: posedge clk, areset==0: if state==S0 and x==1 -> next state S1; else unchanged.
    # Mid changes occur AFTER sampling, before next edge, carried to next row.

    state = "S0"  # power-up unspecified; use S0 as concrete initial
    out = []

    for row in vectors:
        areset = row.get("areset", 0) & 1
        x = row.get("x", 0) & 1

        # R3: asynchronous reset forces state S0 and z=0 (same cycle)
        if areset == 1:
            state = "S0"
            z = 0
        else:
            # R1 / R2: combinational output based on current state
            if state == "S0":
                z = x  # R1
            else:
                z = (~x) & 1  # R2

        out.append({"z": z & 1})

        # R4: posedge clk update (sampled edge already settled).
        # If areset==1, state stays S0 (update_during_reset false).
        if areset == 0:
            if state == "S0" and x == 1:
                state = "S1"
            # else state unchanged

    return out
