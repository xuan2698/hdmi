def reference(vectors):
    # Contract implementation.
    # State: S0 = copy phase (no 1 seen yet), S1 = invert phase (1 seen).
    # R3: areset high asynchronously forces state S0 and z=0 (same cycle).
    # R1: state S0 -> z = x (combinational, same cycle).
    # R2: state S1 -> z = ~x (combinational, same cycle).
    # R4: posedge clk, areset==0: if S0 and x==1 -> S1; else unchanged.
    # Global interpretation: conversion begins immediately when areset deasserts.
    # Mid changes occur AFTER sampling, before other edge, carried to next row.

    state = "S0"
    out = []

    for row in vectors:
        areset = row.get("areset", 0) & 1
        x = row.get("x", 0) & 1

        # R3: async reset forces state S0 and z=0 (same cycle)
        if areset == 1:
            state = "S0"
            z = 0
        else:
            # R1 / R2: combinational output based on current state
            if state == "S0":
                z = x  # R1
            else:
                z = (~x) & 1  # R2

        out.append({"z": z & 1})

        # R4: posedge clk update (sample after edge settled).
        # areset==0: if S0 and x==1 -> S1; else unchanged.
        if areset == 0:
            if state == "S0" and x == 1:
                state = "S1"
            # else state unchanged

        # Mid changes: applied AFTER sampling, carried to next row.
        # (No mid fields in this contract's rows; nothing to do.)

    return out
