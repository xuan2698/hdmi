def reference(vectors):
    state = 'A'
    areset = 1
    x = 0
    out = []

    for row in vectors:
        if isinstance(row, dict):
            if 'areset' in row:
                areset = row['areset']
            if 'x' in row:
                x = row['x']

        # async_reset_assert: active-high asynchronous reset forces state A and z=0.
        if areset == 1:
            state = 'A'

        # Positive-edge sampling with reset priority.
        if areset == 1:
            # async_reset_assert holds state A while reset is asserted.
            state = 'A'
        else:
            if state == 'A':
                if x == 0:
                    # A_on_x0
                    state = 'A'
                else:
                    # A_on_x1
                    state = 'B'
            elif state == 'B':
                if x == 0:
                    # B_on_x0
                    state = 'B'
                else:
                    # B_on_x1
                    state = 'C'
            else:  # state == 'C'
                if x == 0:
                    # C_on_x0
                    state = 'B'
                else:
                    # C_on_x1
                    state = 'C'

        # Moore output after the posedge has settled: A=0, B=1, C=0.
        z = 1 if state == 'B' else 0
        out.append({'z': z})

        # mid changes occur after sampling and are carried into the next row.
        if isinstance(row, dict) and 'mid' in row:
            mid = row['mid']
            mids = mid if isinstance(mid, (list, tuple)) else [mid]
            for m in mids:
                if isinstance(m, dict):
                    if 'areset' in m:
                        areset = m['areset']
                    if 'x' in m:
                        x = m['x']
                    # async_reset_assert: mid reset assertion takes effect immediately.
                    if areset == 1:
                        state = 'A'

    return out
