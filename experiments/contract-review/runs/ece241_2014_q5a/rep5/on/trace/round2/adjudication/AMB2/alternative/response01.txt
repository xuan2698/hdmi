def reference(vectors):
    # State constants
    A, B, C = 0, 1, 2

    # Power-up / reset values: concrete unsigned values even when unspecified.
    state = A
    current_areset = 0
    x = 0
    z = 0
    held_z = 0  # alternate async_reset_assert interpretation: retain z while reset asserted
    result = []

    for row in vectors:
        # Drive row inputs before the sampled posedge.
        if 'areset' in row:
            new_areset = row['areset']
            if new_areset != current_areset:
                if new_areset == 1:
                    # async_reset_assert: active-high asynchronous reset.
                    # Alternate interpretation selected here: z retains its previous value.
                    held_z = z
                    state = A
                    z = held_z
                    current_areset = 1
                else:
                    # Reset release: state is A at release; output returns to state A's Moore value.
                    current_areset = 0
                    state = A
                    z = 0
            else:
                if current_areset == 1:
                    # Keep reset state and retained output while areset remains asserted.
                    state = A
                    z = held_z

        if 'x' in row:
            x = row['x']

        if current_areset == 1:
            # async_reset_assert dominates the posedge while asserted.
            state = A
            z = held_z
        else:
            # Posedge state transitions when reset is inactive.
            if state == A:
                if x == 0:
                    # A_on_x0: A, x=0 -> A, z=0
                    state = A
                    z = 0
                else:
                    # A_on_x1: A, x=1 -> B, z=1
                    state = B
                    z = 1
            elif state == B:
                if x == 0:
                    # B_on_x0: B, x=0 -> B, z=1
                    state = B
                    z = 1
                else:
                    # B_on_x1: B, x=1 -> C, z=0
                    state = C
                    z = 0
            else:  # state == C
                if x == 0:
                    # C_on_x0: C, x=0 -> B, z=1
                    state = B
                    z = 1
                else:
                    # C_on_x1: C, x=1 -> C, z=0
                    state = C
                    z = 0

        # Sample after the posedge has settled.
        result.append({'z': int(z) & 1})

        # Mid changes occur after sampling, before the next sampled edge,
        # and are carried into the next row's starting inputs.
        if 'mid' in row:
            mid_value = row['mid']
            if isinstance(mid_value, list):
                mids = mid_value
            else:
                mids = [mid_value]

            for mid in mids:
                if 'x' in mid:
                    x = mid['x']

                if 'areset' in mid:
                    new_areset = mid['areset']
                    if new_areset != current_areset:
                        if new_areset == 1:
                            # async_reset_assert: asynchronous assertion after sampling.
                            held_z = z
                            state = A
                            z = held_z
                            current_areset = 1
                        else:
                            # Asynchronous release after sampling.
                            current_areset = 0
                            state = A
                            z = 0
                    else:
                        if current_areset == 1:
                            state = A
                            z = held_z

    return result
