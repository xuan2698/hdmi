def reference(vectors):
    # Moore FSM for serial 2's complement (LSB first).
    # States: S0 = haven't seen first '1' yet (z=0), S1 = seen first '1' (z=1)
    # Output z depends only on state: S0 -> 0, S1 -> 1
    # Transitions on x:
    #   S0: x=0 -> S0, x=1 -> S1
    #   S1: x=0 -> S1, x=1 -> S1
    # Asynchronous reset (active high) forces state to S0 immediately.
    # Output has 1 cycle latency: z reflects state before the sampled edge.
    # We model: state updates on posedge clk (if not areset), z = current state's output.
    # Sampling: after posedge settles, we read z (which is based on state BEFORE this edge's update? 
    # Actually Moore output = current state. After posedge, state has updated. So z after posedge = new state's output.
    # Latency 1 cycle means: input sampled at edge N affects state at edge N, output visible after edge N (i.e., at cycle N+1 relative to input).
    # We'll compute: for each row, given inputs (clk, areset, x) applied before the sampled edge,
    # determine state after the edge (or reset), then z = output of that state.
    
    state = 0  # 0 = S0, 1 = S1
    results = []
    
    for row in vectors:
        clk = row.get('clk', 0)
        areset = row.get('areset', 0)
        x = row.get('x', 0)
        
        # Asynchronous reset: if areset is high, state forced to 0 immediately (before edge)
        if areset:
            state = 0
        else:
            # Positive edge triggered: if clk is 1 (posedge), update state based on x
            if clk == 1:
                if state == 0:
                    if x == 1:
                        state = 1
                    # else stay 0
                else:  # state == 1
                    # stay 1 regardless of x
                    pass
        
        # Moore output z = state's output
        z = state  # S0 -> 0, S1 -> 1
        
        results.append({'z': z})
    
    return results
