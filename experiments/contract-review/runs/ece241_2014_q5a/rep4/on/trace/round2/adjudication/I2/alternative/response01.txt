def reference(vectors):
    # R_RESET: active-high asynchronous reset; chosen interpretation:
    # on reset assertion state returns to START but z retains its previous value.
    # R_CONVERT: LSB-first serial 2's complement conversion on posedge clk.
    state = "START"
    z = 0
    cur_areset = 0
    cur_x = 0
    result = []

    for row in vectors:
        # Drive this row's non-mid input values before the sampled posedge.
        if "areset" in row:
            cur_areset = row["areset"]
        if "x" in row:
            cur_x = row["x"]

        # R_RESET: asynchronous reset has priority and returns state to START.
        # Chosen interpretation: z is not forced to 0; it retains its previous value.
        if cur_areset == 1:
            state = "START"

        # R_CONVERT: posedge conversion only when reset is deasserted.
        if cur_areset == 0:
            if state == "START":
                if cur_x == 0:
                    state = "START"
                    z = 0
                else:
                    state = "INV1"
                    z = 1
            elif state == "INV0":
                if cur_x == 0:
                    state = "INV1"
                    z = 1
                else:
                    state = "INV0"
                    z = 0
            else:  # state == "INV1"
                if cur_x == 0:
                    state = "INV1"
                    z = 1
                else:
                    state = "INV0"
                    z = 0

        # Output after the sampled posedge has settled.
        result.append({"z": int(z)})

        # Mid changes occur after sampling and carry to the start of the next row.
        if "mid" in row:
            mid = row["mid"]
            if isinstance(mid, dict):
                if "x" in mid:
                    cur_x = mid["x"]
                if "areset" in mid:
                    cur_areset = mid["areset"]

                # R_RESET: asynchronous assertion through mid changes also
                # returns state to START while z retains its previous value.
                if cur_areset == 1:
                    state = "START"

    return result
