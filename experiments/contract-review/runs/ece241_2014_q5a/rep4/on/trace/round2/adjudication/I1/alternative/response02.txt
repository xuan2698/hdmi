def reference(vectors):
    # R_RESET: active-low areset (0 asserted) asynchronously forces START and z=0.
    state = "START"
    areset = 1
    x = 0
    result = []

    for row in vectors:
        # Apply each row's driven values before the sampled posedge.
        if 'areset' in row:
            areset = row['areset']
        if 'x' in row:
            x = row['x']

        # R_RESET: asynchronous reset has priority before posedge sampling.
        if areset == 0:
            state = "START"
            z = 0
        else:
            # R_CONVERT: posedge transition and Moore output after the edge settles.
            if state == "START":
                if x == 1:
                    state = "INV1"
                else:
                    state = "START"
            elif state == "INV0":
                if x == 0:
                    state = "INV1"
                else:
                    state = "INV0"
            elif state == "INV1":
                if x == 0:
                    state = "INV1"
                else:
                    state = "INV0"

            if state == "INV1":
                z = 1
            else:
                z = 0

        result.append({"z": z})

        # Mid changes occur after sampling and are carried to the next row.
        if 'mid' in row:
            mid = row['mid']
            if 'areset' in mid:
                areset = mid['areset']
                # R_RESET: asynchronous active-low reset assertion after edge.
                if areset == 0:
                    state = "START"
            if 'x' in mid:
                x = mid['x']

    return result
