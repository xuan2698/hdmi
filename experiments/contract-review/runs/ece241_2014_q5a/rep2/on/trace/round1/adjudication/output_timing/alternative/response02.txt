def reference(vectors):
    # State encoding: 0=S0, 1=S1, 2=S2
    state = 0
    cur_areset = 0
    cur_x = 0
    out = []

    for row in vectors:
        # Each row's values are driven before the sampled edge.
        if 'areset' in row:
            cur_areset = row['areset']
        if 'x' in row:
            cur_x = row['x']

        # async_reset_z0: asynchronously force state S0 and z=0 while reset asserted.
        if cur_areset == 1:
            state = 0
            z = 0
        else:
            # global: Registered Moore output is sampled from the pre-edge state
            # (one-cycle latency).  z is determined by the current state before transition.
            if state == 0:
                z = 0
            elif state == 1:
                z = 1
            else:  # state == 2
                z = 0

            # State transitions at posedge when areset=0.
            if state == 0:
                if cur_x == 0:
                    state = 0  # s0_x0
                else:
                    state = 1  # s0_x1
            elif state == 1:
                if cur_x == 0:
                    state = 1  # s1_x0
                else:
                    state = 2  # s1_x1
            else:  # state == 2
                if cur_x == 0:
                    state = 1  # s2_x0
                else:
                    state = 2  # s2_x1

        out.append({'z': z})

        # Optional mid input changes occur AFTER sampling, before the next clock edge.
        # They are carried to the start of the next row.
        if 'mid' in row:
            m = row['mid']
            cls = m.__class__.__name__
            if cls == 'dict':
                if 'areset' in m:
                    cur_areset = m['areset']
                    # async_reset_z0: mid-cycle assertion of reset forces state S0.
                    if cur_areset == 1:
                        state = 0
                if 'x' in m:
                    cur_x = m['x']
            elif cls == 'list' or cls == 'tuple':
                for sub in m:
                    if sub.__class__.__name__ == 'dict':
                        if 'areset' in sub:
                            cur_areset = sub['areset']
                            if cur_areset == 1:
                                state = 0
                        if 'x' in sub:
                            cur_x = sub['x']

    return out
