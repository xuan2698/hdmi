def reference(vectors):
    # State encoding
    S0 = 0  # no 1-bit seen yet
    S1 = 1  # at least one 1-bit seen

    # Contract rules implemented:
    # R1: In S0, z = x (combinational, same cycle)
    # R2: In S1, z = NOT x (combinational, same cycle)
    # R3: posedge transitions: S0->S1 when x=1, S0->S0 when x=0, S1->S1 always
    # R4: async reset (areset=1) forces state to S0 and z=0 same cycle
    # global: conversion starts sampling x on first posedge where areset=0

    state = S0
    results = []

    for row in vectors:
        areset = row.get('areset', 0) & 1
        x = row.get('x', 0) & 1

        # R4: asynchronous reset dominates - same cycle z forced to 0, state -> S0
        if areset == 1:
            state = S0
            z = 0
            results.append({'z': z & 1})
            continue

        # R1 / R2: combinational output based on current state
        if state == S0:
            z = x  # R1
        else:
            z = (~x) & 1  # R2

        results.append({'z': z & 1})

        # R3: posedge state transition (effective next cycle)
        if state == S0:
            if x == 1:
                state = S1
            else:
                state = S0
        else:
            state = S1

    return results
