def reference(vectors):
    # Contract implementation for a serial 2's complementer Moore machine.
    # States: S0 (no 1 seen yet), S1 (at least one 1 seen).
    # R4: async reset (areset=1) forces state to S0 and z=0, same cycle.
    # R1: in S0, z = x (combinational, same cycle).
    # R2: in S1, z = NOT x (combinational, same cycle).
    # R3: posedge transition: S0->S1 if x=1 else S0; S1->S1 always.
    #     Same-cycle z uses current state; transition effective next cycle.

    state = "S0"  # power-up state (unspecified in spec, but S0 is natural)
    results = []

    for row in vectors:
        areset = row.get("areset", 0) & 1
        x = row.get("x", 0) & 1

        # R4: asynchronous reset - same cycle effect
        if areset == 1:
            state = "S0"
            z = 0
            results.append({"z": z})
            # Reset holds state; no posedge transition this row.
            continue

        # R1 / R2: combinational output based on current state
        if state == "S0":
            z = x  # R1
        else:  # S1
            z = 1 - x  # R2

        results.append({"z": z})

        # R3: posedge transition (effective next cycle)
        if state == "S0":
            if x == 1:
                state = "S1"
            else:
                state = "S0"
        else:  # S1
            state = "S1"

    return results
