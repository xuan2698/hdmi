def reference(vectors):
    # Contract implementation.
    # State encoding: 0 = S0 (no 1 seen yet), 1 = S1 (complementing).
    # R4: async reset (areset=1) forces state to S0 and z=0, same cycle.
    # R1: in S0, z = x (combinational, same cycle).
    # R2: in S1, z = NOT x (combinational, same cycle).
    # R3: posedge transition: S0->S1 when x=1; S0->S0 when x=0; S1->S1 always.
    #     Transition takes effect after current cycle's output.
    # Reset release: first posedge after areset deasserted samples x from S0.
    # Mid changes occur after sampling, carried to next row.

    state = 0  # power-up unspecified; start in S0
    out = []

    for row in vectors:
        areset = row.get('areset', 0) & 1
        x = row.get('x', 0) & 1

        # R4: asynchronous reset - same cycle, priority 0
        if areset:
            state = 0
            z = 0
            out.append({'z': z & 1})
            # reset holds state; no posedge transition during reset
            continue

        # R1 / R2: combinational output based on current state
        if state == 0:
            z = x  # R1
        else:
            z = (1 - x) & 1  # R2

        out.append({'z': z & 1})

        # R3: posedge transition (takes effect after this cycle's output)
        if state == 0:
            if x == 1:
                state = 1
            else:
                state = 0
        else:
            state = 1

    return out
