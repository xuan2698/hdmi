def reference(vectors):
    # Contract implementation for 2's complementer Moore state machine.
    # State: S0 = no 1 seen yet (pass through), S1 = 1 seen (complement).
    # R4: async reset (areset=1) forces state S0 and z=0, same cycle.
    # R1: In S0, z = x (combinational, same cycle).
    # R2: In S1, z = NOT x (combinational, same cycle).
    # R3: posedge transitions: S0->S1 if x=1 else S0; S1->S1 always.
    # Mid changes occur after sampling, carried to next row.

    state = "S0"  # power-up unspecified but we start at S0
    results = []

    for row in vectors:
        areset = row.get("areset", 0)
        x = row.get("x", 0)

        # R4: asynchronous reset - highest priority, same cycle
        if areset == 1:
            state = "S0"
            z = 0
            results.append({"z": z})
            # reset is async; state stays S0. No posedge transition during reset.
            continue

        # Combinational output based on current state (R1 / R2)
        if state == "S0":
            z = x  # R1
        else:
            z = 1 - x  # R2 (NOT x)

        results.append({"z": z})

        # R3: posedge state transition (sampled at this edge)
        if state == "S0":
            if x == 1:
                state = "S1"
            else:
                state = "S0"
        else:
            state = "S1"

    return results
