def reference(vectors):
    # Contract implementation for 2's complementer Moore FSM.
    # States: S0 (no 1 seen yet, pass-through), S1 (1 seen, complement).
    # R4: async reset high -> state S0, z=0 same cycle.
    # R1: S0 -> z = x (combinational, same cycle).
    # R2: S1 -> z = NOT x (combinational, same cycle).
    # R3: posedge: S0->S1 if x=1 else S0; S1->S1 always.
    # Mid changes occur after sampling edge, carried to next row start.

    state = "S0"
    out = []

    for row in vectors:
        areset = row.get("areset", 0) & 1
        x = row.get("x", 0) & 1

        # R4: asynchronous reset asserted -> state S0, z forced 0 (same cycle)
        if areset:
            state = "S0"
            z = 0
            out.append({"z": z & 1})
            # reset is async; no posedge state update while reset asserted
            continue

        # R1 / R2: combinational output based on current state
        if state == "S0":
            z = x  # R1
        else:
            z = (1 - x) & 1  # R2

        out.append({"z": z & 1})

        # R3: posedge state transition (takes effect after this cycle's output)
        if state == "S0":
            if x == 1:
                state = "S1"
            else:
                state = "S0"
        else:
            state = "S1"

    return out
