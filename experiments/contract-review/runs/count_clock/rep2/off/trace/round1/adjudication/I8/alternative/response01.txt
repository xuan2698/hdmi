def reference(vectors):
    # State: ss_lo, ss_hi, mm_lo, mm_hi, hh_lo, hh_hi, pm_reg
    # Initial (power-up) state: 12:00 AM -> hh=0x12, mm=0x00, ss=0x00, pm=0
    ss_lo = 0
    ss_hi = 0
    mm_lo = 0
    mm_hi = 0
    hh_lo = 2
    hh_hi = 1
    pm_reg = 0

    out = []
    for row in vectors:
        reset = row.get('reset', 0)
        ena = row.get('ena', 0)

        # Sample current outputs (before edge)
        ss = (ss_hi << 4) | ss_lo
        mm = (mm_hi << 4) | mm_lo
        hh = (hh_hi << 4) | hh_lo
        out.append({'pm': pm_reg, 'hh': hh, 'mm': mm, 'ss': ss})

        # Compute next state at posedge
        if reset:
            # R1_reset: reset to 12:00 AM
            ss_lo, ss_hi = 0, 0
            mm_lo, mm_hi = 0, 0
            hh_lo, hh_hi = 2, 1
            pm_reg = 0
        elif not ena:
            # R2_hold: hold
            pass
        else:
            # ena == 1, not reset
            # R3_seconds: increment seconds if ss != 0x59
            if ss != 0x59:
                # increment BCD seconds
                if ss_lo == 9:
                    ss_lo = 0
                    ss_hi += 1
                else:
                    ss_lo += 1
            else:
                # ss == 0x59
                ss_lo, ss_hi = 0, 0
                if mm != 0x59:
                    # R4_seconds_roll: roll seconds, increment minutes
                    if mm_lo == 9:
                        mm_lo = 0
                        mm_hi += 1
                    else:
                        mm_lo += 1
                else:
                    # mm == 0x59
                    mm_lo, mm_hi = 0, 0
                    if hh != 0x12:
                        # R5_minutes_roll: roll minutes, increment hours
                        # hours 01-12 BCD
                        if hh_lo == 9:
                            hh_lo = 0
                            hh_hi += 1
                        elif hh_lo == 2 and hh_hi == 1:
                            # 12 -> 01
                            hh_lo = 1
                            hh_hi = 0
                        else:
                            hh_lo += 1
                    else:
                        # R6_hours_roll_am_pm: hh == 0x12, roll to 01, toggle pm
                        hh_lo, hh_hi = 1, 0
                        pm_reg = 1 - pm_reg
    return out
