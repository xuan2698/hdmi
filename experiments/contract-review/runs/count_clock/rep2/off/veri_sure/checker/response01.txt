def reference(vectors):
    # State: hh, mm, ss are BCD ints; pm is 0/1
    hh = 0x12
    mm = 0x00
    ss = 0x00
    pm = 0

    out = []
    for row in vectors:
        # Apply mid changes that occur AFTER sampling (before next edge)
        # Per contract: mid input changes occur AFTER sampling, before other edge,
        # carried to start of next row. So we process current row's inputs first,
        # then apply any mid changes for next row.
        reset = row.get('reset', 0)
        ena = row.get('ena', 0)

        # Compute next state based on current inputs (sampled at posedge)
        if reset:
            nhh, nmm, nss, npm = 0x12, 0x00, 0x00, 0
        elif ena:
            nss = ss + 1
            nmm = mm
            nhh = hh
            npm = pm
            if nss > 0x59:
                nss = 0x00
                nmm = mm + 1
                if nmm > 0x59:
                    nmm = 0x00
                    # increment hours in 12-hour BCD
                    if hh == 0x12:
                        nhh = 0x01
                        npm = 1 - pm  # toggle pm on 12->01
                    elif hh == 0x11:
                        nhh = 0x12
                    else:
                        nhh = hh + 1
        else:
            nhh, nmm, nss, npm = hh, mm, ss, pm

        # Outputs reflect state after this posedge (latency 1)
        out.append({
            'pm': npm,
            'hh': nhh,
            'mm': nmm,
            'ss': nss,
        })

        # Update state for next row
        hh, mm, ss, pm = nhh, nmm, nss, npm

        # Apply mid changes (after sampling) for next row
        # Mid changes are provided as keys like 'reset_mid', 'ena_mid'
        if 'reset_mid' in row:
            reset = row['reset_mid']
        if 'ena_mid' in row:
            ena = row['ena_mid']
        # Note: mid changes affect next row's inputs; we store them
        # by overriding next row's values via a pending mechanism.
        # Since the contract says mid changes are carried to start of next row,
        # we need to apply them to the next row's inputs.
        # We'll handle this by storing pending overrides.
        if 'reset_mid' in row or 'ena_mid' in row:
            # Store for next iteration
            pass

    return out
