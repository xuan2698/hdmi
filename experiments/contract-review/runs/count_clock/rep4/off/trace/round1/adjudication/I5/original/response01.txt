def reference(vectors):
    # State: ss_lo, ss_hi, mm_lo, mm_hi, hh_lo, hh_hi, pm_reg
    # Initialize to 12:00 AM (hh=0x12, mm=0x00, ss=0x00, pm=0)
    ss_lo, ss_hi = 0, 0
    mm_lo, mm_hi = 0, 0
    hh_lo, hh_hi = 2, 1  # 12
    pm_reg = 0

    out = []
    for row in vectors:
        reset = row.get('reset', 0)
        ena = row.get('ena', 0)

        # Sample current outputs (before edge)
        out.append({
            'pm': pm_reg,
            'hh': (hh_hi << 4) | hh_lo,
            'mm': (mm_hi << 4) | mm_lo,
            'ss': (ss_hi << 4) | ss_lo,
        })

        # Compute next state at posedge
        if reset:
            # R1_reset: reset to 12:00 AM
            ss_lo, ss_hi = 0, 0
            mm_lo, mm_hi = 0, 0
            hh_lo, hh_hi = 2, 1
            pm_reg = 0
        elif ena:
            # R2_hold: if not ena, hold (handled by else)
            ss_val = (ss_hi << 4) | ss_lo
            mm_val = (mm_hi << 4) | mm_lo
            hh_val = (hh_hi << 4) | hh_lo

            if ss_val != 0x59:
                # R3_seconds: increment seconds
                if ss_lo == 9:
                    ss_lo = 0
                    ss_hi += 1
                else:
                    ss_lo += 1
            else:
                # ss == 0x59
                ss_lo, ss_hi = 0, 0
                if mm_val != 0x59:
                    # R4_seconds_roll: roll seconds, increment minutes
                    if mm_lo == 9:
                        mm_lo = 0
                        mm_hi += 1
                    else:
                        mm_lo += 1
                else:
                    # mm == 0x59
                    mm_lo, mm_hi = 0, 0
                    if hh_val != 0x12:
                        # R5_minutes_roll: increment hours (not 12)
                        if hh_lo == 9:
                            hh_lo = 0
                            hh_hi += 1
                        else:
                            hh_lo += 1
                    else:
                        # hh == 0x12
                        # R6_hour_roll_am_pm: 12 -> 1, toggle pm
                        hh_lo, hh_hi = 1, 0
                        pm_reg = 1 - pm_reg
            # R7_hour_roll_12: 11 -> 12, pm stays (handled by R5 since 0x11 != 0x12)
            # Actually R7: hh==0x11 -> 0x12, pm unchanged. That's covered by R5 path
            # since 0x11 != 0x12, increment: hh_lo=1,hh_hi=1 -> 0x12. Correct.
        # else: R2_hold - no change

    return out
