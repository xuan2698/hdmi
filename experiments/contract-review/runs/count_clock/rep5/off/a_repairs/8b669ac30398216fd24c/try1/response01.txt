def reference(vectors):
    # Contract rules implemented:
    # R1_reset: synchronous active-high reset -> 12:00 AM (pm=0, hh=0x12, mm=0, ss=0)
    # R2_seconds: increment ss when ena and not reset and ss != 0x59
    # R3_seconds_rollover: ss==0x59 -> ss=0, mm++ (when mm != 0x59)
    # R4_minutes_rollover: ss==0x59 && mm==0x59 && hh != 0x12 -> mm=0, hh++
    # R5_hour_rollover_am_pm: ss==0x59 && mm==0x59 && hh==0x11 -> hh=0x12, pm toggles
    # R6_hold: ena==0 and not reset -> hold all
    # hh is BCD 12-hour: 0x12 = 12 o'clock, 0x01 = 1 o'clock, etc.

    # State: BCD digits
    ss_ones = 0
    ss_tens = 0
    mm_ones = 0
    mm_tens = 0
    hh_ones = 2  # 12 -> ones=2, tens=1
    hh_tens = 1
    pm_state = 0

    results = []
    for row in vectors:
        reset = row.get('reset', 0)
        ena = row.get('ena', 0)

        # R1_reset: reset has priority, synchronous
        if reset == 1:
            ss_ones = 0
            ss_tens = 0
            mm_ones = 0
            mm_tens = 0
            hh_ones = 2
            hh_tens = 1
            pm_state = 0
        elif ena == 1:
            # R2_seconds / R3_seconds_rollover
            ss_val = ss_tens * 10 + ss_ones
            if ss_val != 59:
                # increment seconds BCD
                if ss_ones == 9:
                    ss_ones = 0
                    ss_tens += 1
                else:
                    ss_ones += 1
            else:
                # R3_seconds_rollover: ss 59 -> 00, increment minutes
                ss_ones = 0
                ss_tens = 0
                mm_val = mm_tens * 10 + mm_ones
                if mm_val != 59:
                    if mm_ones == 9:
                        mm_ones = 0
                        mm_tens += 1
                    else:
                        mm_ones += 1
                else:
                    # R4_minutes_rollover: mm 59 -> 00, increment hours
                    mm_ones = 0
                    mm_tens = 0
                    hh_val = hh_tens * 10 + hh_ones
                    if hh_val != 12:
                        # increment hours BCD (1..12)
                        if hh_ones == 9:
                            hh_ones = 0
                            hh_tens += 1
                        elif hh_ones == 2 and hh_tens == 1:
                            # 12 -> 1
                            hh_ones = 1
                            hh_tens = 0
                        else:
                            hh_ones += 1
                    else:
                        # R5_hour_rollover_am_pm: 12 -> 1, toggle pm
                        hh_ones = 1
                        hh_tens = 0
                        pm_state = 1 - pm_state
        # R6_hold: ena==0 and not reset -> no change

        hh = (hh_tens << 4) | hh_ones
        mm = (mm_tens << 4) | mm_ones
        ss = (ss_tens << 4) | ss_ones

        results.append({
            'pm': pm_state,
            'hh': hh,
            'mm': mm,
            'ss': ss,
        })

    return results
