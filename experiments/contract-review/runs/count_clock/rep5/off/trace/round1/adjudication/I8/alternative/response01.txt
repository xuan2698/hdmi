def reference(vectors):
    # State: ss_ones, ss_tens, mm_ones, mm_tens, hh_ones, hh_tens, pm_state
    # Initial state (power-up unspecified, but we need concrete values; use 12:00 AM)
    # R1_reset: reset sets to 12:00 AM => hh=0x12, mm=0x00, ss=0x00, pm=0
    ss_ones = 0
    ss_tens = 0
    mm_ones = 0
    mm_tens = 0
    hh_ones = 2
    hh_tens = 1
    pm_state = 0

    results = []
    for row in vectors:
        reset = row.get('reset', 0)
        ena = row.get('ena', 0)

        # Compute next state based on current state and inputs (sampled at posedge)
        if reset == 1:
            # R1_reset: reset to 12:00 AM
            n_ss_ones = 0
            n_ss_tens = 0
            n_mm_ones = 0
            n_mm_tens = 0
            n_hh_ones = 2
            n_hh_tens = 1
            n_pm = 0
        elif ena == 1:
            # R2_seconds, R3_seconds_rollover, R4_minutes_rollover, R5_hour_rollover_am_pm
            ss_val = ss_tens * 10 + ss_ones
            mm_val = mm_tens * 10 + mm_ones
            hh_val = hh_tens * 10 + hh_ones

            if ss_val != 59:
                # R2_seconds: increment seconds
                n_ss_ones = ss_ones + 1
                n_ss_tens = ss_tens
                if n_ss_ones == 10:
                    n_ss_ones = 0
                    n_ss_tens = ss_tens + 1
                n_mm_ones = mm_ones
                n_mm_tens = mm_tens
                n_hh_ones = hh_ones
                n_hh_tens = hh_tens
                n_pm = pm_state
            elif mm_val != 59:
                # R3_seconds_rollover: seconds roll to 00, minutes increment
                n_ss_ones = 0
                n_ss_tens = 0
                n_mm_ones = mm_ones + 1
                n_mm_tens = mm_tens
                if n_mm_ones == 10:
                    n_mm_ones = 0
                    n_mm_tens = mm_tens + 1
                n_hh_ones = hh_ones
                n_hh_tens = hh_tens
                n_pm = pm_state
            elif hh_val != 12:
                # R4_minutes_rollover: mm rolls to 00, hour increments
                n_ss_ones = 0
                n_ss_tens = 0
                n_mm_ones = 0
                n_mm_tens = 0
                # Increment hour
                if hh_val == 11:
                    # 11 -> 12, pm toggles per candidate interpretation
                    n_hh_ones = 2
                    n_hh_tens = 1
                    n_pm = 1 - pm_state  # pm toggles when hour rolls from 11 to 12
                else:
                    n_hh_ones = hh_ones + 1
                    n_hh_tens = hh_tens
                    if n_hh_ones == 10:
                        n_hh_ones = 0
                        n_hh_tens = hh_tens + 1
                    n_pm = pm_state
            else:
                # R5_hour_rollover_am_pm: 12:59:59 -> 1:00:00, pm toggles
                n_ss_ones = 0
                n_ss_tens = 0
                n_mm_ones = 0
                n_mm_tens = 0
                n_hh_ones = 1
                n_hh_tens = 0
                n_pm = 1 - pm_state
        else:
            # R6_hold: no change
            n_ss_ones = ss_ones
            n_ss_tens = ss_tens
            n_mm_ones = mm_ones
            n_mm_tens = mm_tens
            n_hh_ones = hh_ones
            n_hh_tens = hh_tens
            n_pm = pm_state

        # Output current state (before edge)
        hh_out = (hh_tens << 4) | hh_ones
        mm_out = (mm_tens << 4) | mm_ones
        ss_out = (ss_tens << 4) | ss_ones
        results.append({
            'pm': pm_state,
            'hh': hh_out,
            'mm': mm_out,
            'ss': ss_out
        })

        # Update state for next cycle
        ss_ones = n_ss_ones
        ss_tens = n_ss_tens
        mm_ones = n_mm_ones
        mm_tens = n_mm_tens
        hh_ones = n_hh_ones
        hh_tens = n_hh_tens
        pm_state = n_pm

    return results
