```python
def reference(vectors):
    # State: ss_ones, ss_tens, mm_ones, mm_tens, hh_ones, hh_tens, pm
    # Reset: hh=0x12, mm=0x00, ss=0x00, pm=0
    ss_ones, ss_tens = 0, 0
    mm_ones, mm_tens = 0, 0
    hh_ones, hh_tens = 2, 1  # 0x12
    pm = 0
    
    results = []
    for v in vectors:
        reset = v['reset']
        ena = v['ena']
        
        # Compute current outputs
        ss = (ss_tens << 4) | ss_ones
        mm = (mm_tens << 4) | mm_ones
        hh = (hh_tens << 4) | hh_ones
        
        # Apply reset or enable
        if reset:
            ss_ones, ss_tens = 0, 0
            mm_ones, mm_tens = 0, 0
            hh_ones, hh_tens = 2, 1
            pm = 0
        elif ena:
            # Increment seconds
            if ss_ones < 9:
                ss_ones += 1
            else:
                ss_ones = 0
                if ss_tens < 5:
                    ss_tens += 1
                else:
                    ss_tens = 0
                    # Roll minutes
                    if mm_ones < 9:
                        mm_ones += 1
                    else:
                        mm_ones = 0
                        if mm_tens < 5:
                            mm_tens += 1
                        else:
                            mm_tens = 0
                            # Roll hours
                            hh_val = (hh_tens << 4) | hh_ones
                            if hh_val == 0x12:  # 12 -> 1
                                hh_ones, hh_tens = 1, 0
                                pm = 1 - pm
                            elif hh_val == 0x11:  # 11 -> 12
                                hh_ones, hh_tens = 2, 1
                                pm = 1 - pm
                            else:
                                # Increment hour
                                if hh_ones < 9:
                                    hh_ones += 1
                                else:
                                    hh_ones = 0
                                    hh_tens += 1
        
        # Output current state (after update)
        ss = (ss_tens << 4) | ss_ones
        mm = (mm_tens << 4) | mm_ones
        hh = (hh_tens << 4) | hh_ones
        results.append({'pm': pm, 'hh': hh, 'mm': mm, 'ss': ss})
    
    return results
```
