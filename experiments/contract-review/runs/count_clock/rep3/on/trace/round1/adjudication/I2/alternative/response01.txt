def reference(vectors):
    # Initial power-up convention: use reset values for concrete unsigned outputs.
    pm = 0
    hh = 0x12
    mm = 0x00
    ss = 0x00
    cur_reset = 0
    cur_ena = 0
    result = []

    for row in vectors:
        if isinstance(row, dict):
            if "reset" in row and row["reset"] is not None:
                cur_reset = int(row["reset"]) & 1
            if "ena" in row and row["ena"] is not None:
                cur_ena = int(row["ena"]) & 1

        if cur_reset == 1:
            # reset_priority: synchronous reset has priority over ena.
            pm = 0
            hh = 0x12
            mm = 0x00
            ss = 0x00
        else:
            if cur_ena == 0:
                # hold_clock: hold all state when ena is low.
                pass
            else:
                # increment_clock: advance one displayed second in BCD.
                carry_ss = False
                if ss == 0x59:
                    ss = 0x00
                    carry_ss = True
                else:
                    lower = ss & 0x0F
                    upper = ss & 0xF0
                    if lower == 9:
                        ss = upper + 0x10
                    else:
                        ss = ss + 1

                if carry_ss:
                    carry_mm = False
                    if mm == 0x59:
                        mm = 0x00
                        carry_mm = True
                    else:
                        lower = mm & 0x0F
                        upper = mm & 0xF0
                        if lower == 9:
                            mm = upper + 0x10
                        else:
                            mm = mm + 1

                    if carry_mm:
                        # increment_clock override: toggle pm on hh 0x12 -> 0x01,
                        # and do not toggle pm on hh 0x11 -> 0x12.
                        if hh == 0x12:
                            hh = 0x01
                            pm = pm ^ 1
                        else:
                            lower = hh & 0x0F
                            upper = hh & 0xF0
                            if lower == 9:
                                hh = upper + 0x10
                            else:
                                hh = hh + 1

        result.append({
            "pm": pm & 1,
            "hh": hh & 0xFF,
            "mm": mm & 0xFF,
            "ss": ss & 0xFF
        })

        if isinstance(row, dict) and "mid" in row:
            mid = row["mid"]
            if isinstance(mid, dict):
                if "reset" in mid and mid["reset"] is not None:
                    cur_reset = int(mid["reset"]) & 1
                if "ena" in mid and mid["ena"] is not None:
                    cur_ena = int(mid["ena"]) & 1

    return result
