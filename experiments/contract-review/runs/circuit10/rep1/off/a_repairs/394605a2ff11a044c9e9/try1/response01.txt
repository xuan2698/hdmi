def reference(vectors):
    # Contract implementation.
    # R1: state <= q on posedge (visible next cycle). state is the flip-flop.
    # R2: q = state XOR a XOR b (combinational, same cycle).
    # Reset: none. Power-up unspecified -> use 0.
    # Mid changes occur after sampling; carried to next row.

    state = 0  # power-up unspecified; concrete default 0
    out = []

    for vec in vectors:
        a = vec.get('a', 0) & 1
        b = vec.get('b', 0) & 1

        # R2: combinational output q = state XOR a XOR b
        q = (state ^ a ^ b) & 1

        # Emit outputs for this row (sampled after posedge settled)
        out.append({'q': q, 'state': state})

        # R1: on posedge, state captures current q (visible next cycle)
        state = q

    return out
