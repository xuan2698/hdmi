def reference(vectors):
    # Contract implementation.
    # R1: state <= a AND b on posedge (next_cycle visible).
    # R2 (resolved): q = a AND b (combinational).
    # No reset. Power-up state unspecified -> use 0 as concrete default.
    # Mid changes occur after sampling; carried to next row start.
    # We process each row: sample inputs at posedge, compute outputs,
    # then apply mid changes for the next row.

    state = 0  # concrete power-up default (unspecified in spec)
    results = []

    for vec in vectors:
        a = vec.get('a', 0)
        b = vec.get('b', 0)

        # R2: q = a AND b (combinational)
        q = a & b

        # Outputs visible for this row (state is current stored value)
        results.append({'q': q & 1, 'state': state & 1})

        # R1: on posedge, state <= a AND b
        state = a & b

        # Mid changes: apply after sampling, carried to next row
        mid = vec.get('mid')
        if mid:
            # mid is a dict of input changes applied after sampling
            pass  # values already carried via next row's a/b

    return results
