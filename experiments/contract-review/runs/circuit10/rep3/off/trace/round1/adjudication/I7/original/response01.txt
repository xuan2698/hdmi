def reference(vectors):
    # Contract implementation.
    # R1: state is a single flip-flop updated on posedge clk to a XOR b.
    # R2: q is combinational passthrough of state (same cycle).
    # Reset: none; power-up state is unspecified -> use 0 as concrete value.
    state = 0
    out = []
    for row in vectors:
        a = row.get('a', 0)
        b = row.get('b', 0)
        # R2: q = state (combinational, same cycle)
        q = state & 1
        out.append({'q': q, 'state': state & 1})
        # R1: posedge update -> next state = a XOR b
        state = (a ^ b) & 1
    return out
