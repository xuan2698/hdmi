def reference(vectors):
    # Contract implementation.
    # R1: state is a single flip-flop updated on posedge clk to a XOR b.
    # R2: q = state (combinational passthrough).
    # Power-up state is unspecified; we model it as 0 (concrete value required).
    # Mid changes occur after sampling and carry to next row; we process each
    # row's inputs as the values present at the sampling posedge.

    state = 0  # R1: initial flip-flop value (unspecified -> concrete 0)
    out = []

    for row in vectors:
        a = row.get('a', 0)
        b = row.get('b', 0)

        # R1: next state = a XOR b, sampled on posedge.
        # The row's inputs are driven before the sampled edge, so they
        # determine the new state.
        state = (a ^ b) & 1

        # R2: q = state, combinational passthrough (same cycle).
        q = state

        out.append({'q': q & 1, 'state': state & 1})

    return out
