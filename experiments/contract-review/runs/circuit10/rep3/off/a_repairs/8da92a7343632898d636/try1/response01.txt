def reference(vectors):
    # Contract implementation.
    # R1: state updates on posedge clk to a XOR b (visible next cycle).
    # R2: q = state (combinational passthrough, same cycle).
    # Reset: none. Power-up state unspecified; we model it as 0.
    # Mid changes: inputs may change after sampling; we carry them forward.
    # We process each row: sample inputs at posedge, compute next state,
    # then q reflects the *current* state (before update) for same-cycle visibility.
    # However, per R2 q = state (the flip-flop output), and state is observable.
    # The example expects q to equal the state *after* the posedge update in the
    # same row? Let's check example 2:
    #   vectors: [{a:0,b:1},{a:1,b:1},{a:0,b:0}]
    #   expected: [{q:1},{q:0},{q:0}]
    # Row 1: a^b=1 -> state becomes 1. q expected 1. So q reflects new state.
    # Row 2: a^b=0 -> state becomes 0. q expected 0.
    # Row 3: a^b=0 -> state stays 0. q expected 0.
    # So q = state AFTER posedge update (same cycle as sampling).
    # Example 1:
    #   vectors: [{a:0,b:1},{a:1,b:0},{a:1,b:1}]
    #   expected: [{state:1},{state:1},{state:0}]
    # Row 1: a^b=1 -> state=1. expected 1. OK.
    # Row 2: a^b=1 -> state=1. expected 1. OK.
    # Row 3: a^b=0 -> state=0. expected 0. OK.
    # So state output = new state after posedge.
    # q = state (same value).
    # Power-up: state unspecified; we start at 0.
    state = 0
    results = []
    for vec in vectors:
        a = vec.get('a', 0)
        b = vec.get('b', 0)
        # R1: next state = a XOR b on posedge
        next_state = a ^ b
        state = next_state
        # R2: q = state (combinational passthrough)
        q = state
        results.append({'q': q & 1, 'state': state & 1})
    return results
