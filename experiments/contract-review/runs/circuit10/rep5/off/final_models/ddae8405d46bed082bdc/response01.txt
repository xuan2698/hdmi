def reference(vectors):
    # Contract implementation.
    # r1: q is the registered value of state (q equals state after each posedge).
    # r2: next state = (a XOR b) XOR state, sampled at posedge.
    # reset: none; power-up state unspecified -> start at 0.
    # mid: optional mid input changes occur AFTER sampling, before the other edge,
    #      and are carried to the start of the next row.

    state = 0  # power-up unspecified; choose 0
    q = 0
    out = []

    for vec in vectors:
        a = vec.get('a', 0) & 1
        b = vec.get('b', 0) & 1

        # r1: q reflects current registered state (visible after posedge)
        q = state

        # r2: compute next state at posedge
        state_next = (a ^ b) ^ state

        out.append({'q': q & 1, 'state': state & 1})

        # posedge update
        state = state_next

    return out
