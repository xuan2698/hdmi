def reference(vectors):
    # Contract implementation.
    # State: out (32-bit sticky falling-edge flags), prev_in (32-bit previous input).
    # R2: prev_in initialized to 0 before the first sampled cycle.
    out = 0
    prev_in = 0
    results = []
    for row in vectors:
        reset = row.get('reset', 0) & 1
        inp = row.get('in', 0) & 0xFFFFFFFF
        # R1_reset_clear: synchronous reset clears out on posedge (priority 1).
        if reset == 1:
            out = 0
        else:
            # R2_capture_falling: set sticky bits where prev_in bit is 1 and in bit is 0.
            falling = prev_in & (~inp & 0xFFFFFFFF)
            out = (out | falling) & 0xFFFFFFFF
            # R3_hold: bits not falling remain unchanged (implicit in OR above).
        # Update prev_in for next cycle (sampled value).
        prev_in = inp
        results.append({'out': out & 0xFFFFFFFF})
    return results
