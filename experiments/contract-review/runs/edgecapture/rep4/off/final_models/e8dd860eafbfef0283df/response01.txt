def reference(vectors):
    # Contract implementation.
    # State: out_reg (32-bit sticky flags), in_prev (32-bit previous input),
    #        initialized flag (whether in_prev has been seeded after reset release).
    # Timing: sample at posedge clk. Each row's inputs are driven before the edge.
    # Mid changes occur AFTER sampling and are carried to the start of the next row.
    # Outputs are registered: visible next_cycle (per R2/R3 examples).

    out_reg = 0          # R1_reset_clear: reset value 0
    in_prev = 0
    initialized = False  # in_prev seeded on first cycle after reset release

    results = []

    for row in vectors:
        reset = row.get("reset", 0) & 1
        in_val = row.get("in", 0) & 0xFFFFFFFF

        # --- Sampling at posedge ---
        if reset == 1:
            # R1_reset_clear: synchronous active-high reset clears out_reg.
            out_reg = 0
            # After reset, in_prev must be re-seeded to avoid spurious capture.
            initialized = False
            # in_prev is updated to current in for next cycle bookkeeping.
            in_prev = in_val
        else:
            if not initialized:
                # First cycle after reset release: seed in_prev to current in,
                # so no spurious 1->0 capture on this cycle.
                in_prev = in_val
                initialized = True
                # out_reg unchanged (holds, which is 0 after reset).
            else:
                # R2_capture_falling: bits that were 1 prev cycle and 0 now.
                falling = (~in_val) & in_prev & 0xFFFFFFFF
                # R3_hold: sticky OR (idempotent set, already-1 bits remain 1).
                out_reg = (out_reg | falling) & 0xFFFFFFFF
                in_prev = in_val

        # Output visible next_cycle: report current registered out_reg.
        results.append({"out": out_reg & 0xFFFFFFFF})

        # Mid changes: applied AFTER sampling, carried to next row.
        # (No mid changes present in this contract's vectors; if present,
        # they would modify in_val for the next row's start. We honor the
        # contract by not applying them before sampling.)
        if "mid" in row and isinstance(row["mid"], dict):
            # Mid changes affect the input seen at the start of the next row,
            # but the next row's "in" already carries that value per protocol.
            pass

    return results
