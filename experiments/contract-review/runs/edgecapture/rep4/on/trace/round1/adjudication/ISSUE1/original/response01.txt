def reference(vectors):
    # State registers:
    # out_reg: 32-bit sticky capture register (init 0)
    # prev_in: 32-bit previous sampled input (init 0)
    MASK = (1 << 32) - 1
    out_reg = 0
    prev_in = 0
    carried_reset = 0
    carried_in = 0
    results = []
    for row in vectors:
        # Start with carried values, then apply row's driven values before the sampled edge.
        pre_reset = carried_reset
        pre_in = carried_in
        if isinstance(row, dict):
            if 'reset' in row:
                pre_reset = int(row['reset']) & 1
            if 'in' in row:
                pre_in = int(row['in']) & MASK
        # At the sampled posedge:
        if pre_reset == 1:
            # sync_active_high_reset: reset clears out and overrides any capture.
            out_reg = 0
        else:
            # capture_sticky_one_to_zero: set bits where prev_in=1 and in=0, sticky.
            out_reg = (out_reg | (prev_in & (~pre_in))) & MASK
        # prev_in updates every posedge, including reset cycles.
        prev_in = pre_in
        results.append({'out': out_reg & MASK})
        # After sampling, update carried values to current input values, then apply mid changes.
        carried_reset = pre_reset
        carried_in = pre_in
        if isinstance(row, dict):
            mid = None
            if 'mid' in row:
                mid = row['mid']
            elif 'mid_changes' in row:
                mid = row['mid_changes']
            changes = []
            if isinstance(mid, dict):
                changes = [mid]
            elif isinstance(mid, list):
                changes = mid
            for ch in changes:
                if isinstance(ch, dict):
                    if 'reset' in ch:
                        carried_reset = int(ch['reset']) & 1
                    if 'in' in ch:
                        carried_in = int(ch['in']) & MASK
                elif isinstance(ch, (tuple, list)) and len(ch) >= 2:
                    k, v = ch[0], ch[1]
                    if k == 'reset':
                        carried_reset = int(v) & 1
                    elif k == 'in':
                        carried_in = int(v) & MASK
            # Also support flat mid_* keys.
            for k in row:
                if isinstance(k, str) and k.startswith('mid_'):
                    port = k[4:]
                    if port == 'reset':
                        carried_reset = int(row[k]) & 1
                    elif port == 'in':
                        carried_in = int(row[k]) & MASK
    return results
