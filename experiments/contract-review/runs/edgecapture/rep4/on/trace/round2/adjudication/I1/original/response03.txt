def reference(vectors):
    MASK = (1 << 32) - 1
    out_reg = 0
    prev_in = 0
    cur_reset = 0
    cur_in = 0
    result = []

    for row in vectors:
        if row is not None:
            if 'reset' in row:
                cur_reset = row['reset'] & 1
            if 'in' in row:
                cur_in = row['in'] & MASK

        # Rule sync_active_high_reset: active-high synchronous reset clears out; priority over capture.
        if cur_reset == 1:
            out_reg = 0
        else:
            # Rule capture_sticky_one_to_zero: sticky per-bit capture of sampled 1-to-0 transitions.
            transitions = prev_in & (~cur_in & MASK)
            out_reg = (out_reg | transitions) & MASK

        prev_in = cur_in & MASK
        result.append({'out': out_reg & MASK})

        # Optional mid changes occur after sampling and carry to the next row.
        mid = None
        if row is not None:
            if 'mid' in row:
                mid = row['mid']
            elif 'mid_changes' in row:
                mid = row['mid_changes']
            elif 'mid_inputs' in row:
                mid = row['mid_inputs']

        if mid:
            if isinstance(mid, dict):
                if 'reset' in mid:
                    cur_reset = mid['reset'] & 1
                if 'in' in mid:
                    cur_in = mid['in'] & MASK
            elif isinstance(mid, (list, tuple)):
                for m in mid:
                    if isinstance(m, dict):
                        if 'reset' in m:
                            cur_reset = m['reset'] & 1
                        if 'in' in m:
                            cur_in = m['in'] & MASK
                    elif isinstance(m, (list, tuple)) and len(m) >= 2:
                        name = m[0]
                        val = m[1]
                        if name == 'reset':
                            cur_reset = val & 1
                        elif name == 'in':
                            cur_in = val & MASK

    return result
