def reference(vectors):
    # Rule capture_sticky_one_to_zero:
    # out_reg is the 32-bit sticky capture register.
    # Power-up is unspecified; use concrete zero initialization.
    out_reg = 0

    # prev_in holds the in value sampled at the immediately preceding posedge.
    prev_in = 0

    # Carried input values for the start of the next row, including optional mid changes.
    cur_reset = 0
    cur_in = 0

    result = []

    for row in vectors:
        # Each row's values are driven before the sampled posedge.
        if "reset" in row:
            cur_reset = row["reset"]
        if "in" in row:
            cur_in = row["in"] & 0xFFFFFFFF

        # Sample at the posedge.
        # Rule sync_active_high_reset: synchronous active-high reset clears out.
        if cur_reset == 1:
            out_reg = 0
        else:
            # Rule capture_sticky_one_to_zero:
            # Capture per-bit sampled 1-to-0 transitions, sticky until reset.
            ones_to_zeros = prev_in & ((~cur_in) & 0xFFFFFFFF)
            out_reg = (out_reg | ones_to_zeros) & 0xFFFFFFFF

        # prev_in updates on every posedge, including reset cycles.
        prev_in = cur_in

        # Output is visible in the same cycle as the sampled posedge.
        result.append({"out": out_reg & 0xFFFFFFFF})

        # Optional mid-row changes occur after sampling and before the next posedge.
        # They update carried input values for the start of the next row but do not
        # alter out until the next sampled posedge.
        if "mid" in row:
            mid = row["mid"]
            if "reset" in mid:
                cur_reset = mid["reset"]
            if "in" in mid:
                cur_in = mid["in"] & 0xFFFFFFFF

    return result
