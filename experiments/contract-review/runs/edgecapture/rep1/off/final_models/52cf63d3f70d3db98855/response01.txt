def reference(vectors):
    # State: out_reg (32-bit sticky flag), prev_in (32-bit previous input)
    # R1_reset_clear: reset==1 at posedge -> out cleared to 0 (priority 1)
    # R2_capture_falling_edge: reset==0 and (prev_in & ~in)!=0 -> out |= (prev_in & ~in)
    # R3_hold_until_reset: reset==0 and no falling edge -> out holds
    # prev_in updates every posedge to current in (after sampling)
    out_reg = 0
    prev_in = 0
    results = []
    for row in vectors:
        reset = row.get('reset', 0) & 1
        cur_in = row.get('in', 0) & 0xFFFFFFFF
        # Sample at posedge: compute next state based on current inputs and prev_in
        if reset == 1:
            # R1_reset_clear
            next_out = 0
        else:
            falling = prev_in & (~cur_in & 0xFFFFFFFF)
            if falling != 0:
                # R2_capture_falling_edge
                next_out = out_reg | falling
            else:
                # R3_hold_until_reset
                next_out = out_reg
        out_reg = next_out & 0xFFFFFFFF
        prev_in = cur_in
        results.append({'out': out_reg})
    return results
