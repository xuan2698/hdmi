def reference(vectors):
    # State: out_reg (sticky falling-edge flags), prev_in (previous input vector)
    # R1_reset_clear: reset==1 -> out_reg cleared (priority 1)
    # R2_capture_falling_edge: reset==0 and (prev_in & ~in) -> set bits (priority 2)
    # R3_hold_until_reset: reset==0 and no falling edge -> hold out_reg (priority 3)
    # R4_prev_in_tracks_all_cycles: prev_in updated only when reset==0 (priority 4)
    out_reg = 0
    prev_in = 0
    results = []
    for row in vectors:
        reset = row.get('reset', 0) & 1
        in_val = row.get('in', 0) & 0xFFFFFFFF
        # Sampling at posedge clk
        if reset == 1:
            # R1_reset_clear
            out_reg = 0
            # R4_prev_in_tracks_all_cycles: prev_in holds during reset
        else:
            # R2_capture_falling_edge
            falling = prev_in & (~in_val & 0xFFFFFFFF)
            if falling != 0:
                out_reg = (out_reg | falling) & 0xFFFFFFFF
            # R3_hold_until_reset: out_reg unchanged otherwise
            # R4_prev_in_tracks_all_cycles
            prev_in = in_val
        results.append({'out': out_reg & 0xFFFFFFFF})
        # mid changes occur AFTER sampling; carried to next row start
        if 'mid' in row and isinstance(row['mid'], dict):
            if 'in' in row['mid']:
                in_val = row['mid']['in'] & 0xFFFFFFFF
            if 'reset' in row['mid']:
                reset = row['mid']['reset'] & 1
            # mid changes affect next row's starting values; but prev_in
            # is only updated at posedge, so mid 'in' does not alter prev_in.
    return results
