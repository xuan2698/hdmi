def reference(vectors):
    # State: out_reg (32-bit sticky flag), prev_in (32-bit previous input)
    out_reg = 0
    prev_in = 0
    results = []
    for row in vectors:
        reset = row.get('reset', 0) & 1
        inp = row.get('in', 0) & 0xFFFFFFFF
        # R1_reset_clear: synchronous reset clears out_reg (priority 1)
        if reset == 1:
            out_reg = 0
        else:
            # R2_capture_falling_edge: set bits where prev_in=1 and in=0
            falling = prev_in & (~in) & 0xFFFFFFFF
            out_reg = (out_reg | falling) & 0xFFFFFFFF
            # R3_hold_until_reset: out_reg retains previous value otherwise
        # Output visible after posedge (same cycle for reset, next for capture)
        results.append({'out': out_reg & 0xFFFFFFFF})
        # Update prev_in after sampling (mid changes carried to next row)
        prev_in = inp
    return results
