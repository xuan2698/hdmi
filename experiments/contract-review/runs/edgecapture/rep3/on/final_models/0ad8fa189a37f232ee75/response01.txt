def reference(vectors):
    MASK = (1 << 32) - 1
    out_reg = 0       # 32-bit capture register
    prev_in = 0       # previous sampled input history
    cur_reset = 0
    cur_in = 0
    result = []

    for row in vectors:
        mid = None

        # Drive row inputs before the sampled posedge.
        if isinstance(row, dict):
            if "reset" in row and row["reset"] is not None:
                cur_reset = int(row["reset"]) & 1
            if "in" in row and row["in"] is not None:
                cur_in = int(row["in"]) & MASK
            if "mid" in row:
                mid = row["mid"]
        elif isinstance(row, (list, tuple)):
            if len(row) >= 1:
                cur_reset = int(row[0]) & 1
            if len(row) >= 2:
                cur_in = int(row[1]) & MASK
            if len(row) >= 3:
                mid = row[2]

        sampled_reset = cur_reset & 1
        sampled_in = cur_in & MASK

        if sampled_reset == 1:
            # sync_reset_clear_out: active-high synchronous reset; out becomes 0.
            out_reg = 0
        else:
            # falling_edge_capture_and_hold: capture 1-to-0 transitions and hold set bits.
            falling = prev_in & ((~sampled_in) & MASK)
            out_reg = (out_reg | falling) & MASK

        # Contract assumption for uncertain rule: prev_in updates on every posedge,
        # including reset cycles.
        prev_in = sampled_in

        # Output is visible in the same sampled cycle after the posedge update.
        result.append({"out": out_reg & MASK})

        # mid changes occur AFTER sampling, before the other edge, and carry to next row.
        if mid is not None:
            if isinstance(mid, dict):
                if "reset" in mid and mid["reset"] is not None:
                    cur_reset = int(mid["reset"]) & 1
                if "in" in mid and mid["in"] is not None:
                    cur_in = int(mid["in"]) & MASK
            elif isinstance(mid, (list, tuple)):
                if len(mid) >= 1:
                    cur_reset = int(mid[0]) & 1
                if len(mid) >= 2:
                    cur_in = int(mid[1]) & MASK

    return result
