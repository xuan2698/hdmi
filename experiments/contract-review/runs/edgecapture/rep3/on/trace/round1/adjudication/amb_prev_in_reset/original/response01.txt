def reference(vectors):
    # sync_reset_clear_out: active high synchronous reset clears out_reg at posedge.
    # falling_edge_capture_and_hold: capture 1->0 transitions using prev_in, hold set bits.
    out_reg = 0
    prev_in = 0
    cur = {'reset': 0, 'in': 0}
    result = []
    for row in vectors:
        # Each row's values are driven before the sampled posedge.
        for k, v in row.items():
            if k != 'mid':
                cur[k] = v

        reset = cur.get('reset', 0) & 1
        inp = cur.get('in', 0) & 0xFFFFFFFF

        if reset == 1:
            # sync_reset_clear_out: out becomes 0 on reset at sampled posedge.
            out_next = 0
        else:
            # falling_edge_capture_and_hold: out_next = out | (prev_in & ~in)
            out_next = out_reg | (prev_in & (~in & 0xFFFFFFFF))
            out_next &= 0xFFFFFFFF

        # prev_in updates on every posedge, including reset cycles (contract assumption).
        prev_in_next = inp

        result.append({'out': out_next})

        # Update sequential state after sampling the posedge.
        out_reg = out_next
        prev_in = prev_in_next

        # mid changes occur after sampling, before the next edge, and are carried
        # to the start of the next row as pre-edge input values.
        if 'mid' in row and row['mid'] is not None:
            for k, v in row['mid'].items():
                cur[k] = v

    return result
