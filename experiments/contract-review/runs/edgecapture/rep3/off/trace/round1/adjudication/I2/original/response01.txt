def reference(vectors):
    # Contract implementation.
    # State: out_reg (32-bit), prev_in (32-bit) for falling-edge detection.
    # Timing: sample at posedge clk after settling; mid changes occur AFTER
    # sampling and are carried to the start of the next row.
    # Power-up: out_reg = 0, prev_in = 0 (concrete, unsigned).

    out_reg = 0
    prev_in = 0
    results = []

    for row in vectors:
        reset = row.get("reset", 0) & 1
        in_val = row.get("in", 0) & 0xFFFFFFFF

        # R1_reset_clear: reset==1 -> out cleared (same_cycle, priority 1)
        # R2_capture_falling: reset==0 and (prev_in & ~in)!=0 -> out |= prev_in & ~in
        # R3_hold: reset==0 and no falling edge -> out unchanged
        if reset == 1:
            out_reg = 0
        else:
            falling = prev_in & (~in_val & 0xFFFFFFFF)
            if falling != 0:
                out_reg = (out_reg | falling) & 0xFFFFFFFF
            # else hold

        # prev_in updates to current in for next cycle's edge detection
        prev_in = in_val

        results.append({"out": out_reg & 0xFFFFFFFF})

    return results
