def reference(vectors):
    # Contract implementation.
    # State: out_reg (32-bit), prev_in (32-bit, previous sampled input).
    # Power-up: out_reg = 0, prev_in = 0 (unspecified but concrete).
    # R1_reset_clear: reset==1 -> out <= 0 (sync, posedge, same cycle visible).
    # R2_capture_falling: reset==0 and (prev_in & ~in)!=0 -> out <= out | (prev_in & ~in).
    # R3_hold: reset==0 and no falling -> out <= out.
    # Global interpretation: on first posedge with reset==0, capture falling edge
    #   rather than clearing (prev_in initialized to 0, so first cycle prev_in&~in==0
    #   unless mid changes set prev_in; we honor capture over clear on first release).
    # Mid changes occur AFTER sampling, before next edge, carried to next row.

    out_reg = 0
    prev_in = 0
    first_released = True  # tracks first posedge with reset==0
    results = []

    for row in vectors:
        reset = row.get('reset', 0) & 1
        cur_in = row.get('in', 0) & 0xFFFFFFFF

        # Sampling at posedge clk (values driven before edge).
        if reset == 1:
            # R1_reset_clear
            out_reg = 0
            # reset also resets first_released tracking? Keep flag; reset clears state.
            first_released = True
        else:
            fall = prev_in & (~cur_in & 0xFFFFFFFF)
            if first_released:
                # Global interpretation: on first posedge with reset==0,
                # capture falling edge rather than clearing.
                if fall != 0:
                    out_reg = out_reg | fall  # R2_capture_falling
                else:
                    out_reg = out_reg  # R3_hold (no change)
                first_released = False
            else:
                if fall != 0:
                    out_reg = out_reg | fall  # R2_capture_falling
                else:
                    out_reg = out_reg  # R3_hold

        # prev_in updates to current sampled input for next cycle comparison.
        prev_in = cur_in

        # Mid changes: occur AFTER sampling, before other edge, carried to next row.
        # They affect prev_in for the next row's comparison (i.e., the "previous"
        # value seen at next posedge is the mid-updated value).
        if 'mid' in row and row['mid'] is not None:
            mid = row['mid']
            if isinstance(mid, dict):
                if 'in' in mid:
                    prev_in = mid['in'] & 0xFFFFFFFF
            else:
                prev_in = mid & 0xFFFFFFFF

        results.append({'out': out_reg & 0xFFFFFFFF})

    return results
