def reference(vectors):
    # Contract implementation.
    # State: out_reg (32-bit), prev_in (32-bit), initialized per R2 uncertainty.
    # We choose prev_in initialized to 0 (no first-cycle fall detected) — one of
    # the allowed candidates for the R2 first-cycle uncertainty.
    out_reg = 0
    prev_in = 0
    results = []
    for row in vectors:
        reset = row.get('reset', 0) & 1
        cur_in = row.get('in', 0) & 0xFFFFFFFF
        # R1_reset_clear: synchronous reset clears out_reg (priority 1)
        if reset == 1:
            out_reg = 0
        else:
            # R2_capture_falling: capture 1->0 transitions
            fall = prev_in & (~cur_in & 0xFFFFFFFF)
            if fall != 0:
                out_reg = (out_reg | fall) & 0xFFFFFFFF
            # R3_hold: otherwise hold (no change)
        # prev_in updates after sampling (mid changes carried to next row)
        prev_in = cur_in
        results.append({'out': out_reg & 0xFFFFFFFF})
    return results
