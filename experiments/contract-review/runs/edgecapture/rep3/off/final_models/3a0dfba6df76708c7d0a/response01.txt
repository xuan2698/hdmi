def reference(vectors):
    # Contract implementation.
    # State: out_reg (32-bit), prev_in (32-bit, last sampled input).
    # Timing: sample at posedge clk after settling. Mid changes occur AFTER
    # sampling and are carried to the start of the next row.
    # Power-up: out_reg = 0, prev_in = 0 (concrete, unsigned).

    out_reg = 0
    prev_in = 0
    results = []

    for row in vectors:
        reset = int(row.get("reset", 0)) & 1
        in_val = int(row.get("in", 0)) & 0xFFFFFFFF

        # R1_reset_clear: reset==1 -> out cleared (sync, posedge, same cycle visible)
        if reset == 1:
            out_reg = 0
        else:
            # R2_capture_falling: detect 1->0 transitions vs prev_in, OR into out
            falling = prev_in & (~in_val & 0xFFFFFFFF)
            if falling != 0:
                out_reg = (out_reg | falling) & 0xFFFFFFFF
            # R3_hold: otherwise out_reg unchanged (explicit no-op)

        # Update prev_in to current sampled input (posedge sampling)
        prev_in = in_val

        # Mid changes occur AFTER sampling; carried to next row start.
        # (No mid field in this contract's rows; nothing to apply.)

        results.append({"out": out_reg & 0xFFFFFFFF})

    return results
