def reference(vectors):
    # State: out_reg (32-bit sticky flag), prev_in (32-bit previous input)
    # R1_reset_clear: reset==1 at posedge -> out=0 (same cycle visible)
    # R2_capture_falling: reset==0 and (prev_in & ~in)!=0 -> out |= (prev_in & ~in), visible next cycle
    # R3_hold: reset==0 and no falling edge -> out holds
    # global: prev_in initialized to all ones at time 0
    out_reg = 0
    prev_in = 0xFFFFFFFF  # global interpretation: init to all ones
    results = []
    for row in vectors:
        reset = row['reset'] & 1
        inp = row['in'] & 0xFFFFFFFF
        # At posedge: compute next state
        if reset == 1:
            # R1_reset_clear
            out_reg = 0
        else:
            # R2_capture_falling / R3_hold
            falling = prev_in & (~inp & 0xFFFFFFFF)
            out_reg = (out_reg | falling) & 0xFFFFFFFF
        # Output visible after this posedge (same cycle for reset, next cycle for capture)
        results.append({'out': out_reg & 0xFFFFFFFF})
        # prev_in updated to current input (mid changes carried to next row start)
        prev_in = inp
    return results
