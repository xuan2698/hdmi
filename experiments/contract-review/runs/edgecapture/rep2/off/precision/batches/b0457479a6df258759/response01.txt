def reference(vectors):
    out_reg = 0
    prev_in = 0
    results = []
    for v in vectors:
        reset = v['reset']
        in_val = v['in']
        # output for this cycle
        out = 0 if reset == 1 else out_reg
        results.append({'out': out})
        # compute next state
        if reset == 1:

        prev_in_next = in_val & 0xFFFFFFFF
        out_reg = out_reg_next
        prev_in = prev_in_next
    return results
