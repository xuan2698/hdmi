def reference(vectors):
    # States: 0=WALK_LEFT, 1=WALK_RIGHT, 2=FALL, 3=DIG, 4=SPLAT
    # resume_dir: walking direction to resume after fall/dig (0=left, 1=right)
    # fall_count: number of cycles spent falling (incremented each cycle while falling)

    results = []
    state = 0
    fall_count = 0
    resume_dir = 0
    prev_areset = 0

    for row in vectors:
        areset = row.get('areset', 0)
        # Async positive-edge reset
        if areset == 1 and prev_areset == 0:
            state = 0
            fall_count = 0
            resume_dir = 0
        prev_areset = areset

        bump_left = row.get('bump_left', 0)
        bump_right = row.get('bump_right', 0)
        ground = row.get('ground', 0)
        dig = row.get('dig', 0)

        # Moore outputs from current state
        walk_left = 1 if state == 0 else 0
        walk_right = 1 if state == 1 else 0
        aaah = 1 if state == 2 else 0
        digging = 1 if state == 3 else 0

        results.append({
            'walk_left': walk_left,
            'walk_right': walk_right,
            'aaah': aaah,
            'digging': digging,
        })

        # Compute next state
        if state == 4:
            next_state, next_fall, next_resume = 4, 0, resume_dir
        elif state == 0:  # WALK_LEFT
            if ground == 0:
                next_state, next_fall, next_resume = 2, 1, 0
            elif dig == 1:
                next_state, next_fall, next_resume = 3, 0, 0
            elif bump_left == 1 or bump_right == 1:
                next_state, next_fall, next_resume = 1, 0, 0
            else:
                next_state, next_fall, next_resume = 0, 0, 0
        elif state == 1:  # WALK_RIGHT
            if ground == 0:
                next_state, next_fall, next_resume = 2, 1, 1
            elif dig == 1:
                next_state, next_fall, next_resume = 3, 0, 1
            elif bump_left == 1 or bump_right == 1:
                next_state, next_fall, next_resume = 0, 0, 1
            else:
                next_state, next_fall, next_resume = 1, 0, 1
        elif state == 2:  # FALL
            if ground == 1:
                if fall_count > 20:
                    next_state, next_fall, next_resume = 4, 0, resume_dir
                else:
                    next_state = resume_dir
                    next_fall = 0
                    next_resume = resume_dir
            else:
                next_state, next_fall, next_resume = 2, fall_count + 1, resume_dir
        elif state == 3:  # DIG

            else:
                next_state, next_fall, next_resume = 3, 0, resume_dir
        else:
            next_state, next_fall, next_resume = state, fall_count, resume_dir

        state = next_state
        fall_count = next_fall
        resume_dir = next_resume

    return results
