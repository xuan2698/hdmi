def reference(vectors):
    # State encoding
    WALK_LEFT, WALK_RIGHT, FALL_LEFT, FALL_RIGHT, DIG_LEFT, DIG_RIGHT, SPLAT = range(7)

    def outputs(state):
        # R2, R3, R4, R5, R6: combinational outputs per state
        if state == WALK_LEFT:
            return {"walk_left": 1, "walk_right": 0, "aaah": 0, "digging": 0}
        if state == WALK_RIGHT:
            return {"walk_left": 0, "walk_right": 1, "aaah": 0, "digging": 0}
        if state in (FALL_LEFT, FALL_RIGHT):
            return {"walk_left": 0, "walk_right": 0, "aaah": 1, "digging": 0}
        if state in (DIG_LEFT, DIG_RIGHT):
            return {"walk_left": 0, "walk_right": 0, "aaah": 0, "digging": 1}
        # SPLAT
        return {"walk_left": 0, "walk_right": 0, "aaah": 0, "digging": 0}

    state = WALK_LEFT
    fall_count = 0
    result = []

    for row in vectors:
        areset = row.get("areset", 0)
        bump_left = row.get("bump_left", 0)
        bump_right = row.get("bump_right", 0)
        ground = row.get("ground", 1)
        dig = row.get("dig", 0)

        # R1: asynchronous reset forces WALK_LEFT (visible same cycle)
        if areset == 1:
            state = WALK_LEFT
            fall_count = 0

        # Output for this cycle (combinational from current state)
        result.append(outputs(state))

        # Compute next state at posedge
        nxt = state
        nfall = fall_count

        if state == SPLAT:
            # R20: SPLAT stays SPLAT
            nxt = SPLAT
            nfall = 0
        elif state == WALK_LEFT:
            if ground == 0:
                # R7: fall precedence
                nxt = FALL_LEFT
                nfall = 1
            elif dig == 1:
                # R9: start digging
                nxt = DIG_LEFT
                nfall = 0
            elif bump_left == 1:
                # R11, R13: bump left (or both) -> walk right
                nxt = WALK_RIGHT
                nfall = 0
            else:
                nxt = WALK_LEFT
                nfall = 0
        elif state == WALK_RIGHT:
            if ground == 0:
                # R8
                nxt = FALL_RIGHT
                nfall = 1
            elif dig == 1:
                # R10
                nxt = DIG_RIGHT
                nfall = 0
            elif bump_right == 1:
                # R12, R14: bump right (or both) -> walk left
                nxt = WALK_LEFT
                nfall = 0
            else:
                nxt = WALK_RIGHT
                nfall = 0
        elif state == FALL_LEFT:
            if ground == 1:
                if fall_count > 20:
                    # R17: splat
                    nxt = SPLAT
                    nfall = 0
                else:
                    # R15: resume walking left
                    nxt = WALK_LEFT
                    nfall = 0
            else:
                nxt = FALL_LEFT
                nfall = fall_count + 1
        elif state == FALL_RIGHT:
            if ground == 1:
                if fall_count > 20:
                    # R17
                    nxt = SPLAT
                    nfall = 0
                else:
                    # R16
                    nxt = WALK_RIGHT
                    nfall = 0
            else:
                nxt = FALL_RIGHT
                nfall = fall_count + 1
        elif state == DIG_LEFT:
            if ground == 0:
                # R18: fall
                nxt = FALL_LEFT
                nfall = 1
            else:
                # R19: stay digging
                nxt = DIG_LEFT
                nfall = 0
        elif state == DIG_RIGHT:
            if ground == 0:
                # R18
                nxt = FALL_RIGHT
                nfall = 1
            else:
                # R19
                nxt = DIG_RIGHT
                nfall = 0

        state = nxt
        fall_count = nfall

    return result
