def reference(vectors):
    # State encoding
    WALK_LEFT, WALK_RIGHT, FALL_LEFT, FALL_RIGHT, DIG_LEFT, DIG_RIGHT, SPLAT = range(7)

    # Outputs per state (walk_left, walk_right, aaah, digging)
    outputs = {
        WALK_LEFT:  (1, 0, 0, 0),
        WALK_RIGHT: (0, 1, 0, 0),
        FALL_LEFT:  (0, 0, 1, 0),
        FALL_RIGHT: (0, 0, 1, 0),
        DIG_LEFT:   (0, 0, 0, 1),
        DIG_RIGHT:  (0, 0, 0, 1),
        SPLAT:      (0, 0, 0, 0),
    }

    state = WALK_LEFT
    fall_count = 0
    result = []

    for row in vectors:
        areset = row.get('areset', 0)
        bump_left = row.get('bump_left', 0)
        bump_right = row.get('bump_right', 0)
        ground = row.get('ground', 0)
        dig = row.get('dig', 0)

        # R1_reset: async reset to WALK_LEFT
        if areset == 1:
            state = WALK_LEFT
            fall_count = 0

        # Compute outputs (combinational, same cycle)
        wl, wr, aa, dg = outputs[state]
        result.append({
            'walk_left': wl,
            'walk_right': wr,
            'aaah': aa,
            'digging': dg,
        })

        # Compute next state (posedge)
        if state == SPLAT:
            # R10_splat_stay
            next_state = SPLAT
            next_fall = 0
        elif state in (WALK_LEFT, WALK_RIGHT):
            if ground == 0:
                # R7_walk_to_fall
                next_state = FALL_LEFT if state == WALK_LEFT else FALL_RIGHT
                next_fall = 0
            elif dig == 1:
                # R6_walk_transition: dig precedence
                next_state = DIG_LEFT if state == WALK_LEFT else DIG_RIGHT
                next_fall = 0
            elif bump_left == 1 or bump_right == 1:
                # R6_walk_transition: switch direction
                next_state = WALK_RIGHT if state == WALK_LEFT else WALK_LEFT
                next_fall = 0
            else:
                next_state = state
                next_fall = 0
        elif state in (FALL_LEFT, FALL_RIGHT):
            if ground == 1:
                # R8_fall_transition
                if fall_count > 20:
                    next_state = SPLAT
                    next_fall = 0
                else:
                    next_state = WALK_LEFT if state == FALL_LEFT else WALK_RIGHT
                    next_fall = 0
            else:
                next_state = state
                next_fall = fall_count + 1
        elif state in (DIG_LEFT, DIG_RIGHT):
            if ground == 0:
                # R9_dig_transition
                next_state = FALL_LEFT if state == DIG_LEFT else FALL_RIGHT
                next_fall = 0
            else:
                next_state = state
                next_fall = 0
        else:
            next_state = state
            next_fall = 0

        state = next_state
        fall_count = next_fall

    return result
