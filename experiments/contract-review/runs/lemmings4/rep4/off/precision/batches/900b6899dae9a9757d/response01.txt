We need to implement the Lemmings FSM as a Moore machine with async reset. Key details:

- States: WALK_LEFT, WALK_RIGHT, FALL_LEFT, FALL_RIGHT, DIG_LEFT, DIG_RIGHT, SPLAT
- Fall counter: counts cycles in fall state; if >20 when landing → SPLAT
- Priority: fall > dig > bump switch
- Async reset: areset=1 → WALK_LEFT immediately (same cycle)
- Outputs are combinational from state

The `reference(vectors)` function takes a list of input dicts (each cycle's inputs) and returns a list of output dicts. Since areset is async and visible same cycle, we check areset first each cycle: if areset=1, state becomes WALK_LEFT and outputs are walk_left=1.

Otherwise, we compute outputs from current state, then compute next state based on inputs.

Fall counter: we need to track how many cycles we've been falling. When entering FALL from WALK/DIG, counter starts at 0 (this cycle is the first fall cycle). def reference(vectors):
    WALK_LEFT, WALK_RIGHT, FALL_LEFT, FALL_RIGHT, DIG_LEFT, DIG_RIGHT, SPLAT = range(7)
    state = WALK_LEFT
    fall_count = 0
    results = []
    for v in vectors:
        areset = v['areset']
        bl = v['bump_left']
        br = v['bump_right']
        g = v['ground']
        d = v['dig']
        if areset:
            state = WALK_LEFT
            fall_count = 0
            results.append({'walk_left':1,'walk_right':0,'aaah':0,'digging':0})
            continue
        # outputs from current state
        if state == WALK_LEFT:
            out = {'walk_left':1,'walk_right':0,'aaah':0,'digging':0}
        elif state == WALK_RIGHT:
            out = {'walk_left':0,'walk_right':1,'aaah':0,'digging':0}
        elif state in (FALL_LEFT, FALL_RIGHT):
            out = {'walk_left':0,'walk_right':0,'aaah':1,'digging':0}
        elif state in (DIG_LEFT, DIG_RIGHT):
            out = {'walk_left':0,'walk_right':0,'aaah':0,'digging':1}
        else:
            out = {'walk_left':0,'walk_right':0,'aaah':0,'digging':0}
        results.append(out)
        # next state
        if state == WALK_LEFT:
            if g == 0:
                state = FALL_LEFT
                fall_count = 0
            elif d == 1:
                state = DIG_LEFT
            elif bl == 1 or br == 1:
                state = WALK_RIGHT
        elif state == WALK_RIGHT:
            if g == 0:
                state = FALL_RIGHT
                fall_count = 0
            elif d == 1:
                state = DIG_RIGHT
            elif bl == 1 or br == 1:
                state = WALK_LEFT
        elif state == FALL_LEFT:
            if g == 1:
                if fall_count > 20:
                    state = SPLAT
                else:
                    state = WALK_LEFT
            else:
                fall_count += 1
        elif state == FALL_RIGHT:
            if g == 1:
                if fall_count > 20:
                    state = SPLAT
                else:
                    state = WALK_RIGHT
            else:
                fall_count += 1
        elif state == DIG_LEFT:
            if g == 0:
                state = FALL_LEFT
                fall_count = 0
        elif state == DIG_RIGHT:
            if g == 0:
                state = FALL_RIGHT
                fall_count = 0
        # SPLAT stays
    return results

def reference(vectors):
    WL, WR, FL, FR, DL, DR, SP = 0, 1, 2, 3, 4, 5, 6
    state = WL
    fc = 0
    res = []
    for v in vectors:
        a = v['areset']
        bl = v['bump_left']
        br = v['bump_right']
        g = v['ground']
        d = v['dig']
        if a:
            state = WL
            fc = 0
            res.append({'walk_left':1,'walk_right':0,'aaah':0,'digging':0})
            continue
        if state == WL:
            res.append({'walk_left':1,'walk_right':0,'aaah':0,'digging':0})
        elif state == WR:
            res.append({'walk_left':0,'walk_right':1,'aaah':0,'digging':0})
        elif state == FL or state == FR:
            res.append({'walk_left':0,'walk_right':0,'aaah':1,'digging':0})
        elif state == DL or state == DR:
            res.append({'walk_left':0,'walk_right':0,'aaah':0,'digging':1})
        else:
            res.append({'walk_left':0,'walk_right':0,'aaah':0,'digging':0})
        if state == WL:
            if g == 0:
                state = FL
                fc = 0
            elif d == 1:
                state = DL
            elif bl == 1 or br == 1:
                state = WR
        elif state == WR:
            if g == 0:
                state = FR
                fc = 0
            elif d == 1:
                state = DR
            elif bl == 1 or br == 1:
                state = WL
        elif state == FL:
            if g == 1:
                if fc > 20:
                    state = SP
                else:
                    state = WL
            else:
                fc += 1
        elif state == FR:
            if g == 1:
                if fc > 20:
                    state = SP
                else:
                    state = WR
            else:

        elif state == DL:
            if g == 0:
                state = FL
                fc = 0
        elif state == DR:
            if g == 0:
                state = FR
                fc = 0
