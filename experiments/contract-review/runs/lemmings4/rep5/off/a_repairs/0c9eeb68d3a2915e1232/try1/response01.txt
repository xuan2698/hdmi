```python
def reference(vectors):
    # Moore FSM for Lemmings.
    # States: WALK_LEFT, WALK_RIGHT, FALL_LEFT, FALL_RIGHT, DIG_LEFT, DIG_RIGHT, SPLAT
    # Outputs are combinational from state.
    # areset is async active-high, resets to WALK_LEFT.
    # Fall counter counts cycles in FALL_*; splat if >20 on ground contact.

    WALK_LEFT, WALK_RIGHT = 0, 1
    FALL_LEFT, FALL_RIGHT = 2, 3
    DIG_LEFT, DIG_RIGHT = 4, 5
    SPLAT = 6

    def outputs(state):
        # R2_walk_outputs
        if state == WALK_LEFT:
            return {"walk_left": 1, "walk_right": 0, "aaah": 0, "digging": 0}
        if state == WALK_RIGHT:
            return {"walk_left": 0, "walk_right": 1, "aaah": 0, "digging": 0}
        # R3_fall_outputs
        if state in (FALL_LEFT, FALL_RIGHT):
            return {"walk_left": 0, "walk_right": 0, "aaah": 1, "digging": 0}
        # R4_dig_outputs
        if state in (DIG_LEFT, DIG_RIGHT):
            return {"walk_left": 0, "walk_right": 0, "aaah": 0, "digging": 1}
        # R5_splat_outputs
        return {"walk_left": 0, "walk_right": 0, "aaah": 0, "digging": 0}

    state = WALK_LEFT
    fall_count = 0
    result = []

    for row in vectors:
        areset = row.get("areset", 0)
        bump_left = row.get("bump_left", 0)
        bump_right = row.get("bump_right", 0)
        ground = row.get("ground", 0)
        dig = row.get("dig", 0)

        # R1_reset: async active-high reset to WALK_LEFT, visible same cycle
        if areset:
            state = WALK_LEFT
            fall_count = 0
            result.append(outputs(state))
            continue

        # Outputs are based on current state (Moore, combinational)
        result.append(outputs(state))

        # Compute next state on posedge
        ns = state
        nfc = fall_count

        if state == WALK_LEFT or state == WALK_RIGHT:
            # R6_walk_transitions: fall > dig > bump
            if ground == 0:
                ns = FALL_LEFT if state == WALK_LEFT else FALL_RIGHT
                nfc = 0
            elif dig == 1:
                ns = DIG_LEFT if state == WALK_LEFT else DIG_RIGHT
                nfc = 0
            else:
                if bump_left and bump_right:
                    # switch direction
                    ns = WALK_RIGHT if state == WALK_LEFT else WALK_LEFT
                elif bump_left:
                    ns = WALK_RIGHT
                elif bump_right:
                    ns = WALK_LEFT
                nfc = 0
        elif state == FALL_LEFT or state == FALL_RIGHT:
            # R7_fall_transitions / R9_splat_transition
            if ground == 1:
                # ground reappears; splat if fell > 20 cycles
                if fall_count > 20:
                    ns = SPLAT
                else:
                    ns = WALK_LEFT if state == FALL_LEFT else WALK_RIGHT
                nfc = 0
            else:
                ns = state
                nfc = fall_count + 1
        elif state == DIG_LEFT or state == DIG_RIGHT:
            # R8_dig_transitions
            if ground == 0:
                ns = FALL_LEFT if state == DIG_LEFT else FALL_RIGHT
                nfc = 0
            else:
                ns = state
                nfc = 0
        elif state == SPLAT:
            # R9_splat_transition: stay forever
            ns = SPLAT
            nfc = 0

        state = ns
        fall_count = nfc

    return result
```
