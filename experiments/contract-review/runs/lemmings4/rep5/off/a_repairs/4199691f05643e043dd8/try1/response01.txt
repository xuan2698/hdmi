def reference(vectors):
    # State encoding
    WALK_LEFT, WALK_RIGHT, FALL_LEFT, FALL_RIGHT, DIG_LEFT, DIG_RIGHT, SPLAT = range(7)
    
    def outputs(state):
        # R2_walk_outputs, R3_fall_outputs, R4_dig_outputs, R5_splat_outputs
        if state == WALK_LEFT:
            return {"walk_left":1,"walk_right":0,"aaah":0,"digging":0}
        if state == WALK_RIGHT:
            return {"walk_left":0,"walk_right":1,"aaah":0,"digging":0}
        if state in (FALL_LEFT, FALL_RIGHT):
            return {"walk_left":0,"walk_right":0,"aaah":1,"digging":0}
        if state in (DIG_LEFT, DIG_RIGHT):
            return {"walk_left":0,"walk_right":0,"aaah":0,"digging":1}
        # SPLAT
        return {"walk_left":0,"walk_right":0,"aaah":0,"digging":0}
    
    def next_state(state, inp, fall_count):
        bl = inp["bump_left"]; br = inp["bump_right"]
        g = inp["ground"]; d = inp["dig"]
        bump = bl or br
        if state == WALK_LEFT:
            if not g: return FALL_LEFT, 0
            if d: return DIG_LEFT, 0
            if bump: return WALK_RIGHT, 0
            return WALK_LEFT, 0
        if state == WALK_RIGHT:
            if not g: return FALL_RIGHT, 0
            if d: return DIG_RIGHT, 0
            if bump: return WALK_LEFT, 0
            return WALK_RIGHT, 0
        if state == FALL_LEFT:
            if g:
                # R9: splat if fell > 20 cycles
                if fall_count + 1 > 20:
                    return SPLAT, 0
                return WALK_LEFT, 0
            return FALL_LEFT, fall_count + 1
        if state == FALL_RIGHT:
            if g:
                if fall_count + 1 > 20:
                    return SPLAT, 0
                return WALK_RIGHT, 0
            return FALL_RIGHT, fall_count + 1
        if state == DIG_LEFT:
            if not g: return FALL_LEFT, 0
            return DIG_LEFT, 0
        if state == DIG_RIGHT:
            if not g: return FALL_RIGHT, 0
            return DIG_RIGHT, 0
        # SPLAT
        return SPLAT, 0
    
    state = WALK_LEFT
    fall_count = 0
    result = []
    for inp in vectors:
        # R1_reset: async reset
        if inp["areset"]:
            state = WALK_LEFT
            fall_count = 0
            result.append(outputs(state))
            continue
        ns, nf = next_state(state, inp, fall_count)
        state = ns
        fall_count = nf
        result.append(outputs(state))
    return result
