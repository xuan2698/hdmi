def reference(vectors):
    # State encoding
    WALK_LEFT, WALK_RIGHT, FALL_LEFT, FALL_RIGHT, DIG_LEFT, DIG_RIGHT, SPLAT = range(7)

    # Outputs per state (Moore)
    def outputs(state):
        if state == WALK_LEFT:
            return {"walk_left": 1, "walk_right": 0, "aaah": 0, "digging": 0}
        if state == WALK_RIGHT:
            return {"walk_left": 0, "walk_right": 1, "aaah": 0, "digging": 0}
        if state == FALL_LEFT or state == FALL_RIGHT:
            return {"walk_left": 0, "walk_right": 0, "aaah": 1, "digging": 0}
        if state == DIG_LEFT or state == DIG_RIGHT:
            return {"walk_left": 0, "walk_right": 0, "aaah": 0, "digging": 1}
        # SPLAT
        return {"walk_left": 0, "walk_right": 0, "aaah": 0, "digging": 0}

    state = WALK_LEFT
    fall_count = 0
    results = []

    for row in vectors:
        areset = row.get("areset", 0)
        bump_left = row.get("bump_left", 0)
        bump_right = row.get("bump_right", 0)
        ground = row.get("ground", 1)
        dig = row.get("dig", 0)

        # R1_reset: async reset to WALK_LEFT, visible same cycle
        if areset:
            state = WALK_LEFT
            fall_count = 0
            results.append(outputs(state))
            continue

        # Combinational outputs for current state (R2-R5)
        results.append(outputs(state))

        # Compute next state on posedge (R6-R9)
        if state == WALK_LEFT:
            if ground == 0:
                # R6: fall has highest precedence
                state = FALL_LEFT
                fall_count = 0
            elif dig == 1:
                # R6: dig next
                state = DIG_LEFT
            elif bump_left == 1 or bump_right == 1:
                # R6: switch direction
                state = WALK_RIGHT
        elif state == WALK_RIGHT:
            if ground == 0:
                state = FALL_RIGHT
                fall_count = 0
            elif dig == 1:
                state = DIG_RIGHT
            elif bump_left == 1 or bump_right == 1:
                state = WALK_LEFT
        elif state == FALL_LEFT:
            if ground == 1:
                # R7/R9: resume or splat
                if fall_count > 20:
                    state = SPLAT
                else:
                    state = WALK_LEFT
                fall_count = 0
            else:
                fall_count += 1
        elif state == FALL_RIGHT:
            if ground == 1:
                if fall_count > 20:
                    state = SPLAT
                else:
                    state = WALK_RIGHT
                fall_count = 0
            else:
                fall_count += 1
        elif state == DIG_LEFT:
            if ground == 0:
                # R8: fall
                state = FALL_LEFT
                fall_count = 0
        elif state == DIG_RIGHT:
            if ground == 0:
                state = FALL_RIGHT
                fall_count = 0
        elif state == SPLAT:
            # R9: stay splat forever
            state = SPLAT

    return results
