def reference(vectors):
    # State encoding with named states.
    S0 = "S0"
    S1 = "S1"
    S11 = "S11"
    S110 = "S110"
    FOUND = "FOUND"

    state = S0
    start_shifting = 0
    inputs = {"reset": 0, "data": 0}
    outputs = []

    for row in vectors:
        # Drive row values before the sampled posedge.
        for name in inputs:
            if name in row:
                inputs[name] = row[name]

        reset = inputs["reset"]
        data = inputs["data"]

        # R_RESET_CLEAR: synchronous active-high reset forces state S0 and start_shifting=0 on the settled posedge.
        if reset == 1:
            state = S0
            start_shifting = 0
        else:
            # Explicit transitions for the 1101 sequence detector.
            if state == S0:
                if data == 1:
                    state = S1
                else:
                    state = S0
            elif state == S1:
                if data == 1:
                    state = S11
                else:
                    state = S0
            elif state == S11:
                if data == 1:
                    state = S11
                else:
                    state = S110
            elif state == S110:
                # R_SET_OR_HOLD_1101: completing 1101 transitions to FOUND and sets start_shifting=1.
                if data == 1:
                    state = FOUND
                else:
                    state = S0
            elif state == FOUND:
                # R_SET_OR_HOLD_1101: FOUND is sticky until reset, keeping start_shifting=1.
                state = FOUND

            # R_SET_OR_HOLD_1101: start_shifting=1 in FOUND.
            # R_KEEP_ZERO_UNTIL_FOUND: start_shifting=0 when not in FOUND.
            if state == FOUND:
                start_shifting = 1
            else:
                start_shifting = 0

        # Output is visible after the posedge has settled.
        outputs.append({"start_shifting": start_shifting})

        # Optional mid input changes occur AFTER sampling and are carried to the start of next row.
        mid = row.get("mid")
        if isinstance(mid, dict):
            for name, value in mid.items():
                if name in inputs:
                    inputs[name] = value

    return outputs
