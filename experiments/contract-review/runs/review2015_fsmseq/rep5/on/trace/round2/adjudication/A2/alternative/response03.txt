def reference(vectors):
    # R_RESET_CLEAR: candidate interpretation applies reset==1 by clearing start_shifting to 0
    # while leaving the search state unchanged.
    # R_SET_OR_HOLD_1101: entering FOUND produces start_shifting=1 on the same settled posedge.
    # R_KEEP_ZERO_UNTIL_FOUND: start_shifting remains 0 unless the settled state is FOUND.

    S0 = 0
    S1 = 1
    S11 = 2
    S110 = 3
    FOUND = 4

    state = S0
    cur_reset = 0
    cur_data = 0
    result = []

    for row in vectors:
        # Inputs are driven before the sampled posedge.
        if "reset" in row:
            cur_reset = row["reset"]
        if "data" in row:
            cur_data = row["data"]

        reset = cur_reset
        data = cur_data

        if reset == 1:
            # R_RESET_CLEAR (candidate): clear output to 0, preserve search state.
            next_state = state
            start_shifting = 0
        else:
            # Explicit state transitions for sequence 1101.
            if state == S0:
                next_state = S1 if data == 1 else S0
            elif state == S1:
                next_state = S11 if data == 1 else S0
            elif state == S11:
                next_state = S11 if data == 1 else S110
            elif state == S110:
                next_state = FOUND if data == 1 else S0
            else:  # FOUND
                next_state = FOUND

            # R_SET_OR_HOLD_1101 / R_KEEP_ZERO_UNTIL_FOUND:
            # same-cycle sticky output based on the settled next state.
            start_shifting = 1 if next_state == FOUND else 0

        state = next_state
        result.append({"start_shifting": start_shifting})

        # Mid input changes occur after sampling, before the other edge,
        # and are carried to the start of the next row.
        if "mid" in row:
            mid = row["mid"]
            if mid is not None:
                if "reset" in mid:
                    cur_reset = mid["reset"]
                if "data" in mid:
                    cur_data = mid["data"]

    return result
