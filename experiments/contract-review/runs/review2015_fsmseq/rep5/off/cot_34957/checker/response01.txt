def reference(vectors):
    # FSM states: 0 = no prefix, 1 = "1", 2 = "11", 3 = "110", 4 = found
    state = 0
    out = []
    for row in vectors:
        reset = row.get('reset', 0) & 1
        data = row.get('data', 0) & 1
        # Synchronous active-high reset wins on the sampled edge
        if reset:
            state = 0
        else:
            if state == 0:
                state = 1 if data == 1 else 0
            elif state == 1:
                state = 2 if data == 1 else 1
            elif state == 2:
                state = 3 if data == 0 else 2
            elif state == 3:
                state = 4 if data == 1 else 1
            elif state == 4:
                state = 4
        start_shifting = 1 if state == 4 else 0
        out.append({'start_shifting': start_shifting})
    return out
