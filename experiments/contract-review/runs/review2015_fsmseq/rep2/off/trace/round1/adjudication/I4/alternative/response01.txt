def reference(vectors):
    # State encoding
    S0, S1, S2, S3, S4 = 0, 1, 2, 3, 4

    # Initialize state and registered output
    state = S0
    start_shifting = 0

    results = []

    for vec in vectors:
        reset = vec['reset']
        data = vec['data']

        # --- Compute next state and next output (sampled at posedge) ---
        # R1: reset == 1 -> start_shifting = 0, state = S0
        if reset == 1:
            next_state = S0
            next_start_shifting = 0
        else:
            # R2: state == S4 -> stay S4, start_shifting = 1
            if state == S4:
                next_state = S4
                next_start_shifting = 1
            # R3: state == S3 and data == 1 -> enter S4
            elif state == S3 and data == 1:
                next_state = S4
                # Registered output: becomes 1 on posedge AFTER entering S4
                next_start_shifting = 0
            else:
                # R4: otherwise search for sequence
                next_start_shifting = 0
                if state == S0:
                    next_state = S1 if data == 1 else S0
                elif state == S1:
                    next_state = S2 if data == 1 else S0
                elif state == S2:
                    next_state = S3 if data == 0 else S2
                elif state == S3:
                    # data == 0 here (data == 1 handled above)
                    next_state = S1 if data == 1 else S0
                else:
                    next_state = S0

        # Output visible same cycle (registered value from previous posedge)
        results.append({'start_shifting': start_shifting})

        # Update registers at posedge
        state = next_state
        start_shifting = next_start_shifting

    return results
