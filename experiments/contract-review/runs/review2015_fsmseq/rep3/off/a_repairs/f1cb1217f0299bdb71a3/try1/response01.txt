def reference(vectors):
    # State encoding
    S0, S1, S2, S3, S4 = 0, 1, 2, 3, 4

    # Power-up state (unspecified, choose S0)
    state = S0
    start_shifting = 0

    out = []
    for row in vectors:
        reset = row.get('reset', 0)
        data = row.get('data', 0)

        # R1: reset == 1 -> start_shifting = 0 (same_cycle, priority 1)
        # R2: reset == 0 and state == S4 -> start_shifting = 1 (priority 2)
        # R3: reset == 0 and state == S3 and data == 1 -> start_shifting = 1 (priority 3)
        # R4: reset == 0 and state != S4 and not (state == S3 and data == 1) -> 0 (priority 4)
        # R5: reset == 0 -> covered by R2/R3/R4
        if reset == 1:
            # R1
            start_shifting = 0
            state = S0
        else:
            # R2
            if state == S4:
                start_shifting = 1
            # R3
            elif state == S3 and data == 1:
                start_shifting = 1
            # R4
            else:
                start_shifting = 0

            # State transition (posedge, reset==0)
            if state == S0:
                state = S1 if data == 1 else S0
            elif state == S1:
                state = S2 if data == 1 else S1 if data == 1 else (S1 if data == 1 else S0)
                # explicit: S1 on '1' -> S2; on '0' -> S0
                if data == 1:
                    state = S2
                else:
                    state = S0
            elif state == S2:
                if data == 1:
                    state = S2
                else:
                    state = S3
            elif state == S3:
                if data == 1:
                    state = S4
                else:
                    state = S0
            elif state == S4:
                state = S4

        out.append({'start_shifting': start_shifting & 1})

    return out
