def reference(vectors):
    # FSM states
    S0, S1, S2, S3, S4 = 0, 1, 2, 3, 4

    # State registers (power-up unspecified -> start at S0)
    state = S0
    start_shifting = 0

    results = []

    for vec in vectors:
        reset = vec.get('reset', 0) & 1
        data = vec.get('data', 0) & 1

        # Compute next state based on current state and data
        if state == S0:
            nxt = S1 if data == 1 else S0
        elif state == S1:
            nxt = S2 if data == 1 else S0
        elif state == S2:
            nxt = S3 if data == 0 else S2
        elif state == S3:
            nxt = S4 if data == 1 else S0
        else:  # S4
            nxt = S4

        # R1: reset == 1 -> start_shifting = 0 (priority 1)
        # R2: reset == 0 and state == S4 -> start_shifting = 1 (priority 2)
        # R3: reset == 0 and state != S4 and next_state == S4 -> start_shifting = 1 (priority 3)
        # R4: reset == 0 and state != S4 and next_state != S4 -> start_shifting = 0 (priority 4)
        # Candidate interpretation: start_shifting is registered, goes high one posedge
        # AFTER the 1101-completing bit is sampled. So output reflects current state
        # (before update), not next_state.
        if reset == 1:
            # R1
            out = 0
            state = S0
        else:
            if state == S4:
                # R2
                out = 1
            else:
                # R3/R4: registered output -> based on current state, not next
                out = 0
            state = nxt

        start_shifting = out
        results.append({'start_shifting': start_shifting & 1})

    return results
