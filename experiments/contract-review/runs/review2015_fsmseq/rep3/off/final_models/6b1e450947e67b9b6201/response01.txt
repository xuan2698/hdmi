def reference(vectors):
    # FSM states
    S0, S1, S2, S3, S4 = 0, 1, 2, 3, 4

    # State registers (power-up unspecified; use S0 as concrete default)
    state = S0
    start_shifting = 0

    results = []

    for vec in vectors:
        reset = vec.get('reset', 0) & 1
        data = vec.get('data', 0) & 1

        # --- Compute next state and next output based on current state and inputs ---
        # R1: reset == 1 -> start_shifting = 0 (priority 1)
        if reset == 1:
            next_state = S0
            next_start_shifting = 0
        else:
            # R2: state == S4 -> start_shifting stays 1 (priority 2)
            if state == S4:
                next_state = S4
                next_start_shifting = 1
            else:
                # Transition logic for sequence 1101
                if state == S0:
                    next_state = S1 if data == 1 else S0
                elif state == S1:
                    next_state = S2 if data == 1 else S0
                elif state == S2:
                    next_state = S3 if data == 0 else S2
                elif state == S3:
                    next_state = S4 if data == 1 else S0
                else:
                    next_state = S0

                # R3: state == S3 and data == 1 -> start_shifting = 1 (priority 3)
                # R4: otherwise (not S4, next_state != S4) -> start_shifting = 0 (priority 4)
                if state == S3 and data == 1:
                    next_start_shifting = 1
                else:
                    next_start_shifting = 0

        # Output visible same_cycle (combinational from current state/inputs per rules)
        results.append({'start_shifting': next_start_shifting})

        # Update state registers at posedge
        state = next_state
        start_shifting = next_start_shifting

    return results
