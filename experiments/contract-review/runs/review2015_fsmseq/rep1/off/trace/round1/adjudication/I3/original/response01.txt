def reference(vectors):
    # Contract rules:
    # R1: reset == 1 -> start_shifting = 0 (sync reset, same cycle)
    # R2: reset == 0 and state == S4 -> start_shifting = 1
    # R3: reset == 0 and state == S3 and data == 1 -> start_shifting = 1
    # R4: reset == 0 and state != S4 and not (state == S3 and data == 1) -> start_shifting = 0
    # States: S0 (no prefix), S1 ('1'), S2 ('11'), S3 ('110'), S4 ('1101' found, latched)
    # Mid input changes occur AFTER sampling, before the other edge, carried to next row.

    S0, S1, S2, S3, S4 = 0, 1, 2, 3, 4

    state = S0
    start_shifting = 0
    out = []

    for vec in vectors:
        reset = vec.get('reset', 0) & 1
        data = vec.get('data', 0) & 1

        # Compute next state and output based on current state and inputs (posedge sampling)
        if reset == 1:
            # R1: synchronous reset
            next_state = S0
            next_out = 0
        else:
            if state == S4:
                # R2: latched forever
                next_state = S4
                next_out = 1
            elif state == S3 and data == 1:
                # R3: completes 1101
                next_state = S4
                next_out = 1
            else:
                # R4: not yet found
                next_out = 0
                # State transitions for prefix matching
                if state == S0:
                    next_state = S1 if data == 1 else S0
                elif state == S1:
                    next_state = S2 if data == 1 else S0
                elif state == S2:
                    next_state = S3 if data == 0 else S2
                elif state == S3:
                    # data == 0 here (data == 1 handled above)
                    next_state = S1 if data == 1 else S0
                else:
                    next_state = S0

        state = next_state
        start_shifting = next_out
        out.append({'start_shifting': start_shifting & 1})

    return out
