def reference(vectors):
    # R_RESET: active-high synchronous reset forces S0 and start_shifting=0.
    # R_COMPLETE_1101: completion of 1101 is visible same cycle at the sampled posedge.
    # R_HOLD_FOUND: FOUND is held until reset, with start_shifting=1.
    # R_ZERO_NOT_FOUND: using candidate override "start_shifting is 1 before any 1101 detection".
    # Mid input changes are applied after sampling, before the other edge, and carried to the next row.
    state = "S0"
    start_shifting = 0
    cur = {"reset": 0, "data": 0}
    result = []

    for row in vectors:
        # Drive row inputs before the sampled posedge.
        if isinstance(row.get("inputs"), dict):
            src = row["inputs"]
            for k in ("reset", "data"):
                if k in src:
                    cur[k] = src[k] & 1
        for k in ("reset", "data"):
            if k in row:
                cur[k] = row[k] & 1

        reset = cur.get("reset", 0) & 1
        data = cur.get("data", 0) & 1
        prev_state = state

        if reset == 1:
            # R_RESET: reset == 1
            state = "S0"
            start_shifting = 0
        else:
            # Explicit normal transitions.
            if prev_state == "S0":
                state = "S1" if data else "S0"
            elif prev_state == "S1":
                state = "S11" if data else "S0"
            elif prev_state == "S11":
                state = "S11" if data else "S110"
            elif prev_state == "S110":
                state = "FOUND" if data else "S0"
            elif prev_state == "FOUND":
                state = "FOUND"
            else:
                state = "S0"

            if prev_state == "S110" and data == 1:
                # R_COMPLETE_1101: same-cycle completion visibility.
                start_shifting = 1
            elif prev_state == "FOUND":
                # R_HOLD_FOUND: forever until reset.
                start_shifting = 1
            else:
                # R_ZERO_NOT_FOUND: candidate override, start_shifting is 1 before detection.
                start_shifting = 1

        result.append({"start_shifting": start_shifting & 1})

        # Optional mid input changes occur AFTER sampling, before the other edge.
        mid = row.get("mid")
        if isinstance(mid, dict):
            if isinstance(mid.get("inputs"), dict):
                mid = mid["inputs"]
            for k in ("reset", "data"):
                if k in mid:
                    cur[k] = mid[k] & 1

        for k in ("reset", "data"):
            if k + "_mid" in row:
                cur[k] = row[k + "_mid"] & 1
            if "mid_" + k in row:
                cur[k] = row["mid_" + k] & 1

    return result
