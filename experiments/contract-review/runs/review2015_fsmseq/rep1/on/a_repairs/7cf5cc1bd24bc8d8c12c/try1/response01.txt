def reference(vectors):
    # State encoding
    S0, S1, S11, S110, FOUND = 0, 1, 2, 3, 4

    state = S0
    current_inputs = {"reset": 0, "data": 0}
    outputs = []

    for row in vectors:
        # Drive the row's primary input values before the sampled posedge.
        if isinstance(row, dict):
            for signal in ("reset", "data"):
                if signal in row:
                    current_inputs[signal] = int(row[signal]) & 1

        reset = current_inputs.get("reset", 0) & 1
        data = current_inputs.get("data", 0) & 1

        # R_RESET: synchronous active-high reset has priority and forces S0,
        # which makes start_shifting 0 for this sampled posedge.
        if reset == 1:
            next_state = S0
        else:
            if state == S0:
                next_state = S1 if data == 1 else S0
            elif state == S1:
                next_state = S11 if data == 1 else S0
            elif state == S11:
                next_state = S11 if data == 1 else S110
            elif state == S110:
                # R_COMPLETE_1101: completing 1101 enters FOUND immediately,
                # so start_shifting is visible as 1 on this posedge.
                next_state = FOUND if data == 1 else S0
            else:
                # R_HOLD_FOUND: once FOUND, any data keeps the machine in FOUND
                # and start_shifting stays 1 until reset.
                next_state = FOUND

        state = next_state

        # R_COMPLETE_1101 / R_HOLD_FOUND / R_ZERO_NOT_FOUND:
        # start_shifting is 1 exactly when the updated state is FOUND.
        start_shifting = 1 if state == FOUND else 0
        outputs.append({"start_shifting": start_shifting})

        # Process optional mid-cycle input changes after sampling.
        # They are carried to the start of the next row.
        if isinstance(row, dict):
            mid = row.get("mid")
            if isinstance(mid, dict):
                for signal, value in mid.items():
                    if signal in ("reset", "data"):
                        current_inputs[signal] = int(value) & 1
            elif isinstance(mid, list):
                for change in mid:
                    if isinstance(change, dict):
                        for signal, value in change.items():
                            if signal in ("reset", "data"):
                                current_inputs[signal] = int(value) & 1

            for key, value in row.items():
                if key.endswith("_mid"):
                    signal = key[:-4]
                    if signal in ("reset", "data"):
                        current_inputs[signal] = int(value) & 1

    return outputs
