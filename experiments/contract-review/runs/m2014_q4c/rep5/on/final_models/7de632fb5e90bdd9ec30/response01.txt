def reference(vectors):
    # State: registered output q and current input values carried from prior mid changes.
    q = 0
    cur = {"d": 0, "r": 0}
    outputs = []

    for row in vectors:
        # Each row's values are driven before the sampled posedge edge.
        for name in ("d", "r"):
            if name in row:
                cur[name] = row[name]

        d = cur.get("d", 0) & 1
        r = cur.get("r", 0) & 1

        # Rule sync_active_high_reset: active high synchronous reset at posedge, priority 2.
        if r == 1:
            q = 0
        else:
            # Rule d_update_when_not_reset: when not reset, q becomes d at posedge.
            q = d & 1

        outputs.append({"q": q & 1})

        # Optional mid changes occur after sampling, before the other edge,
        # and are carried to the start of the next row.
        mid = row.get("mid")
        if isinstance(mid, dict):
            for name in ("d", "r"):
                if name in mid:
                    cur[name] = mid[name]

    return outputs
