def reference(vectors):
    # Power-up state is unspecified by the spec; choose concrete q_zero.
    q_state = 'q_zero'  # named state: q_zero means registered q is 0
    d = 0
    r = 0
    result = []

    for row in vectors:
        # Apply row's explicit inputs driven before the sampled edge.
        if 'd' in row:
            d = row['d'] & 1
        if 'r' in row:
            r = row['r'] & 1

        # Rule q_posedge_update: At each sampled posedge: if r==1 then q becomes 0;
        # otherwise q becomes d. Reset has priority over d.
        if r == 1:
            q_state = 'q_zero'
        else:
            if d == 1:
                q_state = 'q_one'
            else:
                q_state = 'q_zero'

        q_val = 1 if q_state == 'q_one' else 0
        result.append({'q': q_val})

        # Optional mid input changes occur AFTER sampling and are carried to
        # the start of the next row. They do not affect the current sampled q.
        if 'mid' in row:
            mid = row['mid']
            if isinstance(mid, dict):
                if 'd' in mid:
                    d = mid['d'] & 1
                if 'r' in mid:
                    r = mid['r'] & 1
            elif isinstance(mid, list):
                for change in mid:
                    if isinstance(change, dict):
                        if 'd' in change:
                            d = change['d'] & 1
                        if 'r' in change:
                            r = change['r'] & 1

    return result
