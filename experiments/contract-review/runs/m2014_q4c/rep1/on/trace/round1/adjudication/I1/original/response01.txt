def reference(vectors):
    # Carried input state between sampled edges (mid changes update these)
    d = 0
    r = 0
    q = 0
    out = []

    for row in vectors:
        # Drive row values before the sampled posedge.
        # These override any carried mid changes from the previous row.
        if 'd' in row:
            d = int(row['d']) & 1
        if 'r' in row:
            r = int(row['r']) & 1

        # Rule q_posedge_update: At sampled posedge, if r==1 then q becomes 0;
        # otherwise q becomes d. Between posedges q holds.
        if r == 1:
            q = 0
        else:
            q = d & 1

        out.append({'q': q})

        # Optional mid input changes occur AFTER sampling and are carried to
        # the start of the next row.
        mid = row.get('mid')
        if isinstance(mid, dict):
            if 'd' in mid:
                d = int(mid['d']) & 1
            if 'r' in mid:
                r = int(mid['r']) & 1

    return out
