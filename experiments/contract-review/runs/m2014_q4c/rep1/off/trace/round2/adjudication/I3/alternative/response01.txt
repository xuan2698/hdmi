def reference(vectors):
    # State: Q = stored flip-flop value
    Q = 0  # power-up unspecified; choose 0
    out = []
    for row in vectors:
        d = row['d'] & 1
        r = row['r'] & 1
        # R1_sync_reset: active high synchronous reset (reset output to 0)
        # R2_capture_d: capture d on posedge when r == 0
        # Candidate interpretation: reset is asynchronous - q goes 0 immediately when r=1
        if r == 1:
            Q = 0  # R1_sync_reset (async interpretation)
        else:
            Q = d  # R2_capture_d
        out.append({'q': Q & 1})
    return out
