def reference(vectors):
    # State: registered BCD counter value
    q = 0
    reset_val = 0
    outputs = []
    
    for row in vectors:
        # Apply explicit inputs driven before the sampled posedge
        if "reset" in row:
            reset_val = row["reset"] & 1
        
        # posedge clk: sample reset and update state
        if reset_val:
            # rule reset: synchronous active-high reset, q becomes 0
            q = 0
        else:
            # rule bcd_increment: BCD add 1 modulo 10000
            ones = q & 0xF
            tens = (q >> 4) & 0xF
            hundreds = (q >> 8) & 0xF
            thousands = (q >> 12) & 0xF
            
            ones += 1
            if ones == 10:
                ones = 0
                tens += 1
                if tens == 10:
                    tens = 0
                    hundreds += 1
                    if hundreds == 10:
                        hundreds = 0
                        thousands += 1
                        if thousands == 10:
                            thousands = 0
            q = ones | (tens << 4) | (hundreds << 8) | (thousands << 12)
        
        # rule ena_combinational: combinational enable outputs from q
        ones = q & 0xF
        tens = (q >> 4) & 0xF
        hundreds = (q >> 8) & 0xF
        ena = 0
        if ones == 9:
            ena |= 1  # ena[0]
        if tens == 9 and ones == 9:
            ena |= 2  # ena[1]
        if hundreds == 9 and tens == 9 and ones == 9:
            ena |= 4  # ena[2]
        
        outputs.append({"q": q, "ena": ena})
        
        # Process optional mid changes after sampling; carried to next row
        if "mid" in row and isinstance(row["mid"], dict):
            for k, v in row["mid"].items():
                if k == "reset":
                    reset_val = v & 1
        if "reset_mid" in row:
            reset_val = row["reset_mid"] & 1
        if "mid_reset" in row:
            reset_val = row["mid_reset"] & 1
    
    return outputs
