def reference(vectors):
    # Rule reset, bcd_increment: state register initialized to 0 for a concrete power-up value.
    q = 0
    inputs = {}
    results = []

    for row in vectors:
        # Drive row's input values before the sampled posedge.
        if isinstance(row, dict):
            for key, value in row.items():
                if key != "mid":
                    inputs[key] = value
            mid = row.get("mid", None)
        else:
            inputs["reset"] = row
            mid = None

        reset_value = inputs.get("reset", 0) & 1

        # Rule reset: synchronous active-high reset has priority at posedge.
        # Rule bcd_increment: otherwise increment BCD counter modulo 10000.
        if reset_value:
            q = 0
        else:
            ones = q & 0xF
            tens = (q >> 4) & 0xF
            hundreds = (q >> 8) & 0xF
            thousands = (q >> 12) & 0xF

            ones += 1
            if ones == 10:
                ones = 0
                tens += 1
                if tens == 10:
                    tens = 0
                    hundreds += 1
                    if hundreds == 10:
                        hundreds = 0
                        thousands += 1
                        if thousands == 10:
                            thousands = 0

            q = (thousands << 12) | (hundreds << 8) | (tens << 4) | ones

        # Rule ena_combinational: combinational enables from settled q.
        ones = q & 0xF
        tens = (q >> 4) & 0xF
        hundreds = (q >> 8) & 0xF

        ena0 = 1 if ones == 9 else 0
        ena1 = 1 if (tens == 9 and ones == 9) else 0
        ena2 = 1 if (hundreds == 9 and tens == 9 and ones == 9) else 0
        ena = ena0 | (ena1 << 1) | (ena2 << 2)

        results.append({"q": q, "ena": ena})

        # Process mid changes after sampling; they carry to the next row.
        if isinstance(mid, dict):
            for key, value in mid.items():
                inputs[key] = value
        elif isinstance(mid, list):
            for change in mid:
                if isinstance(change, dict):
                    for key, value in change.items():
                        inputs[key] = value

    return results
