    for v in vectors:

        if reset:
            q = 0
            ena = 0
        else:

            tens = (q >> 4) & 0xF
            hundreds = (q >> 8) & 0xF
            thousands = (q >> 12) & 0xF
            if ones != 9:
                ones += 1
            else:
                ones = 0
                if tens != 9:
                    tens += 1
                else:
                    tens = 0
                    if hundreds != 9:
                        hundreds += 1
                    else:
                        hundreds = 0
                        if thousands != 9:
                            thousands += 1
                        else:
                            thousands = 0
            q = (thousands << 12) | (hundreds << 8) | (tens << 4) | ones
