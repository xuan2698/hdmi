def reference(vectors):
    def bcd_inc(q):
        ones = q & 0xF
        tens = (q >> 4) & 0xF
        hundreds = (q >> 8) & 0xF
        thousands = (q >> 12) & 0xF
        if ones != 9:
            ones += 1
        else:
            ones = 0
            if tens != 9:
                tens += 1
            else:
                tens = 0
                if hundreds != 9:
                    hundreds += 1
                else:
                    hundreds = 0
                    if thousands != 9:
                        thousands += 1
                    else:
                        thousands = 0
        return (thousands << 12) | (hundreds << 8) | (tens << 4) | ones

    def compute_ena(q):
        ones = q & 0xF
        tens = (q >> 4) & 0xF
        hundreds = (q >> 8) & 0xF
        ena0 = 1 if ones == 9 else 0
        ena1 = 1 if (ones == 9 and tens == 9) else 0
        ena2 = 1 if (ones == 9 and tens == 9 and hundreds == 9) else 0
        return (ena2 << 2) | (ena1 << 1) | ena0

    q = 0
    results = []
    for v in vectors:
        reset = v['reset']
        if reset:
            q_next = 0
        else:
            q_next = bcd_inc(q)
        ena = compute_ena(q_next)
        results.append({'q': q_next, 'ena': ena})
        q = q_next
