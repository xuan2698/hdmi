def reference(vectors):
    q = 0
    result = []
    for row in vectors:
        # sync_reset: synchronous active-high reset at posedge, priority over increment
        reset = row.get('reset', 0) if isinstance(row, dict) else 0
        if reset:
            q = 0
        else:
            # bcd_increment: 4-digit BCD increment modulo 10000
            ones = q & 0xF
            tens = (q >> 4) & 0xF
            hundreds = (q >> 8) & 0xF
            thousands = (q >> 12) & 0xF

            ones += 1
            if ones >= 10:
                ones = 0
                tens += 1
                if tens >= 10:
                    tens = 0
                    hundreds += 1
                    if hundreds >= 10:
                        hundreds = 0
                        thousands += 1
                        if thousands >= 10:
                            thousands = 0

            q = (thousands << 12) | (hundreds << 8) | (tens << 4) | ones

        # upper_digit_enables: combinational from current q after posedge
        ones = q & 0xF
        tens = (q >> 4) & 0xF
        hundreds = (q >> 8) & 0xF

        bit0 = 1 if ones == 9 else 0
        bit1 = 1 if (ones == 9 and tens == 9) else 0
        bit2 = 1 if (ones == 9 and tens == 9 and hundreds == 9) else 0
        ena = bit0 | (bit1 << 1) | (bit2 << 2)

        result.append({'q': q, 'ena': ena})

    return result
