def reference(vectors):
    # state: decimal value 0..9999 representing the BCD counter
    val = 0
    out = []
    for row in vectors:
        reset = 0
        if isinstance(row, dict):
            reset = row.get('reset', 0)
        else:
            reset = row

        # sync_reset: synchronous active-high reset overrides increment
        if reset == 1:
            val = 0
        else:
            # bcd_increment: increment as 4-digit BCD modulo 10000
            val = (val + 1) % 10000

        # Convert decimal counter value to 16-bit BCD q
        ones = val % 10
        tens = (val // 10) % 10
        hundreds = (val // 100) % 10
        thousands = (val // 1000) % 10
        q = (thousands << 12) | (hundreds << 8) | (tens << 4) | ones

        # upper_digit_enables: combinational from current BCD digits
        bit0 = 1 if ones == 9 else 0
        bit1 = 1 if (ones == 9 and tens == 9) else 0
        bit2 = 1 if (ones == 9 and tens == 9 and hundreds == 9) else 0
        ena = bit0 + (bit1 << 1) + (bit2 << 2)

        out.append({'q': q, 'ena': ena})

    return out
