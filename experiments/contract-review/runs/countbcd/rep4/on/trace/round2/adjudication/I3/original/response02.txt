def reference(vectors):
    q = 0
    outputs = []
    for row in vectors:
        # sync_active_high_reset: synchronous active-high reset sampled at rising edge
        r = row["reset"]
        if r:
            q = 0
            ena = 0
        else:
            # bcd_counter_increment: increment four BCD digits modulo 10000
            d0 = q & 15
            d1 = (q >> 4) & 15
            d2 = (q >> 8) & 15
            d3 = (q >> 12) & 15
            d0 = d0 + 1
            if d0 == 10:
                d0 = 0
                d1 = d1 + 1
                if d1 == 10:
                    d1 = 0
                    d2 = d2 + 1
                    if d2 == 10:
                        d2 = 0
                        d3 = d3 + 1
                        if d3 == 10:
                            d3 = 0
            q = d0 | (d1 << 4) | (d2 << 8) | (d3 << 12)
            # upper_digit_enables: combinational enables based on current q after edge
            e0 = 1 if d0 == 9 else 0
            e1 = 1 if (d0 == 9 and d1 == 9) else 0
            e2 = 1 if (d0 == 9 and d1 == 9 and d2 == 9) else 0
            ena = e0 | (e1 << 1) | (e2 << 2)
        outputs.append({"q": q, "ena": ena})
    return outputs
