def reference(vectors):
    # State encoding
    A, B, C, D = 0, 1, 2, 3
    state = A  # power-up state (unspecified; use A)
    out = 0
    results = []
    for vec in vectors:
        # R1/R2: combinational output based on current state (visible same cycle)
        if state == D:
            out = 1  # R2
        else:
            out = 0  # R1
        # Sample inputs at posedge
        inp = vec.get('in', 0)
        rst = vec.get('reset', 0)
        # R3: synchronous active high reset forces state A
        if rst == 1:
            state = A
        else:
            # R4-R7: state transitions
            if state == A:
                state = B if inp == 1 else A  # R4
            elif state == B:
                state = B if inp == 1 else C  # R5
            elif state == C:
                state = D if inp == 1 else A  # R6
            elif state == D:
                state = B if inp == 1 else C  # R7
        results.append({'out': out})
    return results
