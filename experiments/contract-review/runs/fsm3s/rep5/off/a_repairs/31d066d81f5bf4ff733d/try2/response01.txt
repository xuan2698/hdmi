def reference(vectors):
    # State encoding
    A, B, C, D = 0, 1, 2, 3
    state = A  # power-up state (unspecified; use A)
    out = 0
    results = []
    for vec in vectors:
        # R1/R2: combinational output based on current state (visible same cycle)
        if state == D:
            out = 1  # R2
        else:
            out = 0  # R1
        results.append({"out": out})
        # R3: synchronous active high reset -> state A at posedge
        if vec.get("reset", 0) == 1:
            state = A  # R3
        else:
            inp = vec.get("in", 0)
            # R4-R7: state transitions at posedge
            if state == A:
                state = B if inp == 1 else A  # R4
            elif state == B:
                state = B if inp == 1 else C  # R5
            elif state == C:
                state = D if inp == 1 else A  # R6
            elif state == D:
                state = B if inp == 1 else C  # R7
    return results
