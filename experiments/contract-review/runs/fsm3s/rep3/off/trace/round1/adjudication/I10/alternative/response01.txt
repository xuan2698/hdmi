def reference(vectors):
    # State encoding
    A, B, C, D = 0, 1, 2, 3
    state = A  # power-up state (unspecified, default to A)
    out = 0
    results = []

    for vec in vectors:
        # Apply mid changes carried from previous row (already in vec)
        # R3 (candidate interpretation): async reset forces state A and out=0 immediately
        if vec.get('reset', 0) == 1:
            state = A
            out = 0

        # R1 / R2: combinational output based on current state (visible same cycle)
        if state == D:
            out = 1  # R2
        else:
            out = 0  # R1

        # Record output for this row (sampled after posedge settles)
        results.append({'out': out})

        # R3 (sync portion) / R4-R7: posedge state transition
        if vec.get('reset', 0) == 1:
            state = A  # R3
        else:
            inp = vec.get('in', 0)
            if state == A:
                state = B if inp == 1 else A  # R4
            elif state == B:
                state = B if inp == 1 else C  # R5
            elif state == C:
                state = D if inp == 1 else A  # R6
            elif state == D:
                state = B if inp == 1 else C  # R7

    return results
