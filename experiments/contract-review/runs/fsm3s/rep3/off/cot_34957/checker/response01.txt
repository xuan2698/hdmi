def reference(vectors):
    # State encoding: A=0, B=1, C=2, D=3
    # Moore outputs: A=0, B=0, C=0, D=1
    # Synchronous active-high reset -> state A
    # Transition table:
    #   A: in=0 -> A, in=1 -> B
    #   B: in=0 -> C, in=1 -> B
    #   C: in=0 -> A, in=1 -> D
    #   D: in=0 -> C, in=1 -> B
    out_map = {0: 0, 1: 0, 2: 0, 3: 1}
    next_map = {
        0: {0: 0, 1: 1},
        1: {0: 2, 1: 1},
        2: {0: 0, 1: 3},
        3: {0: 2, 1: 1},
    }

    state = 0  # power-up unspecified; start at A
    results = []

    for row in vectors:
        # Sample inputs at the rising edge (values driven before the edge)
        r = row.get('reset', 0) & 1
        i = row.get('in', 0) & 1

        # Synchronous reset has priority over in
        if r:
            state = 0
        else:
            state = next_map[state][i]

        # Output settles after the edge based on new current state
        results.append({'out': out_map[state] & 1})

    return results
