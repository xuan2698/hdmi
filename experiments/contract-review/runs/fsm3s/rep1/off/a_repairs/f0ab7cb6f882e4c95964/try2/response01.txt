def reference(vectors):
    # State encoding
    A, B, C, D = 0, 1, 2, 3
    state = A  # power-up state (unspecified, but reset will handle it)
    out = 0
    results = []
    for vec in vectors:
        reset = vec.get('reset', 0)
        inp = vec.get('in', 0)
        # R_RESET: synchronous active high reset -> state A, out 0 (visible next cycle)
        # R_A/R_B/R_C/R_D: combinational output based on current state (same cycle)
        # R_TRANS: next state determined by table on posedge when reset == 0
        # Compute output for current state (combinational, same cycle)
        if state == A:
            out = 0  # R_A
        elif state == B:
            out = 0  # R_B
        elif state == C:
            out = 0  # R_C
        else:  # D
            out = 1  # R_D
        results.append({'out': out})
        # Compute next state (posedge update, visible next cycle)
        if reset == 1:
            state = A  # R_RESET
        else:
            if state == A:
                state = A if inp == 0 else B  # R_TRANS
            elif state == B:
                state = C if inp == 0 else B  # R_TRANS
            elif state == C:
                state = A if inp == 0 else D  # R_TRANS
            else:  # D
                state = C if inp == 0 else B  # R_TRANS
    return results
