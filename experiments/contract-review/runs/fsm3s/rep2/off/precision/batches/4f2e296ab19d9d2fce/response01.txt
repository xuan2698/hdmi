```python
def reference(vectors):
    state = 'A'
    results = []
    for v in vectors:
        inp = v['in']
        rst = v['reset']
        # combinational output based on current state
        out = 1 if state == 'D' else 0
        results.append({'out': out})
        # posedge update
        if rst == 1:
            state = 'A'
        else:
            if state == 'A':
                state = 'B' if inp == 1 else 'A'
            elif state == 'B':
                state = 'B' if inp == 1 else 'C'
            elif state == 'C':
                state = 'D' if inp == 1 else 'A'
            elif state == 'D':
                state = 'B' if inp == 1 else 'C'
    return results
```
