def reference(vectors):
    # State encoding
    A, B, C, D = 0, 1, 2, 3
    # R3: synchronous active high reset resets FSM to state A
    # global: initial state is D before any reset (candidate interpretation)
    state = D
    out = 1  # R2: state D -> out 1
    results = []
    for vec in vectors:
        # Sample inputs at posedge (values driven before sampled edge)
        inp = vec.get('in', 0)
        rst = vec.get('reset', 0)
        # R1/R2: combinational output based on current state (visible same cycle)
        if state == D:
            out = 1  # R2
        else:
            out = 0  # R1
        results.append({'out': out})
        # R3: synchronous reset has priority 0
        if rst == 1:
            state = A  # R3
        else:
            # R4-R7: state transitions on posedge, priority 2
            if state == A:
                state = B if inp == 1 else A  # R4
            elif state == B:
                state = C if inp == 0 else B  # R5
            elif state == C:
                state = D if inp == 1 else A  # R6
            elif state == D:
                state = C if inp == 0 else B  # R7
        # mid changes occur AFTER sampling, carried to next row
        # (handled naturally since we read vec at start of each iteration)
    return results
