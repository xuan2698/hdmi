def reference(vectors):
    # Contract implementation.
    # State encoding: A=0, B=1, C=2, D=3
    # R1-R4: combinational output based on current state (same_cycle).
    # R5: synchronous active high reset -> state A on posedge.
    # R6-R9: next-state transitions on posedge.
    # Global: initial state assumed A with output 0 before first posedge.
    # Mid changes occur AFTER sampling, before other edge, carried to next row.

    out_map = {0: 0, 1: 0, 2: 0, 3: 1}  # R1, R2, R3, R4

    state = 0  # global: initial state A
    results = []

    for row in vectors:
        # Sample inputs at this row (driven before sampled edge)
        in_val = row.get('in', 0) & 1
        reset_val = row.get('reset', 0) & 1

        # Combinational output visible same cycle (R1-R4)
        out_val = out_map[state]

        results.append({'out': out_val & 1})

        # Posedge update: compute next state
        if reset_val == 1:
            # R5: synchronous active high reset forces state A
            state = 0
        else:
            # R6-R9: explicit transitions
            if state == 0:  # A
                state = 1 if in_val == 1 else 0  # R6
            elif state == 1:  # B
                state = 1 if in_val == 1 else 2  # R7
            elif state == 2:  # C
                state = 3 if in_val == 1 else 0  # R8
            elif state == 3:  # D
                state = 1 if in_val == 1 else 2  # R9

    return results
