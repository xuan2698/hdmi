def reference(vectors):
    # State encoding: A=0, B=1, C=2, D=3
    # Output: A->0, B->0, C->0, D->1
    # Synchronous active-high reset forces state A on posedge.
    # Mid changes occur AFTER sampling, before the other edge, carried to next row.

    state = 0  # power-up unspecified; start at A (reset default)
    out = 0
    results = []

    for vec in vectors:
        # Apply mid changes from previous row are already reflected in vec inputs
        # (mid changes are carried to start of next row, so vec holds current values)
        inp = vec.get('in', 0)
        rst = vec.get('reset', 0)

        # R1-R4: combinational output based on current state (visible same cycle)
        if state == 0:      # R1: state A -> out 0
            out = 0
        elif state == 1:    # R2: state B -> out 0
            out = 0
        elif state == 2:    # R3: state C -> out 0
            out = 0
        else:               # R4: state D -> out 1
            out = 1

        results.append({'out': out})

        # Posedge update: compute next state
        if rst == 1:
            # R5: synchronous active high reset -> state A
            state = 0
        else:
            # R6-R9: explicit transitions
            if state == 0:      # R6: A: in=0->A, in=1->B
                state = 0 if inp == 0 else 1
            elif state == 1:    # R7: B: in=0->C, in=1->B
                state = 2 if inp == 0 else 1
            elif state == 2:    # R8: C: in=0->A, in=1->D
                state = 0 if inp == 0 else 3
            else:               # R9: D: in=0->C, in=1->B
                state = 2 if inp == 0 else 1

    return results
