def reference(vectors):
    # State encoding: A=0, B=1, C=2, D=3
    # Output: A->0, B->0, C->0, D->1
    # Synchronous active-high reset to state A.
    # R5: reset==1 at posedge forces state A (next_cycle visible)
    # R6-R9: state transitions at posedge based on current state and in
    # R1-R4: combinational output based on current state (same_cycle)

    state = 0  # power-up unspecified; start at A (reset default)
    out = 0
    results = []

    for vec in vectors:
        # Apply inputs for this row (driven before sampled edge)
        in_val = vec.get('in', 0)
        reset_val = vec.get('reset', 0)

        # R1-R4: combinational output from current state (same_cycle)
        if state == 0:      # R1: state A -> out 0
            out = 0
        elif state == 1:    # R2: state B -> out 0
            out = 0
        elif state == 2:    # R3: state C -> out 0
            out = 0
        else:               # R4: state D -> out 1
            out = 1

        # Sample: record output for this row (visible same_cycle)
        results.append({'out': out & 0x1})

        # Posedge update (next_cycle)
        if reset_val == 1:
            # R5: synchronous active high reset forces state A
            state = 0
        else:
            # R6-R9: explicit transitions
            if state == 0:          # R6: A -> A if in==0, B if in==1
                state = 0 if in_val == 0 else 1
            elif state == 1:        # R7: B -> C if in==0, B if in==1
                state = 2 if in_val == 0 else 1
            elif state == 2:        # R8: C -> A if in==0, D if in==1
                state = 0 if in_val == 0 else 3
            else:                   # R9: D -> C if in==0, B if in==1
                state = 2 if in_val == 0 else 1

        # Mid changes (if any) occur after sampling, before next edge,
        # carried to start of next row. Handled implicitly since next
        # iteration reads vec inputs fresh.

    return results
