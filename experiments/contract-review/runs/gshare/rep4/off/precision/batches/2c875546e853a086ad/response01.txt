def reference(vectors):
    # state
    ghr = 0
    pht = [0]*128
    results = []
    for v in vectors:
        areset = v['areset']
        if areset:
            ghr = 0
            pht = [0]*128
            results.append({'predict_taken':0,'predict_history':0})
            continue
        # combinational outputs from current state
        pv = v['predict_valid']
        if pv:
            idx = (v['predict_pc'] ^ ghr) & 0x7f
            pt = (pht[idx] >> 1) & 1
            ph = ghr
        else:
            pt = 0
            ph = 0
        results.append({'predict_taken':pt,'predict_history':ph})
        # posedge updates
        tv = v['train_valid']
        tm = v['train_mispredicted']
        # PHT update
        if tv:
            tidx = (v['train_pc'] ^ v['train_history']) & 0x7f
            c = pht[tidx]
            if v['train_taken']:
                if c < 3: c += 1
            else:
                if c > 0: c -= 1
            pht[tidx] = c
        # GHR update
        if tv and tm:
            ghr = v['train_history'] & 0x7f
        elif pv:
            ghr = ((ghr << 1) | pt) & 0x7f
    return results
