    ghr = 0
    out = []
    for v in vectors:
        areset = v['areset']
        if areset:
            pht = [0]*128
            ghr = 0
            out.append({'predict_taken':0, 'predict_history':0})
            continue
        # combinational outputs
        if v['predict_valid']:
            idx = (v['predict_pc'] ^ ghr) & 0x7F
            pt = (pht[idx] >> 1) & 1
            ph = ghr
        else:
            pt = 0
            ph = 0
        out.append({'predict_taken':pt, 'predict_history':ph})
        # posedge updates
        new_pht = pht[:]  # copy
