def reference(vectors):
    # State
    ghr = 0
    pht = [0] * 128  # 2-bit saturating counters, reset to 0

    outputs = []

    for v in vectors:
        areset = v['areset']
        predict_valid = v['predict_valid']
        predict_pc = v['predict_pc']
        train_valid = v['train_valid']
        train_taken = v['train_taken']
        train_mispredicted = v['train_mispredicted']
        train_history = v['train_history']
        train_pc = v['train_pc']

        if areset:
            # Async reset: clear state, outputs reflect reset
            ghr = 0
            pht = [0] * 128
            outputs.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # Combinational outputs (before edge update)
        if predict_valid:
            idx = (predict_pc ^ ghr) & 0x7F
            ctr = pht[idx]
            predict_taken = 1 if ctr >= 2 else 0
            predict_history = ghr
        else:
            predict_taken = 0
            predict_history = 0

        outputs.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Compute next state (posedge updates)
        new_ghr = ghr
        new_pht = list(pht)

        # Training PHT update
        if train_valid:
            tidx = (train_pc ^ train_history) & 0x7F
            ctr = new_pht[tidx]
            if train_taken:
                if ctr < 3:
                    ctr += 1
            else:
                if ctr > 0:
                    ctr -= 1
            new_pht[tidx] = ctr

        # GHR update: training takes precedence
        if train_valid and train_mispredicted:
            new_ghr = train_history & 0x7F
        elif train_valid:
            # Non-mispredict training: does not update GHR (per candidate 1)
            pass
        elif predict_valid:
            # Shift in predicted direction
            new_ghr = ((ghr << 1) | predict_taken) & 0x7F

        ghr = new_ghr
        pht = new_pht

    return outputs
