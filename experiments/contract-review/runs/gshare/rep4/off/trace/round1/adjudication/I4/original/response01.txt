def reference(vectors):
    # State
    pht = [0] * 128          # 2-bit saturating counters
    ghr = 0                  # 7-bit global history register

    results = []

    for row in vectors:
        # Apply asynchronous reset (R6_async_reset)
        if row.get("areset", 0) == 1:
            pht = [0] * 128
            ghr = 0

        # Combinational prediction outputs (R1_predict_outputs)
        if row.get("areset", 0) == 1:
            predict_taken = 0
            predict_history = 0
        else:
            if row.get("predict_valid", 0) == 1:
                idx = (row["predict_pc"] & 0x7F) ^ (ghr & 0x7F)
                predict_taken = 1 if pht[idx] >= 2 else 0
                predict_history = ghr & 0x7F
            else:
                predict_taken = 0
                predict_history = 0

        results.append({
            "predict_taken": predict_taken & 1,
            "predict_history": predict_history & 0x7F,
        })

        # Sequential updates at posedge (skip if reset active)
        if row.get("areset", 0) == 1:
            continue

        train_valid = row.get("train_valid", 0)
        train_mispredicted = row.get("train_mispredicted", 0)
        predict_valid = row.get("predict_valid", 0)

        # R3_train_pht_update: update PHT toward train_taken
        if train_valid == 1:
            tidx = (row["train_pc"] & 0x7F) ^ (row["train_history"] & 0x7F)
            c = pht[tidx]
            if row.get("train_taken", 0) == 1:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            pht[tidx] = c

        # R4_train_mispredict_ghr_recover / R5_train_precedence_over_predict
        if train_valid == 1 and train_mispredicted == 1:
            # GHR restored to train_history shifted with train_taken
            ghr = ((row["train_history"] & 0x7F) << 1) | (row.get("train_taken", 0) & 1)
            ghr &= 0x7F
        elif predict_valid == 1:
            # R2_predict_ghr_update: shift in predicted direction
            ghr = ((ghr << 1) | (predict_taken & 1)) & 0x7F

    return results
