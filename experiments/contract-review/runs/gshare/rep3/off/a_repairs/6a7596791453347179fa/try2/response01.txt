def reference(vectors):
    # State
    pht = [0] * 128          # 2-bit saturating counters
    ghr = 0                  # 7-bit global history register

    out = []

    for row in vectors:
        areset = row.get('areset', 0)
        predict_valid = row.get('predict_valid', 0)
        predict_pc = row.get('predict_pc', 0) & 0x7F
        train_valid = row.get('train_valid', 0)
        train_taken = row.get('train_taken', 0)
        train_mispredicted = row.get('train_mispredicted', 0)
        train_history = row.get('train_history', 0) & 0x7F
        train_pc = row.get('train_pc', 0) & 0x7F

        # R7_async_reset: asynchronous active-high reset
        if areset:
            pht = [0] * 128
            ghr = 0
            out.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # R1_predict_outputs / R8_index_xor: combinational prediction
        predict_taken = 0
        predict_history = 0
        if predict_valid:
            # R8_index_xor: index = pc xor ghr
            idx = (predict_pc ^ ghr) & 0x7F
            # R6_predict_sees_pre_train_pht: prediction sees pre-train PHT
            predict_taken = 1 if pht[idx] >= 2 else 0
            # R1_predict_outputs: output GHR used for prediction
            predict_history = ghr

        out.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Sequential updates at posedge
        # R5_train_priority_over_predict_ghr: training takes precedence for GHR
        if train_valid:
            # R9_train_index_uses_train_history: index uses train_pc xor train_history
            tidx = (train_pc ^ train_history) & 0x7F
            # R3_train_updates_pht: 2-bit saturating counter update
            c = pht[tidx]
            if train_taken:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            pht[tidx] = c

            # R4_train_mispredicted_recovers_ghr: recover GHR
            if train_mispredicted:
                # GHR becomes train_history with the taken bit shifted in
                ghr = ((train_history << 1) | (train_taken & 1)) & 0x7F
            else:
                # Normal training: shift in actual outcome
                ghr = ((ghr << 1) | (train_taken & 1)) & 0x7F
        elif predict_valid:
            # R2_predict_updates_ghr: update GHR with predicted direction
            ghr = ((ghr << 1) | (predict_taken & 1)) & 0x7F

    return out
