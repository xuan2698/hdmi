def reference(vectors):
    # State
    pht = [0] * 128  # 2-bit saturating counters
    ghr = 0          # 7-bit global history register

    outputs = []

    for row in vectors:
        areset = row['areset']
        predict_valid = row['predict_valid']
        predict_pc = row['predict_pc']
        train_valid = row['train_valid']
        train_taken = row['train_taken']
        train_mispredicted = row['train_mispredicted']
        train_history = row['train_history']
        train_pc = row['train_pc']

        # R7_async_reset: asynchronous active-high reset
        if areset:
            pht = [0] * 128
            ghr = 0
            outputs.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # Combinational outputs (before posedge updates)
        # R1_predict_outputs: predict outputs when predict_valid
        # R8_index_xor: index = pc XOR ghr
        # R6_predict_sees_pre_train_pht: prediction sees pre-train PHT
        if predict_valid:
            idx = (predict_pc ^ ghr) & 0x7F
            counter = pht[idx]
            predict_taken = 1 if counter >= 2 else 0
            predict_history = ghr
        else:
            predict_taken = 0
            predict_history = 0

        outputs.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Sequential updates at posedge
        # R3_train_updates_pht: train updates PHT
        # R9_train_index_uses_train_history: train index uses train_history
        if train_valid:
            tidx = (train_pc ^ train_history) & 0x7F
            c = pht[tidx]
            if train_taken:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            pht[tidx] = c

        # GHR update: training takes precedence over prediction
        # R5_train_priority_over_predict_ghr
        # R4_train_mispredicted_recovers_ghr
        # R2_predict_updates_ghr
        if train_valid:
            if train_mispredicted:
                # Recover GHR to state after mispredicting branch
                ghr = ((train_history << 1) | train_taken) & 0x7F
            # else: training without mispredict does not change GHR
        elif predict_valid:
            ghr = ((ghr << 1) | predict_taken) & 0x7F

    return outputs
