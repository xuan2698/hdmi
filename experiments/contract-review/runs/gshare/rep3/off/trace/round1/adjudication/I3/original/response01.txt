def reference(vectors):
    # State
    pht = [0] * 128  # 2-bit saturating counters, init 0
    ghr = 0          # 7-bit global history register

    out = []

    def predict_taken_for(pc, hist):
        idx = (pc ^ hist) & 0x7F
        return 1 if pht[idx] >= 2 else 0

    for row in vectors:
        areset = row.get('areset', 0)
        predict_valid = row.get('predict_valid', 0)
        predict_pc = row.get('predict_pc', 0) & 0x7F
        train_valid = row.get('train_valid', 0)
        train_taken = row.get('train_taken', 0)
        train_mispredicted = row.get('train_mispredicted', 0)
        train_history = row.get('train_history', 0) & 0x7F
        train_pc = row.get('train_pc', 0) & 0x7F

        # R7_async_reset: async active-high reset forces outputs to 0 and clears state
        if areset:
            pht = [0] * 128
            ghr = 0
            out.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # R1_predict_outputs: combinational prediction using current PHT and GHR
        # R6_predict_sees_pre_train_pht: prediction uses PHT state before this cycle's training
        if predict_valid:
            predict_taken = predict_taken_for(predict_pc, ghr)
            predict_history = ghr
        else:
            predict_taken = 0
            predict_history = 0

        out.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Compute next-state updates at posedge
        new_pht = list(pht)
        new_ghr = ghr

        # R3_train_updates_pht: train PHT using train_history for indexing
        if train_valid:
            idx = (train_pc ^ train_history) & 0x7F
            c = new_pht[idx]
            if train_taken:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            new_pht[idx] = c

        # R4_train_mispredicted_recovers_ghr: recover GHR to train_history + taken
        # R5_train_priority_over_predict_ghr: training takes precedence over prediction for GHR
        if train_valid:
            if train_mispredicted:
                new_ghr = ((train_history << 1) | (train_taken & 1)) & 0x7F
            # If not mispredicted, training does not modify GHR (only PHT)
        elif predict_valid:
            # R2_predict_updates_ghr: update GHR with predicted direction
            new_ghr = ((ghr << 1) | (predict_taken & 1)) & 0x7F

        pht = new_pht
        ghr = new_ghr

    return out
