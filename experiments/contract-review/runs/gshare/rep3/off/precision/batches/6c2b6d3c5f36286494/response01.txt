    # state
    pht = [0]*128
    ghr = 0
    out = []
    for v in vectors:
        areset = v['areset']
        if areset:
            # async reset: outputs 0, state reset
            pht = [0]*128
            ghr = 0
            out.append({'predict_taken':0,'predict_history':0})
            continue
        # combinational outputs
        pv = v['predict_valid']
        if pv:
            idx = (v['predict_pc'] ^ ghr) & 0x7f
            pt = (pht[idx] >> 1) & 1
            ph = ghr
        else:
            pt = 0
            ph = 0
        out.append({'predict_taken':pt,'predict_history':ph})
        # posedge updates
        tv = v['train_valid']
        if tv:
            tidx = (v['train_pc'] ^ v['train_history']) & 0x7f
            c = pht[tidx]
            if v['train_taken']:

            else:

            pht[tidx] = c
            if v['train_mispredicted']:
                ghr = ((v['train_history'] << 1) | v['train_taken']) & 0x7f
        elif pv:
            ghr = ((ghr << 1) | pt) & 0x7f
    return out
