def reference(vectors):
    # State
    pht = [0] * 128          # 2-bit saturating counters
    ghr = 0                  # 7-bit global history register

    outputs = []

    for v in vectors:
        areset = v['areset']
        predict_valid = v['predict_valid']
        predict_pc = v['predict_pc']
        train_valid = v['train_valid']
        train_taken = v['train_taken']
        train_mispredicted = v['train_mispredicted']
        train_history = v['train_history']
        train_pc = v['train_pc']

        # Asynchronous reset: outputs forced to 0, state reset
        if areset:
            pht = [0] * 128
            ghr = 0
            outputs.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # Combinational prediction outputs (see pre-train PHT state)
        if predict_valid:
            idx = (predict_pc ^ ghr) & 0x7F
            counter = pht[idx]
            predict_taken = 1 if counter >= 2 else 0
            predict_history = ghr
        else:
            predict_taken = 0
            predict_history = 0

        outputs.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Sequential updates at posedge
        # Training takes precedence over prediction for GHR update
        if train_valid:
            # Update PHT using train_history for index
            tidx = (train_pc ^ train_history) & 0x7F
            c = pht[tidx]
            if train_taken:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            pht[tidx] = c

            # GHR update
            if train_mispredicted:
                # Recover GHR to state immediately after mispredicting branch
                ghr = ((train_history << 1) | train_taken) & 0x7F
            else:
                # Normal training: shift in actual outcome
                ghr = ((ghr << 1) | train_taken) & 0x7F
        elif predict_valid:
            # Prediction updates GHR with predicted direction
            ghr = ((ghr << 1) | predict_taken) & 0x7F

    return outputs
