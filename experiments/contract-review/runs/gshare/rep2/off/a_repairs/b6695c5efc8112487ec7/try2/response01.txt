def reference(vectors):
    # State
    ghr = 0
    pht = [0] * 128  # 2-bit saturating counters, init 0

    outputs = []

    for row in vectors:
        # Read inputs
        areset = row.get('areset', 0)
        predict_valid = row.get('predict_valid', 0)
        predict_pc = row.get('predict_pc', 0)
        train_valid = row.get('train_valid', 0)
        train_taken = row.get('train_taken', 0)
        train_mispredicted = row.get('train_mispredicted', 0)
        train_history = row.get('train_history', 0)
        train_pc = row.get('train_pc', 0)

        # R1_reset: asynchronous active-high reset
        if areset:
            ghr = 0
            pht = [0] * 128
            outputs.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # Combinational outputs (before posedge updates)
        # R2_predict: predict_valid=1 -> produce prediction
        # R3_predict_no_valid: predict_valid=0 -> predict_taken=0, predict_history=current ghr
        if predict_valid:
            idx = (predict_pc & 0x7F) ^ (ghr & 0x7F)
            counter = pht[idx]
            predict_taken = 1 if counter >= 2 else 0
            predict_history = ghr & 0x7F
        else:
            predict_taken = 0
            predict_history = ghr & 0x7F

        outputs.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Posedge updates
        # R6_pht_train: train_valid -> update PHT
        if train_valid:
            idx = (train_pc & 0x7F) ^ (train_history & 0x7F)
            counter = pht[idx]
            if train_taken:
                if counter < 3:
                    counter += 1
            else:
                if counter > 0:
                    counter -= 1
            pht[idx] = counter

        # R5_ghr_update_train_mispredicted / R7_train_priority_over_predict
        # Training takes precedence over prediction for GHR update
        if train_valid and train_mispredicted:
            # Recover GHR to state immediately after mispredicting branch
            # = train_history shifted left with actual outcome inserted at LSB
            ghr = ((train_history << 1) | (train_taken & 1)) & 0x7F
        elif predict_valid:
            # R4_ghr_update_predict: shift in predicted direction
            ghr = ((ghr << 1) | (predict_taken & 1)) & 0x7F

    return outputs
