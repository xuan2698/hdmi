def reference(vectors):
    # State
    ghr = 0
    pht = [0] * 128

    out = []

    for v in vectors:
        areset = v.get('areset', 0)
        predict_valid = v.get('predict_valid', 0)
        predict_pc = v.get('predict_pc', 0)
        train_valid = v.get('train_valid', 0)
        train_taken = v.get('train_taken', 0)
        train_mispredicted = v.get('train_mispredicted', 0)
        train_history = v.get('train_history', 0)
        train_pc = v.get('train_pc', 0)

        # R1_reset: asynchronous active-high reset clears ghr and pht,
        # outputs forced to 0 in same cycle.
        if areset:
            ghr = 0
            pht = [0] * 128
            out.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # R2_predict: combinational prediction from pre-edge PHT and ghr.
        # R3_predict_no_valid: when predict_valid=0, predict_taken=0 and
        # predict_history=current ghr (assumed).
        if predict_valid:
            idx = (predict_pc & 0x7F) ^ (ghr & 0x7F)
            counter = pht[idx]
            predict_taken = 1 if counter >= 2 else 0
            predict_history = ghr & 0x7F
        else:
            predict_taken = 0
            predict_history = ghr & 0x7F

        out.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Compute next-state updates (applied at posedge).
        new_ghr = ghr

        # R6_pht_train: train PHT entry (uses train_pc XOR train_history).
        if train_valid:
            tidx = (train_pc & 0x7F) ^ (train_history & 0x7F)
            c = pht[tidx]
            if train_taken:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            pht[tidx] = c

        # R5_ghr_update_train_mispredicted: on mispredicted training,
        # ghr <= {train_history[5:0], train_taken}.
        # R7_train_priority_over_predict: training (mispredicted) takes
        # precedence over prediction for ghr update.
        if train_valid and train_mispredicted:
            new_ghr = ((train_history & 0x3F) << 1) | (train_taken & 1)
            new_ghr &= 0x7F
        elif predict_valid:
            # R4_ghr_update_predict: shift in predicted direction.
            new_ghr = ((ghr << 1) | predict_taken) & 0x7F

        ghr = new_ghr

    return out
