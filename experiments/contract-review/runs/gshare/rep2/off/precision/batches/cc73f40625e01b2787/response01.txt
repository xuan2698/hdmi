```python
def reference(vectors):
    ghr = 0
    pht = [0]*128
    out = []
    for v in vectors:
        areset = v['areset']
        if areset:
            ghr = 0
            pht = [0]*128
            out.append({'predict_taken':0, 'predict_history':0})
            continue
        # compute outputs
        pv = v['predict_valid']
        if pv:
            idx = (v['predict_pc'] ^ ghr) & 0x7F
            pt = 1 if pht[idx] >= 2 else 0
            ph = ghr
        else:
            pt = 0
            ph = ghr
        out.append({'predict_taken':pt, 'predict_history':ph})
        # update state at posedge
        tv = v['train_valid']
        tm = v['train_mispredicted']
        tt = v['train_taken']
        th = v['train_history']
        tp = v['train_pc']
        # PHT update
        if tv:
            idx = (tp ^ th) & 0x7F
            if tt:
                if pht[idx] < 3:
                    pht[idx] += 1
            else:
                if pht[idx] > 0:
                    pht[idx] -= 1
        # GHR update per model
        # model 1
        if tv and tm:
            ghr = ((th << 1) | tt) & 0x7F  # {th[5:0], tt}
        elif pv and not tv:
            ghr = ((ghr << 1) | pt) & 0x7F
    return out
```
