def reference(vectors):
    # State
    ghr = 0
    pht = [0] * 128  # 2-bit saturating counters, init 0

    outputs = []

    for vec in vectors:
        areset = vec.get('areset', 0)
        predict_valid = vec.get('predict_valid', 0)
        predict_pc = vec.get('predict_pc', 0)
        train_valid = vec.get('train_valid', 0)
        train_taken = vec.get('train_taken', 0)
        train_mispredicted = vec.get('train_mispredicted', 0)
        train_history = vec.get('train_history', 0)
        train_pc = vec.get('train_pc', 0)

        # R8_async_reset: asynchronous active-high reset
        if areset:
            ghr = 0
            pht = [0] * 128
            outputs.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # R1_predict_outputs: combinational prediction using current ghr and pht
        # R7_predict_sees_pre_train_pht: prediction sees PHT state before training
        # (training only modifies PHT at next posedge, so we read current pht here)
        if predict_valid:
            idx = (predict_pc ^ ghr) & 0x7F
            counter = pht[idx]
            predict_taken = 1 if counter >= 2 else 0
            predict_history = ghr
        else:
            # R2_predict_no_valid: outputs zero when no prediction requested
            predict_taken = 0
            predict_history = 0

        outputs.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Compute next state (posedge updates)
        next_ghr = ghr
        next_pht = list(pht)

        # R4_train_pht_update: train PHT on train_valid
        if train_valid:
            tidx = (train_pc ^ train_history) & 0x7F
            c = next_pht[tidx]
            if train_taken:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            next_pht[tidx] = c

        # R5_train_mispredict_ghr_recover: recover ghr on mispredict
        # R6_train_priority_over_predict_ghr: training takes precedence over prediction
        if train_valid and train_mispredicted:
            next_ghr = ((train_history << 1) | train_taken) & 0x7F
        elif predict_valid:
            # R3_ghr_update_predict: update ghr for predicted branch
            next_ghr = ((ghr << 1) | predict_taken) & 0x7F

        ghr = next_ghr
        pht = next_pht

    return outputs
