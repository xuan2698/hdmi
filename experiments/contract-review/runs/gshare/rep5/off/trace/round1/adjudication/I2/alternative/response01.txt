def reference(vectors):
    # State: pht (128 entries, 2-bit counters), ghr (7-bit)
    # Reset is asynchronous active-high.
    # Contract rules implemented below with IDs in comments.

    # Initialize state (power-up unspecified; use zeros)
    pht = [0] * 128
    ghr = 0

    outputs = []

    for vec in vectors:
        areset = vec.get('areset', 0)
        predict_valid = vec.get('predict_valid', 0)
        predict_pc = vec.get('predict_pc', 0)
        train_valid = vec.get('train_valid', 0)
        train_taken = vec.get('train_taken', 0)
        train_mispredicted = vec.get('train_mispredicted', 0)
        train_history = vec.get('train_history', 0)
        train_pc = vec.get('train_pc', 0)

        # R7_async_reset: asynchronous active-high reset
        if areset:
            pht = [0] * 128
            ghr = 0
            outputs.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # Compute prediction outputs (same cycle, before any state update)
        # R1_predict_outputs: when predict_valid, produce predicted direction
        # and the GHR state used to make the prediction.
        # R6_predict_sees_pre_train_pht: prediction sees PHT state before
        # training (training only modifies PHT at next posedge).
        if predict_valid:
            idx = (predict_pc ^ ghr) & 0x7F
            counter = pht[idx]
            # 2-bit saturating counter: MSB determines taken
            pred_taken = (counter >> 1) & 1
            pred_history = ghr & 0x7F
        else:
            pred_taken = 0
            pred_history = 0

        outputs.append({'predict_taken': pred_taken, 'predict_history': pred_history})

        # Compute next-state updates (applied at posedge)
        new_pht = list(pht)
        new_ghr = ghr

        # R3_train_pht_update: train PHT on train_valid
        if train_valid:
            tidx = (train_pc ^ train_history) & 0x7F
            c = new_pht[tidx]
            if train_taken:
                if c < 3:
                    c += 1
            else:
                if c > 0:
                    c -= 1
            new_pht[tidx] = c

        # R5_train_priority_over_predict_ghr: training takes precedence over
        # prediction for GHR update.
        # R4_train_mispredict_ghr_recover: on mispredict, recover GHR to state
        # immediately after the mispredicting branch completes execution.
        # Candidate interpretation: GHR = train_history shifted left with
        # train_taken (i.e., (train_history << 1 | train_taken) & 0x7F).
        if train_valid:
            if train_mispredicted:
                new_ghr = ((train_history << 1) | train_taken) & 0x7F
            # If training but not mispredicted, GHR is not modified by training
            # (training only recovers GHR on mispredict per spec).
        elif predict_valid:
            # R2_predict_ghr_update: update GHR for predicted branch
            new_ghr = ((ghr << 1) | pred_taken) & 0x7F

        pht = new_pht
        ghr = new_ghr

    return outputs
