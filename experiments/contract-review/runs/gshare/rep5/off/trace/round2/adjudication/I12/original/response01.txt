def reference(vectors):
    # State
    pht = [0] * 128          # 2-bit saturating counters; reset to 0 (R7)
    ghr = 0                  # 7-bit global history register (R7)

    outputs = []

    def sat_update(counter, taken):
        # 2-bit saturating counter update
        if taken:
            return 3 if counter == 3 else counter + 1
        else:
            return 0 if counter == 0 else counter - 1

    for vec in vectors:
        areset = vec['areset']
        predict_valid = vec['predict_valid']
        predict_pc = vec['predict_pc'] & 0x7F
        train_valid = vec['train_valid']
        train_taken = vec['train_taken']
        train_mispredicted = vec['train_mispredicted']
        train_history = vec['train_history'] & 0x7F
        train_pc = vec['train_pc'] & 0x7F

        # R7_async_reset: asynchronous active-high reset
        if areset:
            pht = [0] * 128
            ghr = 0
            outputs.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # R1_predict_outputs: produce prediction using current PHT and ghr
        # R6_predict_sees_pre_train_pht: prediction uses pre-update PHT state
        if predict_valid:
            idx = (predict_pc ^ ghr) & 0x7F
            counter = pht[idx]
            predict_taken = 1 if counter >= 2 else 0
            predict_history = ghr
        else:
            predict_taken = 0
            predict_history = 0

        outputs.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Compute next-state updates (applied at posedge)
        new_pht = list(pht)
        new_ghr = ghr

        # R3_train_pht_update: train PHT
        # R6: training modifies PHT at next posedge (so prediction above saw old value)
        if train_valid:
            tidx = (train_pc ^ train_history) & 0x7F
            new_pht[tidx] = sat_update(new_pht[tidx], train_taken)

        # R4_train_mispredict_ghr_recover: recover ghr on mispredict
        # R5_train_priority_over_predict_ghr: training takes precedence over prediction ghr update
        if train_valid:
            if train_mispredicted:
                # Recover to state immediately after mispredicting branch
                new_ghr = (train_history << 1 | train_taken) & 0x7F
            # else: training without mispredict does not change ghr
        elif predict_valid:
            # R2_predict_ghr_update: update ghr for predicted branch
            new_ghr = ((ghr << 1) | predict_taken) & 0x7F

        pht = new_pht
        ghr = new_ghr

    return outputs
