def reference(vectors):
    # State
    pht = [0] * 128  # 2-bit saturating counters; reset to 0 (R7)
    ghr = 0          # 7-bit global history register

    out = []

    def idx(pc, hist):
        return (pc ^ hist) & 0x7F

    def sat_inc(c):
        return c + 1 if c < 3 else 3

    def sat_dec(c):
        return c - 1 if c > 0 else 0

    for v in vectors:
        areset = v.get('areset', 0)
        predict_valid = v.get('predict_valid', 0)
        predict_pc = v.get('predict_pc', 0) & 0x7F
        train_valid = v.get('train_valid', 0)
        train_taken = v.get('train_taken', 0)
        train_mispredicted = v.get('train_mispredicted', 0)
        train_history = v.get('train_history', 0) & 0x7F
        train_pc = v.get('train_pc', 0) & 0x7F

        # R7_async_reset: asynchronous active-high reset
        if areset:
            pht = [0] * 128
            ghr = 0
            out.append({'predict_taken': 0, 'predict_history': 0})
            continue

        # R1_predict_outputs: same-cycle prediction outputs
        # R6_predict_sees_pre_train_pht: prediction uses pre-edge PHT state
        if predict_valid:
            pidx = idx(predict_pc, ghr)
            counter = pht[pidx]
            predict_taken = 1 if counter >= 2 else 0
            predict_history = ghr
        else:
            predict_taken = 0
            predict_history = 0

        out.append({'predict_taken': predict_taken, 'predict_history': predict_history})

        # Compute next-state updates (posedge)
        new_pht = list(pht)
        new_ghr = ghr

        # R3_train_pht_update: train PHT
        if train_valid:
            tidx = idx(train_pc, train_history)
            if train_taken:
                new_pht[tidx] = sat_inc(new_pht[tidx])
            else:
                new_pht[tidx] = sat_dec(new_pht[tidx])

        # R4_train_mispredict_ghr_recover: recover GHR on mispredict
        # R5_train_priority_over_predict_ghr: training takes precedence over prediction
        if train_valid:
            if train_mispredicted:
                new_ghr = (train_history << 1 | train_taken) & 0x7F
            # else: training without mispredict does not modify GHR
        elif predict_valid:
            # R2_predict_ghr_update: update GHR for predicted branch
            new_ghr = (ghr << 1 | predict_taken) & 0x7F

        pht = new_pht
        ghr = new_ghr

    return out
