def reference(vectors):
    q = 0
    reset = 0
    result = []
    for row in vectors:
        if isinstance(row, dict) and 'reset' in row:
            reset = int(row['reset']) & 1
        if reset:
            q = 0
        else:
            q = (q + 1) % 10
        result.append({'q': q})
        mid = row.get('mid') if isinstance(row, dict) else None
        if isinstance(mid, dict) and 'reset' in mid:
            reset = int(mid['reset']) & 1
    return result
