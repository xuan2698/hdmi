def reference(vectors):
    # State: count (0..9), initial value unspecified -> use 0 as concrete power-up.
    # R1: reset==1 at posedge -> q=0 (sync, active high, priority 1)
    # R2: reset==0 && q==9 at posedge -> q=0 (priority 2)
    # R3: reset==0 && q!=9 at posedge -> q=q+1 (priority 3)
    # Mid changes occur AFTER sampling, before next edge; carried to next row.
    count = 0
    out = []
    for row in vectors:
        # Sample inputs at posedge (values driven before edge)
        reset = row.get('reset', 0)
        # Compute next state per priority rules
        if reset == 1:
            # R1
            count = 0
        elif count == 9:
            # R2
            count = 0
        else:
            # R3
            count = count + 1
        # Output visible same cycle
        out.append({'q': count & 0xF})
        # Mid changes: applied after sampling, carried to next row.
        # (No mid inputs in this contract beyond reset, which is a normal input.)
    return out
