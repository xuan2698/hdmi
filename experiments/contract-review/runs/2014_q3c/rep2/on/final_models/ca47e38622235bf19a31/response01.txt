def reference(vectors):
    # rule valid_y_0_to_4: implement minimized Y0 and z; unused y states treated as don't-cares.
    cur = {'x': 0, 'y': 0}
    result = []
    for row in vectors:
        # Apply direct row drives before the sampled edge.
        if row:
            for key, value in row.items():
                if key != 'mid':
                    cur[key] = value

        x = cur.get('x', 0) & 1
        y = cur.get('y', 0) & 7

        y0 = y & 1
        y1 = (y >> 1) & 1
        y2 = (y >> 2) & 1

        # rule valid_y_0_to_4
        Y0 = (((1 - x) & y0) |
              ((1 - x) & y2) |
              (x & (1 - y2) & (1 - y0))) & 1
        z = (y2 | (y1 & y0)) & 1

        result.append({'Y0': Y0, 'z': z})

        # Optional mid changes occur after sampling and carry to next row.
        if row and 'mid' in row:
            mid = row['mid']
            if isinstance(mid, dict):
                for key, value in mid.items():
                    cur[key] = value
            elif isinstance(mid, list):
                for change in mid:
                    if isinstance(change, dict):
                        for key, value in change.items():
                            cur[key] = value

    return result
