def reference(vectors):
    outputs = []
    for v in vectors:
        x = v["x"] & 1
        y = v["y"] & 7

        # Rule valid_y_0_to_4: Boolean equations for specified present states y=000..100.
        if y <= 4:
            y0 = y & 1
            y1 = (y >> 1) & 1
            y2 = (y >> 2) & 1
            nx = 1 - x
            Y0 = ((nx & y0) | (nx & y2) | (x & (1 - y2) & (1 - y0))) & 1
            z = (y2 | (y1 & y0)) & 1
        else:
            # Rule global: unused present states y=101,110,111 use zero next-state bits and z=0.
            Y0 = 0
            z = 0

        outputs.append({"Y0": Y0, "z": z})

    return outputs
