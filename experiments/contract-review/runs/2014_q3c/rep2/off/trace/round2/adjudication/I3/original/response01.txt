def reference(vectors):
    # State table: next state Y[2:0] for (y, x)
    # y=000: x0->000, x1->001
    # y=001: x0->001, x1->100
    # y=010: x0->010, x1->001
    # y=011: x0->001, x1->010
    # y=100: x0->011, x1->100
    next_table = {
        (0, 0): 0b000, (0, 1): 0b001,
        (1, 0): 0b001, (1, 1): 0b100,
        (2, 0): 0b010, (2, 1): 0b001,
        (3, 0): 0b001, (3, 1): 0b010,
        (4, 0): 0b011, (4, 1): 0b100,
    }
    results = []
    for vec in vectors:
        x = vec['x'] & 1
        y = vec['y'] & 0b111
        # R_Y0: Y0 = LSB of next state selected by (y, x)
        if (y, x) in next_table:
            nxt = next_table[(y, x)]
        else:
            # undefined states 101,110,111 -> treat as 0 (candidate)
            nxt = 0
        Y0 = nxt & 1
        # R_z: z=1 for y=011 or y=100, else 0
        if y in (3, 4):
            z = 1
        else:
            z = 0
        results.append({'Y0': Y0, 'z': z})
    return results
