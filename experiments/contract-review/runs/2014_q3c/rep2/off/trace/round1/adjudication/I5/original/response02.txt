def reference(vectors):
    # R_Y0: combinational next-state LSB selection by (y, x)
    # R_z: combinational output z = 1 for y in {3, 4}
    # Table: y -> (Y_x0, Y_x1)
    table = {
        0: (0, 1),
        1: (1, 4),
        2: (2, 1),
        3: (1, 2),
        4: (3, 4),
    }
    out = []
    for v in vectors:
        x = v["x"] & 1
        y = v["y"] & 7
        # R_Y0: select next state, take LSB
        nxt = table[y][x]
        Y0 = nxt & 1
        # R_z: z=1 for states 011 (3) or 100 (4)
        z = 1 if (y == 3 or y == 4) else 0
        out.append({"Y0": Y0, "z": z})
    return out
