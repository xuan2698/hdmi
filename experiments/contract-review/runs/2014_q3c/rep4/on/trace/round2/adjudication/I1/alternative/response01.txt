def reference(vectors):
    results = []
    for vec in vectors:
        x = int(vec["x"]) & 1
        y = int(vec["y"])
        if y == 0:
            if x == 0:
                # Y0_z_S0_x0
                Y0, z = 0, 0
            else:
                # Y0_z_S0_x1
                Y0, z = 1, 0
        elif y == 1:
            if x == 0:
                # Y0_z_S1_x0
                Y0, z = 1, 0
            else:
                # Y0_z_S1_x1
                Y0, z = 0, 0
        elif y == 2:
            if x == 0:
                # Y0_z_S2_x0
                Y0, z = 0, 0
            else:
                # Y0_z_S2_x1
                Y0, z = 1, 0
        elif y == 3:
            if x == 0:
                # Y0_z_S3_x0
                Y0, z = 1, 1
            else:
                # Y0_z_S3_x1
                Y0, z = 0, 1
        elif y == 4:
            if x == 0:
                # Y0_z_S4_x0
                Y0, z = 1, 1
            else:
                # Y0_z_S4_x1
                Y0, z = 0, 1
        else:
            # Y0_z_unused
            Y0, z = 1, 1
        results.append({"Y0": Y0, "z": z})
    return results
