def reference(vectors):
    # State-assigned table:
    # y=000: x=0 -> Y=000, x=1 -> Y=001
    # y=001: x=0 -> Y=001, x=1 -> Y=100
    # y=010: x=0 -> Y=010, x=1 -> Y=001
    # y=011: x=0 -> Y=001, x=1 -> Y=010
    # y=100: x=0 -> Y=011, x=1 -> Y=100
    # z = 1 for y in {011, 100}, else 0

    # Next-state table indexed by (y, x)
    next_state = {
        (0b000, 0): 0b000, (0b000, 1): 0b001,
        (0b001, 0): 0b001, (0b001, 1): 0b100,
        (0b010, 0): 0b010, (0b010, 1): 0b001,
        (0b011, 0): 0b001, (0b011, 1): 0b010,
        (0b100, 0): 0b011, (0b100, 1): 0b100,
    }

    results = []
    for vec in vectors:
        y = vec["y"] & 0b111
        x = vec["x"] & 0b1

        # R_Y0: Y0 = next-state bit 0 as function of present state y and input x
        Y = next_state[(y, x)]
        Y0 = Y & 0b1

        # R_z: z = 1 when present state y is 011 or 100, else 0 (Moore output)
        z = 1 if y in (0b011, 0b100) else 0

        results.append({"Y0": Y0, "z": z})

    return results
