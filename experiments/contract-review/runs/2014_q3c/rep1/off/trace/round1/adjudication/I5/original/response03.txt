def reference(vectors):
    # R_Y0: Y0 = next-state bit 0 of Y[2:0] selected by x from present state y
    # R_z: z = 1 when present state y is 011 or 100, else 0
    # Next-state table indexed by present state y (0..7):
    #   y=0: x=0 -> 000, x=1 -> 001
    #   y=1: x=0 -> 001, x=1 -> 100
    #   y=2: x=0 -> 010, x=1 -> 001
    #   y=3: x=0 -> 001, x=1 -> 010
    #   y=4: x=0 -> 011, x=1 -> 100
    #   y=5,6,7: unspecified; treat as 000 (safe default)
    next_state = {
        0: (0b000, 0b001),
        1: (0b001, 0b100),
        2: (0b010, 0b001),
        3: (0b001, 0b010),
        4: (0b011, 0b100),
        5: (0b000, 0b000),
        6: (0b000, 0b000),
        7: (0b000, 0b000),
    }
    out = []
    for v in vectors:
        x = v["x"] & 1
        y = v["y"] & 7
        ns = next_state[y][x]
        Y0 = ns & 1  # R_Y0
        z = 1 if (y == 0b011 or y == 0b100) else 0  # R_z
        out.append({"Y0": Y0, "z": z})
    return out
