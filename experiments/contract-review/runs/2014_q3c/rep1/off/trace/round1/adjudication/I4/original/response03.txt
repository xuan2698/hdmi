def reference(vectors):
    # R_Y0: Y0 = next-state bit 0 of Y[2:0] selected by x from present state y
    # R_z: z = 1 when present state y is 011 or 100, else 0
    # State table: next state Y[2:0] for (y, x)
    # y=000: x=0->000, x=1->001
    # y=001: x=0->001, x=1->100
    # y=010: x=0->010, x=1->001
    # y=011: x=0->001, x=1->010
    # y=100: x=0->011, x=1->100
    next_state = {
        0: {0: 0, 1: 1},
        1: {0: 1, 1: 4},
        2: {0: 2, 1: 1},
        3: {0: 1, 1: 2},
        4: {0: 3, 1: 4},
    }
    out = []
    for v in vectors:
        y = v["y"]
        x = v["x"]
        ns = next_state[y][x]
        Y0 = ns & 1  # R_Y0
        z = 1 if y in (3, 4) else 0  # R_z
        out.append({"Y0": Y0, "z": z})
    return out
