def reference(vectors):
    # R_Y0: Y0 = next-state bit 0 of Y[2:0] selected by x from present state y
    # R_z: z = 1 when present state y is 011 or 100, else 0
    # State transition table (present y -> next Y for x=0, x=1)
    # 000 -> 000, 001
    # 001 -> 001, 100
    # 010 -> 010, 001
    # 011 -> 001, 010
    # 100 -> 011, 100
    next_state = {
        0: (0, 1),  # y=000: x=0 -> 000, x=1 -> 001
        1: (1, 4),  # y=001: x=0 -> 001, x=1 -> 100
        2: (2, 1),  # y=010: x=0 -> 010, x=1 -> 001
        3: (1, 2),  # y=011: x=0 -> 001, x=1 -> 010
        4: (3, 4),  # y=100: x=0 -> 011, x=1 -> 100
    }
    results = []
    for vec in vectors:
        x = vec["x"]
        y = vec["y"]
        # R_Y0: combinational next-state bit 0
        Y = next_state[y][x]
        Y0 = Y & 1
        # R_z: combinational output
        z = 1 if y in (3, 4) else 0
        results.append({"Y0": Y0, "z": z})
    return results
