def reference(vectors):
    # R_Y0: Y0 = next-state bit 0 of Y[2:0] selected by x from present state y
    # R_z: z = 1 when present state y is 011 or 100, else 0
    # Present state y[2:0] | Next state Y[2:0] x=0, Next state Y[2:0] x=1 | Output z
    # 000 | 000, 001 | 0
    # 001 | 001, 100 | 0
    # 010 | 010, 001 | 0
    # 011 | 001, 010 | 1
    # 100 | 011, 100 | 1
    next_state = {
        0: (0, 1),
        1: (1, 4),
        2: (2, 1),
        3: (1, 2),
        4: (3, 4),
    }
    out = []
    for v in vectors:
        x = v["x"]
        y = v["y"]
        # R_Y0: combinational, same cycle
        ns = next_state[y][x]
        Y0 = ns & 1
        # R_z: combinational, same cycle
        z = 1 if (y == 3 or y == 4) else 0
        out.append({"Y0": Y0, "z": z})
    return out
