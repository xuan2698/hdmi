def reference(vectors):
    result = []
    for row in vectors:
        x = row.get('x', 0) & 1
        y = row.get('y', 0) & 7

        # Y0_comb: For y in {0,1,2,3}: Y0 = (x XOR (y & 1)) & 1;
        # for y == 4: Y0 = (NOT x) & 1.
        # Candidate interpretation: default Y0=0 for y=5,6,7.
        if y <= 3:
            Y0 = (x ^ (y & 1)) & 1
        elif y == 4:
            Y0 = (1 - x) & 1
        else:  # y in {5,6,7}
            Y0 = 0

        # z_comb: For y in {0,1,2}: z = 0; for y in {3,4}: z = 1.
        # Unspecified invalid states default to 0.
        if y in (3, 4):
            z = 1
        else:
            z = 0

        result.append({'Y0': Y0, 'z': z})
    return result
