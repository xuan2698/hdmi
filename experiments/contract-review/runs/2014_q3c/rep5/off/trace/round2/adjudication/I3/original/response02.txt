def reference(vectors):
    # R_Y0 / R_z: combinational next-state bit 0 and output z from present state y and input x.
    # State-assigned table:
    #   y=000: x=0 -> Y=000, x=1 -> Y=001 ; z=0
    #   y=001: x=0 -> Y=001, x=1 -> Y=100 ; z=0
    #   y=010: x=0 -> Y=010, x=1 -> Y=001 ; z=0
    #   y=011: x=0 -> Y=001, x=1 -> Y=010 ; z=1
    #   y=100: x=0 -> Y=011, x=1 -> Y=100 ; z=1
    # Present state is y (3 bits). Next state Y[2:0] selected by x.
    # Y0 is next-state bit 0.
    next_state = {
        0: (0, 1),  # y=000: x=0 -> 000, x=1 -> 001
        1: (1, 4),  # y=001: x=0 -> 001, x=1 -> 100
        2: (2, 1),  # y=010: x=0 -> 010, x=1 -> 001
        3: (1, 2),  # y=011: x=0 -> 001, x=1 -> 010
        4: (3, 4),  # y=100: x=0 -> 011, x=1 -> 100
    }
    z_table = {0: 0, 1: 0, 2: 0, 3: 1, 4: 1}

    out = []
    for vec in vectors:
        y = vec["y"] & 0x7
        x = vec["x"] & 0x1
        # R_Y0: Y0 = next-state bit 0 as function of y and x per state-assigned table
        Y = next_state[y][x]
        Y0 = Y & 0x1
        # R_z: z = 1 for present states 011 and 100, else 0
        z = z_table[y]
        out.append({"Y0": Y0, "z": z})
    return out
