def reference(vectors):
    # Combinational Moore FSM outputs.
    # Next-state table: y -> (Y when x=0, Y when x=1)
    next_state = {
        0b000: (0b000, 0b001),
        0b001: (0b001, 0b100),
        0b010: (0b010, 0b001),
        0b011: (0b001, 0b010),
        0b100: (0b011, 0b100),
    }
    # z depends only on present state y
    z_table = {
        0b000: 0,
        0b001: 0,
        0b010: 0,
        0b011: 1,
        0b100: 1,
    }

    results = []
    for row in vectors:
        y = row["y"] & 0b111
        x = row["x"] & 0b1
        Y = next_state[y][x]
        Y0 = Y & 0b1
        z = z_table[y]
        results.append({"Y0": Y0, "z": z})
    return results
