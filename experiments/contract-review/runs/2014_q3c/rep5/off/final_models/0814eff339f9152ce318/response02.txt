def reference(vectors):
    # R_Y0, R_z: combinational next-state and output logic per state-assigned table.
    # State encoding: y[2:0] present state; Y[2:0] next state.
    # Table:
    #   y=000: x=0 -> Y=000, x=1 -> Y=001, z=0
    #   y=001: x=0 -> Y=001, x=1 -> Y=100, z=0
    #   y=010: x=0 -> Y=010, x=1 -> Y=001, z=0
    #   y=011: x=0 -> Y=001, x=1 -> Y=010, z=1
    #   y=100: x=0 -> Y=011, x=1 -> Y=100, z=1
    # Unused states (101,110,111) default to Y=000, z=0.

    def next_state(y, x):
        # R_Y0: next-state bit 0 as function of y and x
        table = {
            0: (0, 1),
            1: (1, 4),
            2: (2, 1),
            3: (1, 2),
            4: (3, 4),
        }
        if y in table:
            return table[y][x]
        return 0

    def out_z(y):
        # R_z: z = 1 for present states 011 and 100, else 0
        return 1 if y in (3, 4) else 0

    results = []
    for vec in vectors:
        x = vec["x"] & 1
        y = vec["y"] & 7
        Y = next_state(y, x)
        Y0 = Y & 1
        z = out_z(y)
        results.append({"Y0": Y0, "z": z})
    return results
