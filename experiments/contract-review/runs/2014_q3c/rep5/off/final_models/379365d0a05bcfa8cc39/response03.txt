def reference(vectors):
    # R_Y0, R_z: combinational next-state bit 0 and output z from present state y and input x.
    # State-assigned table (y -> next state for x=0, x=1):
    # 000 -> 000, 001
    # 001 -> 001, 100
    # 010 -> 010, 001
    # 011 -> 001, 010
    # 100 -> 011, 100
    next_state = {
        0: (0, 1),
        1: (1, 4),
        2: (2, 1),
        3: (1, 2),
        4: (3, 4),
    }
    z_table = {3: 1, 4: 1}

    results = []
    for vec in vectors:
        y = vec["y"] & 0x7
        x = vec["x"] & 0x1
        # R_Y0: Y0 = next-state bit 0
        ns = next_state[y][x]
        Y0 = ns & 0x1
        # R_z: z = 1 for present states 011 and 100
        z = z_table.get(y, 0)
        results.append({"Y0": Y0, "z": z})
    return results
