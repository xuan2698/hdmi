def reference(vectors):
    # State table: next_state[y][x] and output z[y]
    next_state = {
        0: {0: 0, 1: 1},
        1: {0: 1, 1: 4},
        2: {0: 2, 1: 1},
        3: {0: 1, 1: 2},
        4: {0: 3, 1: 4},
    }
    z_table = {0: 0, 1: 0, 2: 0, 3: 1, 4: 1}

    results = []
    for row in vectors:
        y = row["y"] & 0x7
        x = row["x"] & 0x1
        ns = next_state[y][x]
        results.append({
            "Y0": ns & 0x1,
            "z": z_table[y] & 0x1,
        })
    return results
