def reference(vectors):
    # R_Y0, R_z: combinational logic from present state y and input x.
    # Next-state table (state-assigned):
    #   y=000: x=0 -> 000, x=1 -> 001
    #   y=001: x=0 -> 001, x=1 -> 100
    #   y=010: x=0 -> 010, x=1 -> 001
    #   y=011: x=0 -> 001, x=1 -> 010
    #   y=100: x=0 -> 011, x=1 -> 100
    # Y0 is next-state bit 0. z is Moore output: 1 for y in {011, 100}.
    next_state = {
        (0, 0): 0b000, (0, 1): 0b001,
        (1, 0): 0b001, (1, 1): 0b100,
        (2, 0): 0b010, (2, 1): 0b001,
        (3, 0): 0b001, (3, 1): 0b010,
        (4, 0): 0b011, (4, 1): 0b100,
    }
    results = []
    for vec in vectors:
        x = vec["x"] & 1
        y = vec["y"] & 0b111
        ns = next_state[(y, x)]
        Y0 = ns & 1
        z = 1 if y in (0b011, 0b100) else 0
        results.append({"Y0": Y0, "z": z})
    return results
