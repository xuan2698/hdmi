def reference(vectors):
    # R_Y0: Y0 = next-state bit 0, combinational from present state y and input x
    # R_z:  z = 1 when present state y is 011 or 100, else 0 (Moore output)
    # State-assigned table:
    #   y=000: x=0 -> Y=000, x=1 -> Y=001
    #   y=001: x=0 -> Y=001, x=1 -> Y=100
    #   y=010: x=0 -> Y=010, x=1 -> Y=001
    #   y=011: x=0 -> Y=001, x=1 -> Y=010
    #   y=100: x=0 -> Y=011, x=1 -> Y=100
    # Build next-state map keyed by (y, x) -> Y[2:0]
    next_state = {
        (0b000, 0): 0b000,
        (0b000, 1): 0b001,
        (0b001, 0): 0b001,
        (0b001, 1): 0b100,
        (0b010, 0): 0b010,
        (0b010, 1): 0b001,
        (0b011, 0): 0b001,
        (0b011, 1): 0b010,
        (0b100, 0): 0b011,
        (0b100, 1): 0b100,
    }
    results = []
    for vec in vectors:
        y = int(vec["y"]) & 0x7
        x = int(vec["x"]) & 0x1
        Y = next_state[(y, x)]
        Y0 = Y & 0x1
        z = 1 if y in (0b011, 0b100) else 0
        results.append({"Y0": Y0, "z": z})
    return results
