def reference(vectors):
    # R_Y0, R_z: combinational logic from present state y and input x.
    # State-assigned table:
    #   y=000: x=0 -> Y=000, x=1 -> Y=001, z=0
    #   y=001: x=0 -> Y=001, x=1 -> Y=100, z=0
    #   y=010: x=0 -> Y=010, x=1 -> Y=001, z=0
    #   y=011: x=0 -> Y=001, x=1 -> Y=010, z=1
    #   y=100: x=0 -> Y=011, x=1 -> Y=100, z=1
    # Next-state table indexed by (y, x) -> Y[2:0]
    next_state = {
        (0, 0): 0b000, (0, 1): 0b001,
        (1, 0): 0b001, (1, 1): 0b100,
        (2, 0): 0b010, (2, 1): 0b001,
        (3, 0): 0b001, (3, 1): 0b010,
        (4, 0): 0b011, (4, 1): 0b100,
    }
    # R_z: Moore output z=1 for y in {011, 100}
    z_table = {0: 0, 1: 0, 2: 0, 3: 1, 4: 1}

    results = []
    for vec in vectors:
        x = vec["x"] & 1
        y = vec["y"] & 0b111
        Y = next_state[(y, x)]
        Y0 = Y & 1  # R_Y0: next-state bit 0
        z = z_table[y]  # R_z: Moore output from present state
        results.append({"Y0": Y0, "z": z})
    return results
