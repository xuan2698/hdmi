def reference(vectors):
    # R_Y0: Y0 = next-state bit 0, combinational from present state y and input x
    # R_z: z = 1 when present state y is 011 or 100, else 0 (Moore output)
    # State-assigned table:
    #   y=000: x=0 -> Y=000, x=1 -> Y=001
    #   y=001: x=0 -> Y=001, x=1 -> Y=100
    #   y=010: x=0 -> Y=010, x=1 -> Y=001
    #   y=011: x=0 -> Y=001, x=1 -> Y=010
    #   y=100: x=0 -> Y=011, x=1 -> Y=100
    # Next-state table indexed by (y, x) -> Y[2:0]
    next_state = {
        (0, 0): 0b000, (0, 1): 0b001,
        (1, 0): 0b001, (1, 1): 0b100,
        (2, 0): 0b010, (2, 1): 0b001,
        (3, 0): 0b001, (3, 1): 0b010,
        (4, 0): 0b011, (4, 1): 0b100,
    }
    # R_z: Moore output based on present state
    z_table = {0: 0, 1: 0, 2: 0, 3: 1, 4: 1}

    results = []
    for vec in vectors:
        y = vec["y"] & 0x7
        x = vec["x"] & 0x1
        Y = next_state[(y, x)]
        Y0 = Y & 0x1  # R_Y0
        z = z_table[y]  # R_z
        results.append({"Y0": Y0, "z": z})
    return results
