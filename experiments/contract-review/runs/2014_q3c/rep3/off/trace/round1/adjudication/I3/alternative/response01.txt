def reference(vectors):
    # R_Y0, R_z: combinational logic from present state y and input x.
    # State-assigned table:
    #   y=000: x=0 -> Y=000, x=1 -> Y=001 ; z=0
    #   y=001: x=0 -> Y=001, x=1 -> Y=100 ; z=0
    #   y=010: x=0 -> Y=010, x=1 -> Y=001 ; z=0
    #   y=011: x=0 -> Y=001, x=1 -> Y=010 ; z=1
    #   y=100: x=0 -> Y=011, x=1 -> Y=100 ; z=1
    # Undefined states 101,110,111: Y0=0, z=0 (don't-care interpretation).
    results = []
    for v in vectors:
        x = v["x"] & 1
        y = v["y"] & 7
        if y == 0:
            Y = 0 if x == 0 else 1
            z = 0
        elif y == 1:
            Y = 1 if x == 0 else 4
            z = 0
        elif y == 2:
            Y = 2 if x == 0 else 1
            z = 0
        elif y == 3:
            Y = 1 if x == 0 else 2
            z = 1
        elif y == 4:
            Y = 3 if x == 0 else 4
            z = 1
        else:
            Y = 0
            z = 0
        Y0 = Y & 1
        results.append({"Y0": Y0, "z": z})
    return results
