def reference(vectors):
    # Initialize concrete input state; unspecified power-up values use 0.
    cur_x = 0
    cur_y = 0
    results = []

    for row in vectors:
        # Drive row values before the sampled edge.
        if isinstance(row, dict):
            if "x" in row:
                cur_x = int(row["x"]) & 1
            if "y" in row:
                cur_y = int(row["y"]) & 7
        elif isinstance(row, (list, tuple)):
            if len(row) >= 1:
                cur_x = int(row[0]) & 1
            if len(row) >= 2:
                cur_y = int(row[1]) & 7

        x = cur_x & 1
        y = cur_y & 7

        Y0 = 0
        z = 0

        # r_y0_000: y == 0 (000); x=0 -> next 000 gives Y0=0, x=1 -> next 001 gives Y0=1; z=0.
        if y == 0:
            Y0 = 1 if x == 1 else 0
            z = 0
        # r_y0_001: y == 1 (001); x=0 -> next 001 gives Y0=1, x=1 -> next 100 gives Y0=0; z=0.
        elif y == 1:
            Y0 = 1 if x == 0 else 0
            z = 0
        # r_y0_010: y == 2 (010); x=0 -> next 010 gives Y0=0, x=1 -> next 001 gives Y0=1; z=0.
        elif y == 2:
            Y0 = 1 if x == 1 else 0
            z = 0
        # r_y0_011: y == 3 (011); x=0 -> next 001 gives Y0=1, x=1 -> next 010 gives Y0=0; z=1.
        elif y == 3:
            Y0 = 1 if x == 0 else 0
            z = 1
        # r_y0_100: y == 4 (100); x=0 -> next 011 gives Y0=1, x=1 -> next 100 gives Y0=0; z=1.
        elif y == 4:
            Y0 = 1 if x == 0 else 0
            z = 1
        else:
            # Unspecified present-state rows: keep outputs concrete and unsigned.
            Y0 = 0
            z = 0

        results.append({"Y0": int(Y0 & 1), "z": int(z & 1)})

        # Optional mid input changes occur AFTER sampling and carry to the next row.
        if isinstance(row, dict) and "mid" in row:
            mid = row["mid"]
            mids = mid if isinstance(mid, (list, tuple)) and not isinstance(mid, dict) else [mid]
            for change in mids:
                if isinstance(change, dict):
                    if "x" in change:
                        cur_x = int(change["x"]) & 1
                    if "y" in change:
                        cur_y = int(change["y"]) & 7

    return results
