def reference(vectors):
    # Contract rules:
    # hold_when_a_high: on posedge, if a==1, q holds (priority 1)
    # increment_when_a_low: on posedge, if a==0, q increments mod 8
    # global interpretation: increment q on posedge iff a sampled at previous posedge was 0; hold otherwise.
    # Power-up q is unspecified; we start at 0 (concrete unsigned value required).
    # Each row: values driven before sampled posedge; mid changes occur after sampling,
    # carried to start of next row. We process rows sequentially, sampling a at each
    # posedge, then applying mid changes for the next row.

    q = 0  # power-up unspecified; concrete start value
    prev_a = None  # a sampled at previous posedge (for global interpretation)
    results = []

    for row in vectors:
        # Determine a value at this posedge: row's 'a' unless a mid change from
        # previous row overrides it. Mid changes are carried to start of next row,
        # so they take effect before this row's posedge sampling.
        a_now = row.get('a', 0)
        if 'mid' in row and row['mid'] is not None:
            # mid changes occur AFTER sampling of previous row, before this row's posedge
            mid = row['mid']
            if isinstance(mid, dict) and 'a' in mid:
                a_now = mid['a']

        # Apply posedge update based on global interpretation:
        # increment iff a sampled at previous posedge was 0; hold otherwise.
        # For the first posedge, prev_a is None -> treat as hold (no prior sample).
        if prev_a is not None and prev_a == 0:
            q = (q + 1) & 0x7  # increment_when_a_low (global interpretation)
        # else hold_when_a_high / no prior sample -> hold

        # Record output for this row (same_cycle visible)
        results.append({'q': q & 0x7})

        # Sample a at this posedge for next cycle's decision
        prev_a = a_now

    return results
