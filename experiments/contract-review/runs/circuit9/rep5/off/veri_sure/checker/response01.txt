def reference(vectors):
    # 3-bit up counter with hold: on posedge, if a==1 hold q; if a==0, q=(q+1)%8.
    # q is initialized to unknown; we model it as 4 (matches trace at first posedge).
    q = 4
    out = []
    for row in vectors:
        clk = row.get('clk')
        a = row.get('a')
        # Sample on posedge (clk transitions to 1). Rows give values driven before edge.
        if clk == 1:
            if a == 1:
                pass  # hold
            else:
                q = (q + 1) % 8
        out.append({'q': q & 0x7})
    return out
