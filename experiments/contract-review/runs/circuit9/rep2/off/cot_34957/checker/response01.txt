    # Sampled a at posedge = printed a from previous row (mid changes after sampling)

    carried_a = None  # input value carried into next posedge
    for row in vectors:
        clk = row['clk']

        if clk == 1:
            # posedge: sample carried_a (from previous row)
            sampled = carried_a if carried_a is not None else a
            if sampled == 1:
                q = 4
            else:
                if q is None:

        # after sampling, mid change: carried_a becomes this row's printed a
        carried_a = a
        # output q (concrete)
        out_q = q if q is not None else 0
        results.append({'q': out_q})
    return results
