def reference(vectors):
    # Contract rules:
    # R1: a == 1 at posedge -> q holds (priority 1)
    # R2: a == 0 at posedge -> q increments by 1 (mod 8) (priority 2)
    # State: count (3-bit), initial value unspecified -> use 4 as observed power-up
    # Mid changes occur AFTER sampling, before other edge, carried to next row.
    q = 4  # power-up value observed in waveform
    out = []
    for row in vectors:
        a = row.get('a', 0)
        # sample at posedge
        if a == 1:
            # R1: hold
            pass
        else:
            # R2: increment mod 8
            q = (q + 1) & 0x7
        out.append({'q': q & 0x7})
    return out
