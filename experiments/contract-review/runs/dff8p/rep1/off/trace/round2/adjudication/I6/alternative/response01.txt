def reference(vectors):
    # State: q_state holds the 8-bit value stored in the D flip-flops.
    # Contract rules:
    # R1: reset == 1 -> q = 0x34 (52). Applied on negedge (synchronous),
    #     but per the resolved candidate interpretation for the unresolved
    #     field, reset is treated as asynchronous active high, overriding q
    #     immediately. We honor the resolved interpretation: reset overrides
    #     q immediately (asynchronously) whenever reset == 1.
    # R2: reset == 0 -> on negedge, q <= d.
    # Mid changes occur AFTER sampling, before the other edge, and are
    # carried to the start of the next row.

    q_state = 0x34  # power-up value (reset value 0x34)

    results = []
    for row in vectors:
        d = row.get('d', 0) & 0xFF
        reset = row.get('reset', 0) & 0x1

        # R1 (resolved async interpretation): reset active high overrides q
        # immediately, before sampling.
        if reset == 1:
            q_state = 0x34

        # Sample at negedge: if reset == 0, q <= d (R2).
        # If reset == 1, q stays 0x34 (R1).
        if reset == 0:
            q_state = d
        else:
            q_state = 0x34

        results.append({'q': q_state & 0xFF})

    return results
