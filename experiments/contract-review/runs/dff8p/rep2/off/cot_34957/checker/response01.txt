def reference(vectors):
    # 8 D flip-flops, active-high synchronous reset to 0x34, negedge clk.
    # q updates only on sampled negedge. reset=1 -> q=0x34, else q=d.
    # Power-up unspecified; we model as 0x34 (any value acceptable pre-reset,
    # but we must return concrete values). We start with 0x34.
    q = 0x34
    out = []
    for row in vectors:
        # Sample inputs driven before the negedge
        d = row.get('d', 0) & 0xFF
        reset = row.get('reset', 0) & 0x1
        # Negedge update
        if reset:
            q = 0x34
        else:
            q = d
        out.append({'q': q & 0xFF})
    return out
