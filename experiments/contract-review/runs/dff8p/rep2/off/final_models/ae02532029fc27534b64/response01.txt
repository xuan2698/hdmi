```python
def reference(vectors):
    # R1_sync_reset: reset==1 at negedge -> q=0x34 (priority 1)
    # R2_dff_capture: reset==0 at negedge -> q=d (priority 2)
    # R3_reset_value: reset value 0x34 (52)
    # State Q: 8-bit, reset 0x34, updates at negedge.
    # Mid changes occur AFTER sampling, before other edge, carried to next row.

    Q = 0x34  # power-up (R3_reset_value)
    pending = {}  # mid overrides carried to next row
    out = []
    for row in vectors:
        # Apply pending mid overrides from previous row
        d = row.get('d', 0) & 0xFF
        reset = row.get('reset', 0) & 0x1
        if 'd' in pending:
            d = pending['d']
        if 'reset' in pending:
            reset = pending['reset']
        pending = {}

        # negedge sampling
        if reset == 1:
            Q = 0x34  # R1_sync_reset, R3_reset_value
        else:
            Q = d  # R2_dff_capture

        out.append({'q': Q & 0xFF})

        # mid changes after sampling, carried to next row
        mid = row.get('mid')
        if isinstance(mid, dict):
            if 'd' in mid:
                pending['d'] = mid['d'] & 0xFF
            if 'reset' in mid:
                pending['reset'] = mid['reset'] & 0x1

    return out
```
