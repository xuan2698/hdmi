def reference(vectors):
    # State: Q is the 8-bit flip-flop output register, reset value 0x34 (52).
    Q = 0x34  # power-up / initial state
    out = []
    for row in vectors:
        d = row.get('d', 0) & 0xFF
        reset = row.get('reset', 0) & 0x1
        # R1: active high synchronous reset, negedge, priority 1
        # R2: negedge triggered DFF load when reset == 0, priority 2
        if reset == 1:
            Q = 0x34  # R1
        else:
            Q = d  # R2
        out.append({'q': Q & 0xFF})
    return out
