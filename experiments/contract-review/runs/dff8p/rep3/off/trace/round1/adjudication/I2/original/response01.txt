def reference(vectors):
    # State: Q is the 8-bit flip-flop output register, reset value 0x34 (52).
    Q = 0x34  # power-up state (unspecified in spec, but must be concrete)
    out = []
    for row in vectors:
        d = row.get('d', 0) & 0xFF
        reset = row.get('reset', 0) & 0x1
        # R1: active high synchronous reset on negedge -> Q = 0x34
        # R2: reset == 0 on negedge -> Q = d
        # Both rules update at negedge; sample after negedge settled.
        if reset == 1:
            Q = 0x34  # R1
        else:
            Q = d  # R2
        out.append({'q': Q & 0xFF})
    return out
