def reference(vectors):
    # State: 32-bit speculative global branch history register.
    history_reg = 0

    # Current input values (default 0 at power-up).
    inputs = {
        'areset': 0,
        'predict_valid': 0,
        'predict_taken': 0,
        'train_mispredicted': 0,
        'train_taken': 0,
        'train_history': 0
    }

    outputs = []

    for row in vectors:
        # Apply row inputs driven before the sampled posedge edge.
        for k in inputs:
            if k in row:
                inputs[k] = row[k]

        # R1_async_reset: asynchronous active-high reset forces history to 0.
        if inputs['areset'] == 1:
            history_reg = 0

        # Clocked updates occur only when reset is not active.
        if inputs['areset'] == 0:
            if inputs['train_mispredicted'] == 1:
                # R2_train_mispredicted_load (alternative interpretation):
                # Load via MSB-side insertion: new = ((train_history >> 1) | (train_taken << 31)) & 0xFFFFFFFF.
                th = inputs['train_history'] & 0xFFFFFFFF
                tt = inputs['train_taken'] & 1
                history_reg = ((th >> 1) | (tt << 31)) & 0xFFFFFFFF
            elif inputs['predict_valid'] == 1:
                # R3_predict_update: shift in predict_taken at LSB side.
                pt = inputs['predict_taken'] & 1
                history_reg = ((history_reg << 1) | pt) & 0xFFFFFFFF
            else:
                # R4_hold: no update, retain previous value.
                history_reg = history_reg

        # Sample output after posedge has settled.
        outputs.append({'predict_history': history_reg & 0xFFFFFFFF})

        # Process optional mid-cycle changes AFTER sampling.
        # These changes affect the next row/edge but not the already sampled output.
        if 'mid' in row and isinstance(row['mid'], dict):
            mid = row['mid']
            for k, v in mid.items():
                if k in inputs:
                    inputs[k] = v
            # Asynchronous reset if areset is asserted mid-cycle.
            if mid.get('areset', inputs['areset']) == 1:
                history_reg = 0

    return outputs
