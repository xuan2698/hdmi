def reference(vectors):
    # state: 32-bit global branch history shift register, bit 0 youngest
    history = 0
    mask = (1 << 32) - 1
    cur = {
        'areset': 0,
        'predict_valid': 0,
        'predict_taken': 0,
        'train_mispredicted': 0,
        'train_taken': 0,
        'train_history': 0,
    }
    out = []
    keys = ('areset', 'predict_valid', 'predict_taken',
            'train_mispredicted', 'train_taken', 'train_history')

    for row in vectors:
        # Drive row inputs before the sampled posedge; mid changes from the
        # previous cycle are already in cur and are overridden here if present.
        for k in keys:
            if k in row:
                cur[k] = row[k]

        # Asynchronous reset is active while areset is high, including just
        # before the clock edge.  rule reset_async priority 0
        if cur['areset'] & 1:
            history = 0

        # Sample after the posedge has settled.
        if cur['areset'] & 1:
            # rule reset_async priority 0: output is zero
            history = 0
        else:
            if cur['train_mispredicted'] & 1:
                # rule mispredict_rollback priority 1:
                # train_history concatenated with train_taken, youngest at bit 0.
                history = (((cur['train_history'] & mask) << 1) |
                           (cur['train_taken'] & 1)) & mask
            elif cur['predict_valid'] & 1:
                # rule predict_shift priority 2:
                # shift in predict_taken from the LSB side.
                history = ((history << 1) | (cur['predict_taken'] & 1)) & mask
            else:
                # rule hold_no_update priority 3: hold previous history.
                pass

        out.append({'predict_history': history & mask})

        # Optional mid-cycle changes occur after the sample and before the
        # other edge; carry them to the start of the next row.
        mid = row.get('mid')
        if mid is not None:
            if isinstance(mid, dict):
                for k in keys:
                    if k in mid:
                        cur[k] = mid[k]
            elif hasattr(mid, 'items'):
                for k, v in mid.items():
                    if k in cur:
                        cur[k] = v
            # Asynchronous reset may also assert/deassert mid-cycle.
            # rule reset_async priority 0
            if cur['areset'] & 1:
                history = 0

    return out
