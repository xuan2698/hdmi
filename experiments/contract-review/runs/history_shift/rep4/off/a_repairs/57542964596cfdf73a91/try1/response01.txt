def reference(vectors):
    # State: 32-bit global branch history shift register.
    # bit 0 is the youngest branch direction.
    history = 0

    outputs = []

    for row in vectors:
        areset = row["areset"]
        predict_valid = row["predict_valid"]
        predict_taken = row["predict_taken"]
        train_mispredicted = row["train_mispredicted"]
        train_taken = row["train_taken"]
        train_history = row["train_history"]

        # R1_async_reset: asynchronous reset forces history to zero,
        # visible same cycle.
        if areset == 1:
            history = 0
            outputs.append({"predict_history": history & 0xFFFFFFFF})
            continue

        # Output is the current value of the history register
        # (before this cycle's posedge update).
        outputs.append({"predict_history": history & 0xFFFFFFFF})

        # Compute next state at posedge clk.
        if train_mispredicted == 1:
            # R2_mispredict_load: load history after completion of the
            # mispredicted branch = train_history concatenated with
            # train_taken (train_taken becomes the new LSB).
            history = ((train_history << 1) | (train_taken & 1)) & 0xFFFFFFFF
        elif predict_valid == 1:
            # R3_predict_shift: shift in predict_taken from LSB side.
            history = ((history << 1) | (predict_taken & 1)) & 0xFFFFFFFF
        else:
            # R4_hold: hold current value.
            history = history & 0xFFFFFFFF

    return outputs
